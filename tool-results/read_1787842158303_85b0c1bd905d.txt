   508→export function DashboardView() {
   509→  const setView = useUiStore((s) => s.setView);
   510→  const setQuickAddOpen = useUiStore((s) => s.setQuickAddOpen);
   511→  const queryClient = useQueryClient();
   512→
   513→  const { data, isLoading, isError, refetch } = useQuery({
   514→    queryKey: ["stats", "dashboard"],
   515→    queryFn: statsApi.dashboard,
   516→  });
   517→
   518→  const weekDays = React.useMemo(() => lastNDays(7), []);
   519→
   520→  // Toggle a focus todo with an optimistic update so the check feels instant.
   521→  const toggleTodo = useMutation({
   522→    mutationFn: (todo: Todo) =>
   523→      todosApi.update(todo.id, { completed: !todo.completed }),
   524→    onMutate: async (todo) => {
   525→      await queryClient.cancelQueries({ queryKey: ["stats", "dashboard"] });
   526→      const prev = queryClient.getQueryData<DashboardStats>([
   527→        "stats",
   528→        "dashboard",
   529→      ]);
   530→      if (prev) {
   531→        queryClient.setQueryData<DashboardStats>(["stats", "dashboard"], {
   532→          ...prev,
   533→          upcomingTodos: prev.upcomingTodos.map((t) =>
   534→            t.id === todo.id
   535→              ? {
   536→                  ...t,
   537→                  completed: !t.completed,
   538→                  completedAt: !t.completed
   539→                    ? new Date().toISOString()
   540→                    : null,
   541→                }
   542→              : t
   543→          ),
   544→        });
   545→      }
   546→      return { prev };
   547→    },
   548→    onError: (err, _todo, ctx) => {
   549→      if (ctx?.prev) {
   550→        queryClient.setQueryData(["stats", "dashboard"], ctx.prev);
   551→      }
   552→      toast.error(
   553→        err instanceof Error ? err.message : "Couldn't update the task"
   554→      );
   555→    },
   556→    onSettled: () => {
   557→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   558→      void queryClient.invalidateQueries({ queryKey: ["todos"] });
   559→    },
   560→  });
   561→
   562→  // Toggle a habit for today, optimistically flipping the chip + week dots.
   563→  const toggleHabit = useMutation({
   564→    mutationFn: (habit: Habit) => habitsApi.toggle(habit.id, todayKey()),
   565→    onMutate: async (habit) => {
   566→      await queryClient.cancelQueries({ queryKey: ["stats", "dashboard"] });
   567→      const prev = queryClient.getQueryData<DashboardStats>([
   568→        "stats",
   569→        "dashboard",
   570→      ]);
   571→      const today = todayKey();
   572→      if (prev) {
   573→        queryClient.setQueryData<DashboardStats>(["stats", "dashboard"], {
   574→          ...prev,
   575→          todayHabits: prev.todayHabits.map((h) => {
   576→            if (h.id !== habit.id) return h;
   577→            const done = !h.doneToday;
   578→            const logs = done
   579→              ? [...h.logs, { id: `tmp-${h.id}`, habitId: h.id, date: today }]
   580→              : h.logs.filter((l) => l.date !== today);
   581→            return { ...h, doneToday: done, logs };
   582→          }),
   583→        });
   584→      }
   585→      return { prev };
   586→    },
   587→    onError: (err, _habit, ctx) => {
   588→      if (ctx?.prev) {
   589→        queryClient.setQueryData(["stats", "dashboard"], ctx.prev);
   590→      }
   591→      toast.error(
   592→        err instanceof Error ? err.message : "Couldn't toggle the habit"
   593→      );
   594→    },
   595→    onSettled: () => {
   596→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   597→      void queryClient.invalidateQueries({ queryKey: ["habits"] });
   598→    },
   599→  });
   600→
   601→  // Onboarding: create three starter habits in one go.
   602→  const addStarterHabits = useMutation({
   603→    mutationFn: async () => {
   604→      await habitsApi.create({
   605→        name: "Drink 8 glasses of water",
   606→        emoji: "💧",
   607→        color: "teal",
   608→        timeOfDay: "anytime",
   609→      });
   610→      await habitsApi.create({
   611→        name: "Read 20 minutes",
   612→        emoji: "📚",
   613→        color: "amber",
   614→        timeOfDay: "evening",
   615→      });
   616→      await habitsApi.create({
   617→        name: "Move your body",
   618→        emoji: "🏃",
   619→        color: "rose",
   620→        timeOfDay: "morning",
   621→      });
   622→    },
   623→    onSuccess: () => {
   624→      toast.success("Starter habits added", {
   625→        description: "Find them under Routine and check them off daily.",
   626→      });
   627→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   628→      void queryClient.invalidateQueries({ queryKey: ["habits"] });
   629→    },
   630→    onError: (err) =>
   631→      toast.error(
   632→        err instanceof Error ? err.message : "Couldn't add starter habits"
   633→      ),
   634→  });
   635→
   636→  if (isLoading) return <DashboardSkeleton />;
   637→  if (isError || !data) {
   638→    return <DashboardError onRetry={() => void refetch()} />;
   639→  }
   640→
   641→  const { today, week, activeGoals, upcomingTodos, todayHabits, recentJournal, quote } =
   642→    data;
   643→  const isEmpty =
   644→    today.todosTotal === 0 &&
   645→    today.habitsTotal === 0 &&
   646→    activeGoals.length === 0 &&
   647→    today.routineTotal === 0;
   648→
   649→  if (isEmpty) {
   650→    return (
   651→      <div className="space-y-6">
   652→        <FadeIn>
   653→          <DashboardHeader score={today.score} />
   654→        </FadeIn>
   655→        <FadeIn delay={0.05}>
   656→          <OnboardingCard
   657→            habitsPending={addStarterHabits.isPending}
   658→            onAddTask={() => setQuickAddOpen(true)}
   659→            onAddHabits={() => addStarterHabits.mutate()}
   660→            onSetGoal={() => setView("goals")}
   661→          />
   662→        </FadeIn>
   663→        <FadeIn delay={0.1}>
   664→          <QuoteCard quote={quote} />
   665→        </FadeIn>
   666→      </div>
   667→    );
   668→  }
   669→
   670→  return (
   671→    <div className="space-y-6">
   672→      <FadeIn>
   673→        <DashboardHeader score={today.score} />
   674→      </FadeIn>
   675→
   676→      <FadeIn delay={0.04}>
   677→        <section aria-label="Quote of the day">
   678→          <QuoteCard quote={quote} />
   679→        </section>
   680→      </FadeIn>
   681→
   682→      <FadeIn delay={0.08}>
   683→        <StatCards today={today} />
   684→      </FadeIn>
   685→
   686→      <FadeIn delay={0.12}>
   687→        <WeekCard week={week} />
   688→      </FadeIn>
   689→
   690→      {today.overdueCount > 0 && (
   691→        <FadeIn delay={0.14}>
   692→          <OverdueBanner
   693→            count={today.overdueCount}
   694→            onReview={() => setView("tasks")}
   695→          />
   696→        </FadeIn>
   697→      )}
   698→
   699→      <FadeIn delay={0.16}>
   700→        <section aria-label="Today's focus">
   701→          <SectionHeading
   702→            title="Today's focus"
   703→            action={<SeeAllButton onClick={() => setView("tasks")} />}
   704→          />
   705→          {upcomingTodos.length === 0 ? (
   706→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   707→              Nothing due in the next week — enjoy the breathing room.
   708→            </p>
   709→          ) : (
   710→            <div className="divide-y divide-border/70 rounded-2xl border bg-card shadow-sm">
   711→              {upcomingTodos.slice(0, 6).map((todo) => (
   712→                <FocusRow
   713→                  key={todo.id}
   714→                  todo={todo}
   715→                  onToggle={(t) => toggleTodo.mutate(t)}
   716→                />
   717→              ))}
   718→            </div>
   719→          )}
   720→        </section>
   721→      </FadeIn>
   722→
   723→      <FadeIn delay={0.2}>
   724→        <section aria-label="Habits today">
   725→          <SectionHeading
   726→            title="Habits today"
   727→            action={<SeeAllButton onClick={() => setView("routine")} />}
   728→          />
   729→          {todayHabits.length === 0 ? (
   730→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   731→              No habits yet — build momentum with one small daily win.
   732→            </p>
   733→          ) : (
   734→            <div className="-mx-1 flex gap-3 overflow-x-auto px-1 pb-2 no-scrollbar">
   735→              {todayHabits.map((habit) => (
   736→                <HabitChip
   737→                  key={habit.id}
   738→                  habit={habit}
   739→                  weekDays={weekDays}
   740→                  onToggle={(h) => toggleHabit.mutate(h)}
   741→                />
   742→              ))}
   743→            </div>
   744→          )}
   745→        </section>
   746→      </FadeIn>
   747→
   748→      <FadeIn delay={0.24}>
   749→        <section aria-label="Active goals">
   750→          <SectionHeading
   751→            title="Active goals"
   752→            action={<SeeAllButton onClick={() => setView("goals")} />}
   753→          />
   754→          {activeGoals.length === 0 ? (
   755→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   756→              No active goals — set one to give your days direction.
   757→            </p>
   758→          ) : (
   759→            <div className="grid gap-3 sm:grid-cols-2">
   760→              {activeGoals.slice(0, 4).map((goal) => {
   761→                const percent = ratio(goal.progress, goal.target);
   762→                return (
   763→                  <button
   764→                    key={goal.id}
   765→                    type="button"
   766→                    onClick={() => setView("goals")}
   767→                    aria-label={`Open goal: ${goal.title}`}
   768→                    className="rounded-2xl border bg-card p-4 text-left shadow-sm transition-all hover:-translate-y-0.5 hover:border-primary/40 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50 active:scale-[0.99]"
   769→                  >
   770→                    <div className="flex items-start justify-between gap-2">
   771→                      <p className="min-w-0 text-sm font-semibold leading-snug line-clamp-1">
   772→                        {goal.title}
   773→                      </p>
   774→                      <Badge
   775→                        variant="outline"
   776→                        className="shrink-0 rounded-full px-2 py-0 text-[10px] font-medium capitalize"
   777→                      >
   778→                        {goal.period}
   779→                      </Badge>
   780→                    </div>
   781→                    <ProgressBar value={percent} className="mt-3" />
   782→                    <p className="mt-1.5 text-xs tabular-nums text-muted-foreground">
   783→                      <span className="font-semibold text-foreground">
   784→                        {goal.progress}
   785→                      </span>
   786→                      /{goal.target}
   787→                      {goal.unit ? ` ${goal.unit}` : ""} · {percent}%
   788→                    </p>
   789→                  </button>
   790→                );
   791→              })}
   792→            </div>
   793→          )}
   794→        </section>
   795→      </FadeIn>
   796→
   797→      <FadeIn delay={0.28}>
   798→        <section aria-label="Recent journal">
   799→          <SectionHeading
   800→            title="Recent journal"
   801→            action={
   802→              <SeeAllButton
   803→                label="Open diary"
   804→                onClick={() => setView("diary")}
   805→              />
   806→            }
   807→          />
   808→          {recentJournal.length === 0 ? (
   809→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   810→              No entries yet — tonight is a good night to write.
   811→            </p>
   812→          ) : (
   813→            <div className="divide-y divide-border/70 rounded-2xl border bg-card shadow-sm">
   814→              {recentJournal.map((entry) => {
   815→                const mood = MOODS.find((m) => m.value === entry.mood);
   816→                return (
   817→                  <button
   818→                    key={entry.id}
   819→                    type="button"
   820→                    onClick={() => setView("diary")}
   821→                    aria-label={`Open diary entry from ${friendlyDay(entry.date)}`}
   822→                    className="flex w-full items-start gap-3 px-4 py-3 text-left transition-colors hover:bg-muted/40"
   823→                  >
   824→                    <span
   825→                      className="mt-0.5 flex size-9 shrink-0 items-center justify-center rounded-xl bg-muted text-base"
   826→                      aria-hidden="true"
   827→                    >
   828→                      {mood ? (
   829→                        mood.emoji
   830→                      ) : (
   831→                        <BookOpen className="size-4 text-muted-foreground" />
   832→                      )}
   833→                    </span>
   834→                    <span className="min-w-0 flex-1">
   835→                      <span className="flex items-baseline justify-between gap-2">
   836→                        <span className="truncate text-sm font-semibold">
   837→                          {entry.title || "Journal entry"}
   838→                        </span>
   839→                        <span className="shrink-0 text-[11px] text-muted-foreground">
   840→                          {friendlyDay(entry.date)}
   841→                        </span>
   842→                      </span>
   843→                      <span className="mt-0.5 block line-clamp-2 text-xs leading-relaxed text-muted-foreground">
   844→                        {entry.content}
   845→                      </span>
   846→                    </span>
   847→                  </button>
   848→                );
   849→              })}
   850→            </div>
   851→          )}
   852→        </section>
   853→      </FadeIn>
   854→    </div>
   855→  );
   856→}
