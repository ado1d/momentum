    45→    45→/** 65 → "1h 5m", 45 → "45m", 120 → "2h" */
    46→    46→function formatMinutes(total: number): string {
    47→    47→  if (total <= 0) return "0m";
    48→    48→  const h = Math.floor(total / 60);
    49→    49→  const m = total % 60;
    50→    50→  if (h === 0) return `${m}m`;
    51→    51→  if (m === 0) return `${h}h`;
    52→    52→  return `${h}h ${m}m`;
    53→    53→}

    78→    78→/** Catmull-Rom → cubic bezier smoothing, clamped to the 0..40 plot band. */
    79→    79→function smoothPath(pts: { x: number; y: number }[]): string {
    80→    80→  if (pts.length === 0) return "";
    81→    81→  if (pts.length === 1) return `M ${pts[0].x.toFixed(2)} ${pts[0].y.toFixed(2)}`;
    82→    82→  const clamp = (v: number) => Math.min(40, Math.max(0, v));
    83→    83→  let d = `M ${pts[0].x.toFixed(2)} ${pts[0].y.toFixed(2)}`;
    84→    84→  for (let i = 0; i < pts.length - 1; i += 1) {
    85→    85→    const p0 = pts[i - 1] ?? pts[i];
    86→    86→    const p1 = pts[i];
    87→    87→    const p2 = pts[i + 1];
    88→    88→    const p3 = pts[i + 2] ?? p2;
    89→    89→    const c1x = p1.x + (p2.x - p0.x) / 6;
    90→    90→    const c1y = clamp(p1.y + (p2.y - p0.y) / 6);
    91→    91→    const c2x = p2.x - (p3.x - p1.x) / 6;
    92→    92→    const c2y = clamp(p2.y - (p3.y - p1.y) / 6);
    93→    93→    d += ` C ${c1x.toFixed(2)} ${c1y.toFixed(2)}, ${c2x.toFixed(2)} ${c2y.toFixed(2)}, ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`;
    94→    94→  }
    95→    95→  return d;
    96→    96→}

   257→   257→function HeatmapCard({ heatmap }: { heatmap: DayStat[] }) {
   258→   258→  const today = todayKey();
   259→   259→
   260→   260→  // Pad the first column so day 1 of the data lands on its Monday-based row.
   261→   261→  const weeks = React.useMemo(() => {
   262→   262→    if (heatmap.length === 0) return [] as (DayStat | null)[][];
   263→   263→    const pad = weekdayOfKey(heatmap[0].date) - 1; // Mon=1 → 0 padding
   264→   264→    const cells: (DayStat | null)[] = [
   265→   265→      ...Array.from({ length: pad }, () => null),
   266→   266→      ...heatmap,
   267→   267→    ];
   268→   268→    const out: (DayStat | null)[][] = [];
   269→   269→    for (let i = 0; i < cells.length; i += 7) out.push(cells.slice(i, i + 7));
   270→   270→    return out;
   271→   271→  }, [heatmap]);
   272→   272→
   273→   273→  // Tiny month label above a column whenever the month changes.
   274→   274→  const monthLabels = React.useMemo(() => {
   275→   275→    let last = -1;
   276→   276→    return weeks.map((week) => {
   277→   277→      const first = week.find((c): c is DayStat => c !== null);
   278→   278→      if (!first) return "";
   279→   279→      const d = keyToDate(first.date);
   280→   280→      if (!isValid(d)) return "";
   281→   281→      const m = d.getFullYear() * 12 + d.getMonth();
   282→   282→      if (m !== last) {
   283→   283→        last = m;
   284→   284→        return format(d, "MMM");
   285→   285→      }
   286→   286→      return "";
   287→   287→    });
   288→   288→  }, [weeks]);
   289→   289→
   290→   290→  const { longestStreak, activeDays, avgScore } = React.useMemo(() => {
   291→   291→    let streak = 0;
   292→   292→    let best = 0;
   293→   293→    let active = 0;
   294→   294→    let sum = 0;
   295→   295→    for (const d of heatmap) {
   296→   296→      if (d.score > 0) {
   297→   297→        streak += 1;
   298→   298→        active += 1;
   299→   299→        if (streak > best) best = streak;
   300→   300→      } else {
   301→   301→        streak = 0;
   302→   302→      }
   303→   303→      sum += d.score;
   304→   304→    }
   305→   305→    return {
   306→   306→      longestStreak: best,
   307→   307→      activeDays: active,
   308→   308→      avgScore: heatmap.length ? Math.round(sum / heatmap.length) : 0,
   309→   309→    };
   310→   310→  }, [heatmap]);
   311→   311→
   312→   312→  const allEmpty = heatmap.length > 0 && heatmap.every((d) => d.score <= 0);

   315→   315→  return (
   316→   316→    <Card className="gap-4 rounded-2xl p-4 sm:p-5">
   317→   317→      <CardHead
   318→   318→        title="Activity"
   319→   319→        subtitle="Last 12 weeks"
   320→   320→        action={
   321→   321→          <div
   322→   322→            className="flex items-center gap-1 text-[10px] font-medium text-muted-foreground"
   323→   323→            aria-hidden="true"
   324→   324→          >
   325→   325→            Less
   326→   326→            {[
   327→   327→              "bg-muted",
   328→   328→              "bg-emerald-500/25",
   329→   329→              "bg-emerald-500/45",

   334→   334→            ))}
   335→   335→            More
   336→   336→          </div>

   338→   338→      />
   339→   339→
   340→   340→      <div className="flex flex-col gap-5 sm:flex-row sm:items-center sm:justify-between">
   341→   341→        <div
   342→   342→          className="no-scrollbar overflow-x-auto"
   343→   343→          role="img"
   344→   344→          aria-label={`Activity heatmap for the last 12 weeks — ${activeDays} of ${totalDays} days active, average score ${avgScore}.`}
   345→   345→        >
   346→   346→          <div className="flex w-max gap-1">
   347→   347→            {/* Weekday hints (M / W / F) */}

   420→   420→// ── Tasks trend (pure inline SVG line/area chart) ────────────
   421→   421→
   422→   422→function TasksTrendCard({ trend }: { trend: TrendPoint[] }) {
   423→   423→  const n = trend.length;
   424→   424→
   425→   425→  const { max, total, best, avg } = React.useMemo(() => {
   426→   426→    const counts = trend.map((p) => p.count);
   427→   427→    const t = counts.reduce((a, b) => a + b, 0);
   428→   428→    // Round odd maxima up so the mid gridline/label stays an integer.
   429→   429→    const m = Math.max(1, ...counts);
   430→   430→    return {
   431→   431→      max: m >= 3 && m % 2 === 1 ? m + 1 : m,
   432→   432→      total: t,
   433→   433→      best: counts.length ? Math.max(...counts) : 0,
   434→   434→      avg: counts.length ? t / counts.length : 0,
   435→   435→    };
   436→   436→  }, [trend]);
   437→   437→
   438→   438→  // Points live in a 100×40 viewBox; the SVG stretches with
   439→   439→  // preserveAspectRatio="none" and strokes use vector-effect so they
   440→   440→  // stay crisp. Text (axis labels) is rendered as HTML so it never distorts.
   441→   441→  const points = React.useMemo(
   442→   442→    () =>
   443→   443→      trend.map((p, i) => ({
   444→   444→        ...p,
   445→   445→        x: ((i + 0.5) * 100) / n,
   446→   446→        y: 40 - (p.count / max) * 40,
   447→   447→        yPct: (1 - p.count / max) * 100,
   448→   448→      })),
   449→   449→    [trend, n, max]
   450→   450→  );
   451→   451→
   452→   452→  const linePath = React.useMemo(() => smoothPath(points), [points]);
   453→   453→  const areaPath = React.useMemo(() => {
   454→   454→    if (points.length < 2) return "";
   455→   455→    const last = points[points.length - 1];
   456→   456→    const first = points[0];
   457→   457→    return `${linePath} L ${last.x.toFixed(2)} 40 L ${first.x.toFixed(2)} 40 Z`;
   458→   458→  }, [linePath, points]);
   459→   459→
   460→   460→  const xLabelIdx = React.useMemo(() => {
   461→   461→    if (n === 0) return [] as number[];

   688→   688→function MoodCard({ distribution }: { distribution: InsightsData["moodDistribution"] }) {
   689→   689→  const total = distribution.reduce((a, d) => a + d.count, 0);
   690→   690→
   691→   691→  // Render in a fixed best→rough order, only moods that were recorded.
   692→   692→  const rows = MOODS.map((m) => {
   693→   693→    const entry = distribution.find((d) => d.mood === m.value);
   694→   694→    return entry ? { ...m, count: entry.count } : null;
   695→   695→  }).filter((r): r is { value: Mood; label: string; emoji: string; count: number } => r !== null);
   696→   696→
   697→   697→  const pctOf = (count: number) => (total > 0 ? Math.round((count / total) * 100) : 0);
   698→   698→
   699→   699→  return (
   700→   700→    <Card className="gap-4 rounded-2xl p-4 sm:p-5">
   701→   701→      <CardHead title="Mood over time" subtitle="From your diary entries" />
   702→   702→
   703→   703→      <div
   704→   704→        className="flex h-3 w-full overflow-hidden rounded-full bg-muted"
   705→   705→        role="img"
   706→   706→        aria-label={`Mood distribution across ${total} diary ${total === 1 ? "entry" : "entries"}: ${rows
   707→   707→          .map((r) => `${r.label} ${pctOf(r.count)}%`)
   708→   708→          .join(", ")}.`}
   709→   709→      >
   710→   710→        {rows.map((r) => (
   711→   711→          <div
   712→   712→            key={r.value}
   713→   713→            className={moodBarStyles[r.value]}
   714→   714→            style={{ width: `${pctOf(r.count)}%` }}
   715→   715→            title={`${r.label} · ${r.count} ${r.count === 1 ? "entry" : "entries"} · ${pctOf(r.count)}%`}
   716→   716→          />
   717→   717→        ))}
   718→   718→      </div>
   719→   719→
   720→   720→      <div className="grid grid-cols-1 gap-x-6 gap-y-2 sm:grid-cols-2">
