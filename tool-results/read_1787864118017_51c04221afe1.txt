   145→// ── Backup import (restore) ─────────────────────────────────
   146→
   147→const MAX_BACKUP_BYTES = 20 * 1024 * 1024; // sanity cap: 20 MB
   148→
   149→// A file counts as a plausible backup when at least one of these is an array.
   150→const BACKUP_ARRAY_KEYS = [
   151→  "todos",
   152→  "subtasks",
   153→  "habits",
   154→  "routineTasks",
   155→  "notes",
   156→  "journal",
   157→  "goals",
   158→] as const;
   159→
   160→interface ParsedBackup {
   161→  fileName: string;
   162→  sizeLabel: string;
   163→  exportedAt: string | null; // pre-formatted friendly label
   164→  data: Record<string, unknown>;
   165→  preview: { label: string; count: number }[];
   166→}
   167→
   168→function isRecord(value: unknown): value is Record<string, unknown> {
   169→  return typeof value === "object" && value !== null && !Array.isArray(value);
   170→}
   171→
   172→function arrayCount(data: Record<string, unknown>, key: string): number {
   173→  const value = data[key];
   174→  return Array.isArray(value) ? value.length : 0;
   175→}
   176→
   177→function formatBytes(bytes: number): string {
   178→  if (bytes < 1024) return `${bytes} B`;
   179→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   180→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   181→}
   182→
   183→function friendlyTimestamp(value: unknown): string | null {
   184→  if (typeof value !== "string" || value.length === 0) return null;
   185→  const date = new Date(value);
   186→  if (Number.isNaN(date.getTime())) return null;
   187→  return date.toLocaleString(undefined, { dateStyle: "medium", timeStyle: "short" });
   188→}
   189→
   190→function pluralize(count: number, one: string, many: string): string {
   191→  return `${count} ${count === 1 ? one : many}`;
   192→}
   193→
   194→function formatImportCounts(counts: ImportCounts): string {
   195→  const parts = [
   196→    pluralize(counts.todos, "task", "tasks"),
   197→    pluralize(counts.habits, "habit", "habits"),
   198→  ];
   199→  if (counts.routineTasks > 0) {
   200→    parts.push(pluralize(counts.routineTasks, "routine task", "routine tasks"));
   201→  }
   202→  parts.push(
   203→    pluralize(counts.notes, "note", "notes"),
   204→    pluralize(counts.journal, "journal entry", "journal entries"),
   205→    pluralize(counts.goals, "goal", "goals"),
   206→  );
   207→  parts.push(`${counts.skipped} skipped`);
   208→  return parts.join(" · ");
   209→}

   211→function ImportSection() {
   212→  const queryClient = useQueryClient();
   213→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   214→
   215→  const [parsed, setParsed] = React.useState<ParsedBackup | null>(null);
   216→  const [mode, setMode] = React.useState<"merge" | "replace">("merge");
   217→  const [replaceConfirmed, setReplaceConfirmed] = React.useState(false);
   218→  const [dragging, setDragging] = React.useState(false);
   219→  const [result, setResult] = React.useState<ImportResult | null>(null);
   220→
   221→  const importBackup = useMutation({
   222→    mutationFn: () => {
   223→      if (!parsed) throw new Error("Choose a backup file first");
   224→      return importApi.restore(parsed.data, mode);
   225→    },
   226→    onSuccess: (res) => {
   227→      toast.success(res.message);
   228→      setResult(res);
   229→      // Reset the pick + confirmation; the success panel stays visible
   230→      // until another file is chosen or it is dismissed.
   231→      setParsed(null);
   232→      setMode("merge");
   233→      setReplaceConfirmed(false);
   234→      // An import can touch every feature (todos, subtasks, habits, routine,
   235→      // notes, journal, goals, settings — plus derived stats / insights /
   236→      // focus / review / search), so invalidate ALL queries — the robust
   237→      // and simplest option after a full restore.
   238→      void queryClient.invalidateQueries();
   239→    },
   240→    onError: (error) => {
   241→      toast.error(error.message || "Import failed — please try again");
   242→    },
   243→  });
   244→
   245→  const isImporting = importBackup.isPending;
   246→
   247→  const openPicker = () => {
   248→    if (!isImporting) fileInputRef.current?.click();
   249→  };
   250→
   251→  const handleFile = (file: File | undefined | null) => {
   252→    if (!file || isImporting) return;
   253→    setResult(null); // choosing a new file clears the previous outcome
   254→    if (file.size > MAX_BACKUP_BYTES) {
   255→      toast.error("File too large — backups up to 20 MB are supported");
   256→      return;
   257→    }
   258→    const reader = new FileReader();
   259→    reader.onerror = () => toast.error("Could not read the file — try again");
   260→    reader.onload = () => {
   261→      let json: unknown;
   262→      try {
   263→        json = JSON.parse(String(reader.result ?? ""));
   264→      } catch {
   265→        toast.error("Invalid JSON file");
   266→        return;
   267→      }
   268→      if (
   269→        !isRecord(json) ||
   270→        !BACKUP_ARRAY_KEYS.some((key) => Array.isArray(json[key]))
   271→      ) {
   272→        toast.error("Not a Momentum backup file");
   273→        return;
   274→      }
   275→      setParsed({
   276→        fileName: file.name,
   277→        sizeLabel: formatBytes(file.size),
   278→        exportedAt: friendlyTimestamp(json.exportedAt),
   279→        data: json,
   280→        preview: [
   281→          { label: "tasks", count: arrayCount(json, "todos") },
   282→          { label: "subtasks", count: arrayCount(json, "subtasks") },
   283→          { label: "habits", count: arrayCount(json, "habits") },
   284→          { label: "routine", count: arrayCount(json, "routineTasks") },
   285→          { label: "notes", count: arrayCount(json, "notes") },
   286→          { label: "journal", count: arrayCount(json, "journal") },
   287→          { label: "goals", count: arrayCount(json, "goals") },
   288→        ].filter((p) => p.count > 0),
   289→      });
   290→      setMode("merge");
   291→      setReplaceConfirmed(false);
   292→    };
   293→    reader.readAsText(file);
   294→  };
   295→
   296→  const clearFile = () => {
   297→    setParsed(null);
   298→    setMode("merge");
   299→    setReplaceConfirmed(false);
   300→  };
   301→
   302→  const canImport =
   303→    parsed !== null && !isImporting && (mode === "merge" || replaceConfirmed);
   304→
   305→  return (
   306→    <Card className="rounded-2xl py-0 shadow-card">
   307→      <CardHeader className="p-4 pb-2 sm:p-6 sm:pb-3">
   308→        <CardTitle className="flex items-center gap-2 text-base">
   309→          <Upload className="size-4.5 text-primary" aria-hidden="true" />
   310→          Import &amp; restore
   311→        </CardTitle>
   312→        <CardDescription>
   313→          Restore a Momentum JSON backup — merge it in, or replace everything.
   314→        </CardDescription>
   315→      </CardHeader>
   316→      <CardContent className="space-y-4 p-4 pt-2 sm:p-6 sm:pt-2">
   317→        {/* Drop / pick zone */}
   318→        <div
   319→          role="button"
   320→          tabIndex={0}
   321→          aria-label="Choose a backup file, or drop one here"
   322→          aria-disabled={isImporting}
   323→          onClick={openPicker}
   324→          onKeyDown={(e) => {
   325→            if (e.key === "Enter" || e.key === " ") {
   326→              e.preventDefault();
   327→              openPicker();
   328→            }
   329→          }}
   330→          onDragOver={(e) => {
   331→            e.preventDefault();
   332→            setDragging(true);
   333→          }}
   334→          onDragLeave={() => setDragging(false)}
   335→          onDrop={(e) => {
   336→            e.preventDefault();
   337→            setDragging(false);
   338→            handleFile(e.dataTransfer.files?.[0]);
   339→          }}
   340→          className={cn(
   341→            "flex cursor-pointer flex-col items-center gap-1.5 rounded-2xl border-2 border-dashed px-4 py-7 text-center transition-colors",
   342→            "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50 hover:border-primary/50 hover:bg-primary/5",
   343→            dragging && "border-primary bg-primary/5",
   344→            isImporting && "pointer-events-none opacity-60"
   345→          )}
   346→        >
   347→          <FileJson className="size-8 text-primary" aria-hidden="true" />
   348→          <p className="mt-1 text-sm font-semibold">
   349→            {parsed ? "Choose a different backup" : "Drag & drop a backup file"}
   350→          </p>
   351→          <p className="text-xs text-muted-foreground">or</p>
   352→          <span className="inline-flex h-9 items-center rounded-xl bg-primary px-4 text-sm font-semibold text-primary-foreground shadow-sm">
   353→            Choose backup file
   354→          </span>
   355→          <p className="mt-1.5 text-[11px] text-muted-foreground/80">
   356→            JSON backups from “Backup (JSON)” · up to 20 MB
   357→          </p>
   358→          <input
   359→            ref={fileInputRef}
   360→            type="file"
   361→            accept=".json,application/json"
   362→            className="hidden"
   363→            onChange={(e) => {
   364→              const file = e.target.files?.[0];
   365→              e.target.value = ""; // allow re-picking the same file
   366→              handleFile(file);
   367→            }}
   368→          />
   369→        </div>
   370→
   371→        {/* Parsed backup preview */}
   372→        {parsed && (
   373→          <div className="rounded-xl border bg-muted/30 p-3.5">
   374→            <div className="flex items-start gap-3">
   375→              <FileJson
   376→                className="mt-0.5 size-4.5 shrink-0 text-primary"
   377→                aria-hidden="true"
   378→              />
   379→              <div className="min-w-0 flex-1">
   380→                <p className="truncate text-sm font-semibold" title={parsed.fileName}>
   381→                  {parsed.fileName}
   382→                </p>
   383→                <p className="mt-0.5 text-xs text-muted-foreground">
   384→                  {parsed.exportedAt ? `Exported ${parsed.exportedAt} · ` : ""}
   385→                  {parsed.sizeLabel}
   386→                </p>
   387→              </div>
   388→              <Button
   389→                type="button"
   390→                variant="ghost"
   391→                size="icon"
   392→                className="size-7 shrink-0 rounded-lg text-muted-foreground"
   393→                onClick={clearFile}
   394→                aria-label="Remove selected backup file"
   395→              >
   396→                <X className="size-4" aria-hidden="true" />
   397→              </Button>
   398→            </div>
   399→            <div className="mt-2.5 flex flex-wrap gap-1.5">
   400→              {parsed.preview.length > 0 ? (
   401→                parsed.preview.map((p) => (
   402→                  <Badge
   403→                    key={p.label}
   404→                    variant="secondary"
   405→                    className="rounded-full px-2.5 py-0.5 text-[11px] font-medium"
   406→                  >
   407→                    {p.count} {p.label}
   408→                  </Badge>
   409→                ))
   410→              ) : (
   411→                <p className="text-xs text-muted-foreground">
   412→                  No items found in this backup.
   413→                </p>
   414→              )}
   415→            </div>
   416→          </div>
   417→        )}
   418→
   419→        {/* Mode selector */}
   420→        <RadioGroup
   421→          value={mode}
   422→          onValueChange={(v) => {
   423→            setMode(v === "replace" ? "replace" : "merge");
   424→            setReplaceConfirmed(false);
   425→          }}
   426→          disabled={isImporting}
   427→          className="grid gap-2.5"
   428→          aria-label="Import mode"
   429→        >
   430→          <label
   431→            htmlFor="import-mode-merge"
   432→            className={cn(
   433→              "flex cursor-pointer items-start gap-3 rounded-xl border p-3.5 transition-colors",
   434→              mode === "merge"
   435→                ? "border-emerald-500/60 bg-emerald-500/5"
   436→                : "hover:bg-muted/40"
   437→            )}
   438→          >
   439→            <RadioGroupItem value="merge" id="import-mode-merge" className="mt-0.5" />
   440→            <span className="min-w-0 flex-1">
   441→              <span className="flex flex-wrap items-center gap-x-2 gap-y-1 text-sm font-semibold">
   442→                <Merge
   443→                  className="size-4 text-emerald-600 dark:text-emerald-400"
   444→                  aria-hidden="true"
   445→                />
   446→                Merge
   447→                <span className="rounded-full bg-emerald-500/10 px-2 py-0 text-[10px] font-semibold uppercase tracking-wide text-emerald-600 dark:text-emerald-400">
   448→                  Safe
   449→                </span>
   450→              </span>
   451→              <span className="mt-1 block text-xs leading-relaxed text-muted-foreground">
   452→                Add only items that don&apos;t already exist. Nothing you have
   453→                today is changed or overwritten.
   454→              </span>
   455→            </span>
   456→          </label>
   457→
   458→          <label
   459→            htmlFor="import-mode-replace"
   460→            className={cn(
   461→              "flex cursor-pointer items-start gap-3 rounded-xl border p-3.5 transition-colors",
   462→              mode === "replace"
   463→                ? "border-destructive/60 bg-destructive/5"
   464→                : "hover:bg-muted/40"
   465→            )}
   466→          >
   467→            <RadioGroupItem value="replace" id="import-mode-replace" className="mt-0.5" />
   468→            <span className="min-w-0 flex-1">
   469→              <span className="flex flex-wrap items-center gap-x-2 gap-y-1 text-sm font-semibold">
   470→                <TriangleAlert
   471→                  className="size-4 text-rose-600 dark:text-rose-400"

   474→                Replace
   475→              </span>
   476→              <span className="mt-1 block text-xs leading-relaxed text-muted-foreground">
   477→                Delete{" "}
   478→                <strong className="font-semibold text-foreground">
   479→                  all current data
   480→                </strong>{" "}
   481→                and restore this backup exactly as it was. This cannot be undone.
   482→              </span>
   483→            </span>
   484→          </label>
   485→        </RadioGroup>
   486→
   487→        {/* Extra explicit confirmation for the destructive mode */}
   488→        {mode === "replace" && (
   489→          <div className="flex items-start gap-3 rounded-xl border border-destructive/40 bg-destructive/5 p-3.5">

   491→              id="import-replace-confirm"
   492→              checked={replaceConfirmed}
   493→              onCheckedChange={(checked) => setReplaceConfirmed(checked === true)}
   494→              disabled={isImporting}
   495→              className="mt-0.5"
   496→            />
   497→            <Label
   498→              htmlFor="import-replace-confirm"
   499→              className="cursor-pointer text-xs font-normal leading-relaxed text-destructive"
   500→            >
   501→              I understand{" "}
   502→              <strong className="font-semibold">
   503→                all current data will be deleted
   504→              </strong>{" "}
   505→              — tasks, habits, routine, notes, journal and goals — before this
   506→              backup is restored.
   507→            </Label>
   508→          </div>
   509→        )}
   510→
   511→        {/* Primary action */}
   512→        <Button
   513→          type="button"
   514→          size="lg"
   515→          variant={mode === "replace" ? "destructive" : "default"}
   516→          onClick={() => importBackup.mutate()}
   517→          disabled={!canImport}
   518→          className="w-full rounded-xl"
   519→          aria-label={
   520→            mode === "replace"
   521→              ? "Delete all data and restore this backup"
   522→              : "Import backup"
   523→          }
   524→        >
   525→          {isImporting ? (
   526→            <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   527→          ) : (
   528→            <Upload className="size-4" aria-hidden="true" />
   529→          )}
   530→          {isImporting
   531→            ? "Importing…"

   533→              ? "Delete everything & restore backup"
   534→              : "Import backup"}
   535→        </Button>
   536→
   537→        {/* Success panel */}
   538→        {result && (
   539→          <div
   540→            className="rounded-xl border border-emerald-500/40 bg-emerald-500/5 p-3.5"
   541→            role="status"
   542→          >
   543→            <div className="flex items-start gap-3">
   544→              <CheckCircle2
   545→                className="mt-0.5 size-4.5 shrink-0 text-emerald-600 dark:text-emerald-400"
   546→                aria-hidden="true"
   547→              />
   548→              <div className="min-w-0 flex-1">

   550→                  {result.mode === "replace" ? "Backup restored" : "Backup merged"}

   588→  const queryClient = useQueryClient();
   589→
   590→  const { data: settings, isLoading, isError, refetch } = useQuery({
   591→    queryKey: ["settings"],
   592→    queryFn: settingsApi.get,
   593→  });
   594→
   595→  const [permission, setPermission] = React.useState<
   596→    NotificationPermission | "unsupported" | "unknown"
   597→  >("unknown");
   598→  const [exporting, setExporting] = React.useState<string | null>(null);
   599→
   600→  React.useEffect(() => {
   601→    setPermission(notificationPermission());
   602→  }, []);
   603→
   604→  // ── Mutations ──
   605→
   606→  const updateSettings = useMutation({
   607→    mutationFn: (vars: { patch: Partial<AppSettings>; message?: string }) =>
   608→      settingsApi.update(vars.patch),
   609→    onMutate: async ({ patch }) => {
   610→      await queryClient.cancelQueries({ queryKey: ["settings"] });
   611→      const previous = queryClient.getQueryData<AppSettings>(["settings"]);
   612→      if (previous) {
   613→        queryClient.setQueryData<AppSettings>(["settings"], { ...previous, ...patch });
   614→      }
   615→      return { previous };
   616→    },
   617→    onError: (error, _vars, context) => {
   618→      if (context?.previous) {
   619→        queryClient.setQueryData(["settings"], context.previous);
   620→      }
   621→      toast.error(error.message || "Could not save setting");
   622→    },
   623→    onSuccess: (_result, vars) => {
   624→      toast.success(vars.message ?? "Setting saved");
   625→    },
   626→    onSettled: () => {
   627→      void queryClient.invalidateQueries({ queryKey: ["settings"] });
   628→    },
   629→  });
   630→
   631→  // ── Handlers ──
   632→
   633→  const handleNotificationsToggle = async (checked: boolean) => {
   634→    if (!checked) {
   635→      updateSettings.mutate({
   636→        patch: { notificationsEnabled: false },
   637→        message: "Notifications disabled",
   638→      });
   639→      return;
   640→    }
   641→    const result = await requestNotificationPermission();
   642→    setPermission(result);
   643→    if (result !== "granted") {
   644→      toast.error(
   645→        result === "unsupported"
   646→          ? "Notifications aren't supported in this browser"
   647→          : "Notification permission was denied — enable it in your browser settings"
   648→      );
   649→      return;
   650→    }
   651→    updateSettings.mutate({
   652→      patch: { notificationsEnabled: true },
   653→      message: "Notifications enabled",
   654→    });

   663→    const toastId = toast.loading(`Preparing ${label}…`);
   664→    try {
   665→      await fn();

   667→    } catch (e) {
   668→      toast.error(e instanceof Error ? e.message : `${label} failed`, { id: toastId });
   669→    } finally {
   670→      setExporting(null);
   671→    }
   672→  };
   673→
   674→  const exportAllMarkdown = () =>
   675→    runExport("all-md", "Markdown export", async () => {
   676→      const md = await exportApi.markdown("all");
   677→      downloadMarkdown(md, "momentum-export.md");

   679→
   680→  const backupJson = () =>

   692→  const exportDiary = () =>

   695→      downloadMarkdown(md, "momentum-diary.md");

   707→      downloadMarkdown(md, "momentum-notes.md");

   714→        ? "granted"
   715→        : permission === "denied"

   717→          : permission === "unsupported"
