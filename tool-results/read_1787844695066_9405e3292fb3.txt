   135→/** Local calendar-day key of a due/completed ISO timestamp (null when unset). */
   136→function isoDayKey(iso: string | null): string | null {
   137→  if (!iso) return null;
   138→  const d = new Date(iso);
   139→  return Number.isNaN(d.getTime()) ? null : dateToKey(d);
   140→}
   141→
   142→function dueDayKey(todo: Todo): string | null {
   143→  return isoDayKey(todo.dueDate);
   144→}
   145→
   146→/** Turn a local day key (+ optional "HH:MM") into a real ISO instant. */
   147→function localDateKeyToIso(key: string, time = "00:00"): string {
   148→  const [y, mo, d] = key.split("-").map(Number);
   149→  const [h, mi] = time.split(":").map(Number);
   150→  return new Date(y, (mo ?? 1) - 1, d ?? 1, h ?? 0, mi ?? 0).toISOString();
   151→}
   152→
   153→function toTimeInputValue(d: Date): string {
   154→  const pad = (n: number) => String(n).padStart(2, "0");
   155→  return `${pad(d.getHours())}:${pad(d.getMinutes())}`;
   156→}
   157→
   158→/** ISO instant -> "YYYY-MM-DDTHH:MM" (local) for datetime-local inputs. */
   159→function isoToLocalInputValue(iso: string): string {
   160→  const d = new Date(iso);
   161→  if (Number.isNaN(d.getTime())) return "";
   162→  const pad = (n: number) => String(n).padStart(2, "0");
   163→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
   164→}
   165→
   166→function matchesTab(todo: Todo, tab: TabValue, today: string): boolean {
   167→  const key = dueDayKey(todo);
   168→  switch (tab) {
   169→    case "all":
   170→      return true;
   171→    case "today":
   172→      return todo.completed
   173→        ? isoDayKey(todo.completedAt) === today
   174→        : key === today;
   175→    case "upcoming":
   176→      return !todo.completed && !!key && key > today;
   177→    case "overdue":
   178→      return !todo.completed && !!key && key < today;
   179→    case "completed":
   180→      return todo.completed;
   181→  }
   182→}

   383→function EditTaskDialog({
   384→  todo,
   385→  saving,
   386→  onCancel,
   387→  onSave,
   388→}: {
   389→  todo: Todo;
   390→  saving: boolean;
   391→  onCancel: () => void;
   392→  onSave: (patch: Partial<TodoInput>) => void;
   393→}) {
   394→  const [title, setTitle] = React.useState(todo.title);
   395→  const [notes, setNotes] = React.useState(todo.notes ?? "");
   396→  const [priority, setPriority] = React.useState<Priority>(todo.priority);
   397→  const [category, setCategory] = React.useState(todo.category);
   398→  const [dueDate, setDueDate] = React.useState(() => {
   399→    if (!todo.dueDate) return "";
   400→    const d = new Date(todo.dueDate);
   401→    return Number.isNaN(d.getTime()) ? "" : dateToKey(d);
   402→  });
   403→  const [dueTime, setDueTime] = React.useState(() => {
   404→    if (!todo.dueDate || !hasTime(todo.dueDate)) return "";
   405→    const d = new Date(todo.dueDate);
   406→    return Number.isNaN(d.getTime()) ? "" : toTimeInputValue(d);
   407→  });
   408→  const [reminder, setReminder] = React.useState(() =>
   409→    todo.reminderAt ? isoToLocalInputValue(todo.reminderAt) : ""
   410→  );
   411→
   412→  const save = () => {
   413→    const trimmed = title.trim();
   414→    if (!trimmed) {
   415→      toast.error("Give the task a title");
   416→      return;
   417→    }
   418→    const dueIso = dueDate
   419→      ? localDateKeyToIso(dueDate, dueTime || "00:00")
   420→      : null;
   421→    const reminderIso = reminder ? new Date(reminder).toISOString() : null;
   422→    onSave({
   423→      title: trimmed,
   424→      notes: notes.trim() || null,
   425→      priority,
   426→      category,
   427→      dueDate: dueIso,
   428→      reminderAt: reminderIso,
   429→    });
   430→  };
   431→
   432→  const labelClass = "text-xs font-medium text-muted-foreground";
   433→
   434→  return (
   435→    <Dialog
   436→      open
   437→      onOpenChange={(open) => {
   438→        if (!open) onCancel();
   439→      }}
   440→    >
   441→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   442→        <DialogHeader>
   443→          <DialogTitle>Edit task</DialogTitle>
   444→          <DialogDescription>

   448→
   449→        <div className="space-y-4">

   451→            <label htmlFor="edit-title" className={labelClass}>

   453→            </label>

   455→              id="edit-title"

   460→          </div>
   461→
   462→          <div className="space-y-1.5">
   463→            <label htmlFor="edit-notes" className={labelClass}>
   464→              Notes
   465→            </label>

   590→  });

   674→    },

   676→      if (ctx?.prev) {
   677→        queryClient.setQueryData(["todos", "full"], ctx.prev);
   678→      }
   679→      toast.error("Couldn't delete the task");
   680→    },

   709→  const filtered = React.useMemo(

   711→      todos.filter(

   724→  const completedTodos = React.useMemo(

   742→      upcoming: [],
   743→      nodate: [],

   745→    for (const t of activeTodos) {
   746→      const key = dueDayKey(t);
   747→      if (key === null) buckets.nodate.push(t);
   748→      else if (key < today) buckets.overdue.push(t);
   749→      else if (key === today) buckets.today.push(t);
   750→      else if (key === tomorrow) buckets.tomorrow.push(t);
   751→      else buckets.upcoming.push(t);

   754→      { id: "overdue", label: "Overdue", tone: "danger" as const, items: buckets.overdue },
   755→      { id: "today", label: "Today", tone: "default" as const, items: buckets.today },
   756→      { id: "tomorrow", label: "Tomorrow", tone: "default" as const, items: buckets.tomorrow },
   757→      { id: "upcoming", label: "Upcoming", tone: "default" as const, items: buckets.upcoming },
   758→      { id: "nodate", label: "No date", tone: "default" as const, items: buckets.nodate },

   761→
   762→  const submitQuickAdd = () => {

   765→      quickAddRef.current?.focus();

   770→      quickDue === "none"

   772→        : localDateKeyToIso(quickDue === "today" ? today : tomorrow);

   774→      { title, priority: quickPriority, dueDate },
   775→      {
   776→        onSuccess: () => {
   777→          setQuickTitle("");
   778→          quickAddRef.current?.focus();
   779→        },
   780→      }

   782→  };
   783→
   784→  const cyclePriority = () =>
   785→    setQuickPriority(
   786→      (p) =>
   787→        PRIORITY_CYCLE[(PRIORITY_CYCLE.indexOf(p) + 1) % PRIORITY_CYCLE.length]

   789→
   790→  const cycleDue = () =>
   791→    setQuickDue(
   792→      (d) => DUE_CYCLE[(DUE_CYCLE.indexOf(d) + 1) % DUE_CYCLE.length]
   793→    );
   794→
   795→  const completedShown = completedTodos.length > 0 || tab === "completed";
   796→  const completedExpanded = tab === "completed" || completedOpen;
   797→  const hasAnyTodos = todos.length > 0;
   798→
   799→  const empty = EMPTY_BY_TAB[tab];
   800→  const categoryMeta = TODO_CATEGORIES.find((c) => c.value === category);
   801→  const categoryScoped = category !== "all" && hasAnyTodos;
   802→  const showEmptyAction = tab === "all" || tab === "today" || tab === "upcoming";
   803→  const focusQuickAdd = () => quickAddRef.current?.focus();

   805→  return (
   806→    <div className="space-y-4 sm:space-y-5">
   807→      <ViewHeader

   809→        subtitle={`${doneTodayCount} done today · ${overdueCount} overdue`}

   811→          completedTotal > 0 ? (
   812→            <Button
   813→              variant="outline"
   814→              size="sm"
   815→              onClick={() => setClearOpen(true)}
   816→              className="gap-1.5 rounded-xl text-muted-foreground hover:border-destructive/40 hover:bg-destructive/10 hover:text-destructive"
   817→            >
   818→              <Trash2 className="size-3.5" aria-hidden="true" />
   819→              Clear completed
   820→            </Button>
   821→          ) : undefined
   822→        }
   823→      />
   824→
   825→      {/* Quick add */}
   826→      <section aria-label="Add a task">
   827→        <div className="rounded-2xl border bg-card p-2 shadow-sm">
   828→          <div className="flex items-center gap-1.5">
   829→            <span
   830→              className="flex size-9 shrink-0 items-center justify-center rounded-xl bg-primary/10 text-primary"
   831→              aria-hidden="true"
   832→            >
   833→              <Plus className="size-4.5" />
   834→            </span>
   835→            <Input
   836→              ref={quickAddRef}
   837→              value={quickTitle}
   838→              onChange={(e) => setQuickTitle(e.target.value)}
   839→              onKeyDown={(e) => {
   840→                if (e.key === "Enter") {
   841→                  e.preventDefault();
   842→                  submitQuickAdd();
   843→                }
   844→              }}
   845→              placeholder="Add a task…"
   846→              aria-label="Task title"
   847→              className="h-9 flex-1 border-0 bg-transparent px-1 shadow-none focus-visible:ring-0 dark:bg-transparent dark:border-0"
   848→            />
   849→            <Button
   850→              size="icon"
   851→              aria-label="Add task"
   852→              onClick={submitQuickAdd}
   853→              disabled={createMutation.isPending}
   854→              className="size-9 shrink-0 rounded-xl"
   855→            >
   856→              {createMutation.isPending ? (
   857→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   858→              ) : (
   859→                <Plus className="size-4" aria-hidden="true" />
   860→              )}
   861→            </Button>
   862→          </div>
   863→          <div className="mt-1.5 flex flex-wrap items-center gap-1.5 px-1 pb-1">
   864→            <button
   865→              type="button"
   866→              onClick={cyclePriority}
   867→              aria-label={`Priority: ${quickPriority}. Tap to change.`}
   868→              title="Tap to change priority"
   869→              className={cn(
   870→                "inline-flex h-8 items-center gap-1.5 rounded-full border px-2.5 text-xs font-medium capitalize transition-colors",
   871→                priorityChipStyles[quickPriority]
   872→              )}
   873→            >
   874→              <span
   875→                className={cn(
   876→                  "size-2 rounded-full",
   877→                  quickPriority === "urgent" && "bg-red-500",
   878→                  quickPriority === "high" && "bg-orange-500",
   879→                  quickPriority === "medium" && "bg-amber-500",
   880→                  quickPriority === "low" && "bg-muted-foreground/50"
   881→                )}
   882→                aria-hidden="true"
   883→              />
   884→              {quickPriority}
   885→            </button>
   886→            <button
   887→              type="button"
   888→              onClick={cycleDue}
   889→              aria-label={`Due: ${quickDue === "none" ? "no date" : quickDue}. Tap to change.`}
   890→              title="Tap to change due date"
   891→              className={cn(
   892→                "inline-flex h-8 items-center gap-1.5 rounded-full border px-2.5 text-xs font-medium capitalize transition-colors",
   893→                quickDue === "none"
   894→                  ? "border-border bg-muted/50 text-muted-foreground"
   895→                  : "border-primary/40 bg-primary/10 text-primary"
   896→              )}
   897→            >
   898→              <CalendarDays className="size-3.5" aria-hidden="true" />
   899→              {quickDue === "none" ? "No date" : quickDue}
   900→            </button>
   901→          </div>
   902→        </div>
   903→      </section>
   904→

   911→                key={t.value}
   912→                value={t.value}
   913→                className="flex-1 whitespace-nowrap px-2 text-xs sm:text-sm"
   914→              >
   915→                {t.label}
   916→              </TabsTrigger>
   917→            ))}
   918→          </TabsList>
   919→        </Tabs>
   920→        <div
   921→          className="flex gap-1.5 overflow-x-auto pb-0.5 no-scrollbar"
   922→          role="group"
   923→          aria-label="Filter by category"
   924→        >
   925→          <CategoryChip
   926→            label="All"
   927→            active={category === "all"}
   928→            onClick={() => setCategory("all")}
   929→          />
   930→          {TODO_CATEGORIES.map((c) => (
   931→            <CategoryChip
   932→              key={c.value}
   933→              label={`${c.label} ${c.emoji}`}
   934→              active={category === c.value}
   935→              onClick={() => setCategory(c.value)}
   936→            />
   937→          ))}
   938→        </div>
   939→      </section>
   940→
   941→      {/* Task list */}
   942→      {isLoading ? (
   943→        <TasksSkeleton />
   944→      ) : isError ? (
   945→        <EmptyState
   946→          icon={CloudOff}
   947→          title="Couldn't load your tasks"
   948→          description="Something went wrong while fetching your tasks. Check your connection and try again."
   949→          actionLabel="Try again"
   950→          onAction={() => void refetch()}
   951→        />
   952→      ) : filtered.length === 0 ? (
   953→        <EmptyState
   954→          icon={categoryScoped ? ListTodo : empty.icon}
   955→          title={
   956→            categoryScoped
   957→              ? `No ${categoryMeta?.label ?? category} tasks here`
   958→              : empty.title
   959→          }
   960→          description={
   961→            categoryScoped
   962→              ? "Try another category filter, or add a new task in this one."
   963→              : empty.description
   964→          }
   965→          actionLabel={showEmptyAction ? "Add a task" : undefined}
   966→          onAction={showEmptyAction ? focusQuickAdd : undefined}
   967→        />
   968→      ) : (
   969→        <div className="divide-y divide-border/70 rounded-2xl border bg-card shadow-sm">
   970→          {visibleGroups.map((g) => (
   971→            <section key={g.id} aria-label={`${g.label} tasks`}>
   972→              <GroupLabel
   973→                label={g.label}
   974→                count={g.items.length}
   975→                danger={g.tone === "danger"}
   976→              />
   977→              <div className="divide-y divide-border/60">
   978→                {g.items.map((t) => (
   979→                  <TodoRow
   980→                    key={t.id}
   981→                    todo={t}
   982→                    onToggle={(todo) => toggleMutation.mutate(todo)}
   983→                    onEdit={setEditing}
   984→                    onDelete={setDeleting}
   985→                  />
   986→                ))}
   987→              </div>
   988→            </section>
   989→          ))}
   990→
   991→          {completedShown && (
   992→            <section aria-label="Completed tasks">
   993→              <div className="sticky top-16 z-10 flex items-center justify-between gap-2 border-b border-border/70 bg-card/95 px-3 py-1 backdrop-blur sm:px-4 lg:top-0">
   994→                <button
   995→                  type="button"
   996→                  onClick={() => setCompletedOpen((v) => !v)}
   997→                  aria-expanded={completedExpanded}
   998→                  className="flex items-center gap-1 rounded-md py-0.5 text-xs font-semibold uppercase tracking-wider text-muted-foreground transition-colors hover:text-foreground"
   999→                >
  1000→                  <ChevronDown
  1001→                    className={cn(
  1002→                      "size-3.5 transition-transform duration-200",
  1003→                      completedExpanded && "rotate-180"
  1004→                    )}
  1005→                    aria-hidden="true"
  1006→                  />
  1007→                  Completed
  1008→                </button>
  1009→                <div className="flex items-center gap-2.5">
  1010→                  <span className="text-xs font-medium tabular-nums text-muted-foreground/70">
  1011→                    {completedTodos.length}
  1012→                  </span>
  1013→                  {completedTodos.length > 0 && (
  1014→                    <button
  1015→                      type="button"
  1016→                      onClick={() => setClearOpen(true)}
  1017→                      className="rounded-md px-1 py-0.5 text-xs font-medium text-muted-foreground transition-colors hover:text-destructive"
  1018→                    >
  1019→                      Clear
  1020→                    </button>
  1021→                  )}
  1022→                </div>
  1023→              </div>
  1024→              {completedExpanded && (
  1025→                <div className="max-h-72 divide-y divide-border/60 overflow-y-auto">
  1026→                  {completedTodos.map((t) => (
  1027→                    <TodoRow
  1028→                      key={t.id}
  1029→                      todo={t}
  1030→                      onToggle={(todo) => toggleMutation.mutate(todo)}
  1031→                      onEdit={setEditing}
  1032→                      onDelete={setDeleting}
  1033→                    />
  1034→                  ))}
  1035→                </div>
  1036→              )}
  1037→            </section>
  1038→          )}
  1039→        </div>
  1040→      )}
  1041→
  1042→      {/* Edit dialog */}
  1043→      {editing && (
  1044→        <EditTaskDialog
  1045→          key={editing.id}
  1046→          todo={editing}
  1047→          saving={updateMutation.isPending}
  1048→          onCancel={() => setEditing(null)}
  1049→          onSave={(patch) =>

  1054→
  1055→      {/* Delete confirm */}
  1056→      <AlertDialog

  1059→          if (!open) setDeleting(null);
  1060→        }}
  1061→      >
  1062→        <AlertDialogContent>
  1063→          <AlertDialogHeader>
  1064→            <AlertDialogTitle>Delete this task?</AlertDialogTitle>
  1065→            <AlertDialogDescription>
  1066→              &ldquo;{deleting?.title}&rdquo; will be permanently removed. This
  1067→              cannot be undone.
  1068→            </AlertDialogDescription>
  1069→          </AlertDialogHeader>
  1070→          <AlertDialogFooter>
  1071→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1072→            <AlertDialogAction
  1073→              onClick={(e) => {
  1074→                e.preventDefault();
  1075→                if (deleting) deleteMutation.mutate(deleting.id);

  1077→              className="bg-destructive text-white hover:bg-destructive/90"
  1078→            >
  1079→              {deleteMutation.isPending && (
  1080→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
  1081→              )}
  1082→              Delete
  1083→            </AlertDialogAction>
  1084→          </AlertDialogFooter>
  1085→        </AlertDialogContent>
  1086→      </AlertDialog>
  1087→
  1088→      {/* Clear completed confirm */}
  1089→      <AlertDialog open={clearOpen} onOpenChange={setClearOpen}>
  1090→        <AlertDialogContent>
  1091→          <AlertDialogHeader>
  1092→            <AlertDialogTitle>Clear completed tasks?</AlertDialogTitle>
  1093→            <AlertDialogDescription>
  1094→              {completedTotal} completed {completedTotal === 1 ? "task" : "tasks"}{" "}
  1095→              will be permanently deleted. This cannot be undone.
  1096→            </AlertDialogDescription>
  1097→          </AlertDialogHeader>
  1098→          <AlertDialogFooter>
  1099→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1100→            <AlertDialogAction
  1101→              onClick={(e) => {
  1102→                e.preventDefault();
  1103→                clearMutation.mutate();
  1104→              }}
  1105→              className="bg-destructive text-white hover:bg-destructive/90"
  1106→            >
  1107→              {clearMutation.isPending && (
  1108→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
  1109→              )}
  1110→              Clear all
  1111→            </AlertDialogAction>
  1112→          </AlertDialogFooter>
  1113→        </AlertDialogContent>
  1114→      </AlertDialog>
  1115→    </div>
  1116→  );
  1117→}
