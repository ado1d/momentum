   536→export function DashboardView() {
   537→  const setView = useUiStore((s) => s.setView);
   538→  const setQuickAddOpen = useUiStore((s) => s.setQuickAddOpen);
   539→  const queryClient = useQueryClient();
   540→
   541→  const { data, isLoading, isError, refetch } = useQuery({
   542→    queryKey: ["stats", "dashboard"],
   543→    queryFn: statsApi.dashboard,
   544→  });
   545→
   546→  const weekDays = React.useMemo(() => lastNDays(7), []);
   547→
   548→  // Toggle a focus todo with an optimistic update so the check feels instant.
   549→  const toggleTodo = useMutation({
   550→    mutationFn: (todo: Todo) =>
   551→      todosApi.update(todo.id, { completed: !todo.completed }),
   552→    onMutate: async (todo) => {
   553→      await queryClient.cancelQueries({ queryKey: ["stats", "dashboard"] });
   554→      const prev = queryClient.getQueryData<DashboardStats>([
   555→        "stats",
   556→        "dashboard",
   557→      ]);
   558→      if (prev) {
   559→        queryClient.setQueryData<DashboardStats>(["stats", "dashboard"], {
   560→          ...prev,
   561→          upcomingTodos: prev.upcomingTodos.map((t) =>
   562→            t.id === todo.id
   563→              ? {
   564→                  ...t,
   565→                  completed: !t.completed,
   566→                  completedAt: !t.completed
   567→                    ? new Date().toISOString()
   568→                    : null,
   569→                }
   570→              : t
   571→          ),
   572→        });
   573→      }
   574→      return { prev };
   575→    },
   576→    onError: (err, _todo, ctx) => {
   577→      if (ctx?.prev) {
   578→        queryClient.setQueryData(["stats", "dashboard"], ctx.prev);
   579→      }
   580→      toast.error(
   581→        err instanceof Error ? err.message : "Couldn't update the task"
   582→      );
   583→    },
   584→    onSettled: () => {
   585→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   586→      void queryClient.invalidateQueries({ queryKey: ["todos"] });
   587→    },
   588→  });
   589→
   590→  // Toggle a habit for today, optimistically flipping the chip + week dots.
   591→  const toggleHabit = useMutation({
   592→    mutationFn: (habit: Habit) => habitsApi.toggle(habit.id, todayKey()),
   593→    onMutate: async (habit) => {
   594→      await queryClient.cancelQueries({ queryKey: ["stats", "dashboard"] });
   595→      const prev = queryClient.getQueryData<DashboardStats>([
   596→        "stats",
   597→        "dashboard",
   598→      ]);
   599→      const today = todayKey();
   600→      if (prev) {
   601→        queryClient.setQueryData<DashboardStats>(["stats", "dashboard"], {
   602→          ...prev,
   603→          todayHabits: prev.todayHabits.map((h) => {
   604→            if (h.id !== habit.id) return h;
   605→            const done = !h.doneToday;
   606→            const logs = done
   607→              ? [...h.logs, { id: `tmp-${h.id}`, habitId: h.id, date: today }]
   608→              : h.logs.filter((l) => l.date !== today);
   609→            return { ...h, doneToday: done, logs };
   610→          }),
   611→        });
   612→      }
   613→      return { prev };
   614→    },
   615→    onError: (err, _habit, ctx) => {
   616→      if (ctx?.prev) {
   617→        queryClient.setQueryData(["stats", "dashboard"], ctx.prev);
   618→      }
   619→      toast.error(
   620→        err instanceof Error ? err.message : "Couldn't toggle the habit"
   621→      );
   622→    },
   623→    onSettled: () => {
   624→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   625→      void queryClient.invalidateQueries({ queryKey: ["habits"] });
   626→    },
   627→  });
   628→
   629→  // Onboarding: create three starter habits in one go.
   630→  const addStarterHabits = useMutation({
   631→    mutationFn: async () => {
   632→      await habitsApi.create({
   633→        name: "Drink 8 glasses of water",
   634→        emoji: "💧",
   635→        color: "teal",
   636→        timeOfDay: "anytime",
   637→      });
   638→      await habitsApi.create({
   639→        name: "Read 20 minutes",
   640→        emoji: "📚",
   641→        color: "amber",
   642→        timeOfDay: "evening",
   643→      });
   644→      await habitsApi.create({
   645→        name: "Move your body",
   646→        emoji: "🏃",
   647→        color: "rose",
   648→        timeOfDay: "morning",
   649→      });
   650→    },
   651→    onSuccess: () => {
   652→      toast.success("Starter habits added", {
   653→        description: "Find them under Routine and check them off daily.",
   654→      });
   655→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   656→      void queryClient.invalidateQueries({ queryKey: ["habits"] });
   657→    },
   658→    onError: (err) =>
   659→      toast.error(
   660→        err instanceof Error ? err.message : "Couldn't add starter habits"
   661→      ),
   662→  });
   663→
   664→  if (isLoading) return <DashboardSkeleton />;
   665→  if (isError || !data) {
   666→    return <DashboardError onRetry={() => void refetch()} />;
   667→  }
   668→
   669→  const { today, week, activeGoals, upcomingTodos, todayHabits, recentJournal, quote } =
   670→    data;
   671→  const isEmpty =
   672→    today.todosTotal === 0 &&
   673→    today.habitsTotal === 0 &&
   674→    activeGoals.length === 0 &&
   675→    today.routineTotal === 0;
   676→
   677→  if (isEmpty) {
   678→    return (
   679→      <div className="space-y-6">
   680→        <FadeIn>
   681→          <DashboardHeader score={today.score} />
   682→        </FadeIn>
   683→        <FadeIn delay={0.05}>
   684→          <OnboardingCard
   685→            habitsPending={addStarterHabits.isPending}
   686→            onAddTask={() => setQuickAddOpen(true)}
   687→            onAddHabits={() => addStarterHabits.mutate()}
   688→            onSetGoal={() => setView("goals")}
   689→          />
   690→        </FadeIn>
   691→        <FadeIn delay={0.1}>
   692→          <QuoteCard quote={quote} />
   693→        </FadeIn>
   694→      </div>
   695→    );
   696→  }
   697→
   698→  return (
   699→    <div className="space-y-6">
   700→      <FadeIn>
   701→        <DashboardHeader score={today.score} />
   702→      </FadeIn>
   703→
   704→      <FadeIn delay={0.04}>
   705→        <section aria-label="Quote of the day">
   706→          <QuoteCard quote={quote} />
   707→        </section>
   708→      </FadeIn>
   709→
   710→      <FadeIn delay={0.08}>
   711→        <StatCards today={today} />
   712→      </FadeIn>
   713→
   714→      <FadeIn delay={0.12}>
   715→        <WeekCard week={week} />
   716→      </FadeIn>
   717→
   718→      {today.overdueCount > 0 && (
   719→        <FadeIn delay={0.14}>
   720→          <OverdueBanner
   721→            count={today.overdueCount}
   722→            onReview={() => setView("tasks")}
   723→          />
   724→        </FadeIn>
   725→      )}
   726→
   727→      <FadeIn delay={0.16}>
   728→        <section aria-label="Today's focus">
   729→          <SectionHeading
   730→            title="Today's focus"
   731→            action={<SeeAllButton onClick={() => setView("tasks")} />}
   732→          />
   733→          {upcomingTodos.length === 0 ? (
   734→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   735→              Nothing due in the next week — enjoy the breathing room.
   736→            </p>
   737→          ) : (
   738→            <div className="divide-y divide-border/70 rounded-2xl border bg-card shadow-card">
   739→              {upcomingTodos.slice(0, 6).map((todo) => (
   740→                <FocusRow
   741→                  key={todo.id}
   742→                  todo={todo}
   743→                  onToggle={(t) => toggleTodo.mutate(t)}
   744→                />
   745→              ))}
   746→            </div>
   747→          )}
   748→        </section>
   749→      </FadeIn>
   750→
   751→      <FadeIn delay={0.2}>
   752→        <section aria-label="Habits today">
   753→          <SectionHeading
   754→            title="Habits today"
   755→            action={<SeeAllButton onClick={() => setView("routine")} />}
   756→          />
   757→          {todayHabits.length === 0 ? (
   758→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   759→              No habits yet — build momentum with one small daily win.
   760→            </p>
   761→          ) : (
   762→            <div className="-mx-1 flex gap-3 overflow-x-auto px-1 pb-2 no-scrollbar">
   763→              {todayHabits.map((habit) => (
   764→                <HabitChip
   765→                  key={habit.id}
   766→                  habit={habit}
   767→                  weekDays={weekDays}
   768→                  onToggle={(h) => toggleHabit.mutate(h)}
   769→                />
   770→              ))}
   771→            </div>
   772→          )}
   773→        </section>
   774→      </FadeIn>
   775→
   776→      <FadeIn delay={0.24}>
   777→        <section aria-label="Active goals">
   778→          <SectionHeading
   779→            title="Active goals"
   780→            action={<SeeAllButton onClick={() => setView("goals")} />}
   781→          />
   782→          {activeGoals.length === 0 ? (
   783→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   784→              No active goals — set one to give your days direction.
   785→            </p>
   786→          ) : (
   787→            <div className="grid gap-3 sm:grid-cols-2">
   788→              {activeGoals.slice(0, 4).map((goal) => {
   789→                const percent = ratio(goal.progress, goal.target);
   790→                return (
   791→                  <button
   792→                    key={goal.id}
   793→                    type="button"
   794→                    onClick={() => setView("goals")}
   795→                    aria-label={`Open goal: ${goal.title}`}
   796→                    className="press rounded-2xl border bg-card p-4 text-left shadow-card transition-all hover:-translate-y-0.5 hover:border-primary/40 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50 active:scale-[0.99]"
   797→                  >
   798→                    <div className="flex items-start justify-between gap-2">
   799→                      <p className="min-w-0 text-sm font-semibold leading-snug line-clamp-1">
   800→                        {goal.title}
   801→                      </p>
   802→                      <Badge
   803→                        variant="outline"
   804→                        className="shrink-0 rounded-full px-2 py-0 text-[10px] font-medium capitalize"
   805→                      >
   806→                        {goal.period}
   807→                      </Badge>
   808→                    </div>
   809→                    <ProgressBar value={percent} className="mt-3" />
   810→                    <p className="mt-1.5 text-xs tabular-nums text-muted-foreground">
   811→                      <span className="font-semibold text-foreground">
   812→                        {goal.progress}
   813→                      </span>
   814→                      /{goal.target}
   815→                      {goal.unit ? ` ${goal.unit}` : ""} · {percent}%
   816→                    </p>
   817→                  </button>
   818→                );
   819→              })}
   820→            </div>
   821→          )}
   822→        </section>
   823→      </FadeIn>
   824→
   825→      <FadeIn delay={0.28}>
   826→        <section aria-label="Recent journal">
   827→          <SectionHeading
   828→            title="Recent journal"
   829→            action={
   830→              <SeeAllButton
   831→                label="Open diary"
   832→                onClick={() => setView("diary")}
   833→              />
   834→            }
   835→          />
   836→          {recentJournal.length === 0 ? (
   837→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   838→              No entries yet — tonight is a good night to write.
   839→            </p>
   840→          ) : (
   841→            <div className="divide-y divide-border/70 rounded-2xl border bg-card shadow-card">
   842→              {recentJournal.map((entry) => {
   843→                const mood = MOODS.find((m) => m.value === entry.mood);
   844→                return (
   845→                  <button
   846→                    key={entry.id}
   847→                    type="button"
   848→                    onClick={() => setView("diary")}
   849→                    aria-label={`Open diary entry from ${friendlyDay(entry.date)}`}
   850→                    className="flex w-full items-start gap-3 px-4 py-3 text-left transition-colors hover:bg-muted/40"
   851→                  >
   852→                    <span
   853→                      className="mt-0.5 flex size-9 shrink-0 items-center justify-center rounded-xl bg-muted text-base"
   854→                      aria-hidden="true"
   855→                    >
   856→                      {mood ? (
   857→                        mood.emoji
   858→                      ) : (
   859→                        <BookOpen className="size-4 text-muted-foreground" />
   860→                      )}
   861→                    </span>
   862→                    <span className="min-w-0 flex-1">
   863→                      <span className="flex items-baseline justify-between gap-2">
   864→                        <span className="truncate text-sm font-semibold">
   865→                          {entry.title || "Journal entry"}
   866→                        </span>
   867→                        <span className="shrink-0 text-[11px] text-muted-foreground">
   868→                          {friendlyDay(entry.date)}
   869→                        </span>
   870→                      </span>
   871→                      <span className="mt-0.5 block line-clamp-2 text-xs leading-relaxed text-muted-foreground">
   872→                        {entry.content}
   873→                      </span>
   874→                    </span>
   875→                  </button>
   876→                );
   877→              })}
   878→            </div>
   879→          )}
   880→        </section>
   881→      </FadeIn>
   882→    </div>
   883→  );
   884→}
