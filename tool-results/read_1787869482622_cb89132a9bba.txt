   104→   104→/** Longest run of consecutive day keys in a list of entries */
   105→   105→function longestStreakOf(entries: JournalEntry[]): number {
   106→   106→  const keys = Array.from(new Set(entries.map((e) => e.date))).sort();
   107→   107→  let best = 0;
   108→   108→  let run = 0;
   109→   109→  let prev: string | null = null;
   110→   110→  for (const k of keys) {
   111→   111→    run = prev !== null && addDaysToKey(prev, 1) === k ? run + 1 : 1;
   112→   112→    if (run > best) best = run;
   113→   113→    prev = k;
   114→   114→  }
   115→   115→  return best;
   116→   116→}

   122→   122→}

   421→   421→    for (const e of [...entries].sort((a, b) => b.date.localeCompare(a.date))) {
   422→   422→      const month = monthLabel(e.date);

   426→   426→    }

   432→   432→  const save = () => {

   435→   435→      !title.trim() &&
   436→   436→      !content.trim() &&
   437→   437→      !gratitude.trim() &&
   438→   438→      mood === null
   439→   439→    ) {
   440→   440→      toast.error("Write something first — even one line counts");
   441→   441→      contentRef.current?.focus();
   442→   442→      return;
   443→   443→    }
   444→   444→    saveEntry.mutate();
   445→   445→  };

   447→   447→  const openDay = (e: JournalEntry) => {
   448→   448→    setDate(e.date);
   449→   449→    setExpandedId(null);
   450→   450→    window.scrollTo({ top: 0, behavior: "smooth" });
   451→   451→  };
   452→   452→
   453→   453→  const focusTodayEditor = () => {
   454→   454→    setDate(today);
   455→   455→    window.scrollTo({ top: 0, behavior: "smooth" });
   456→   456→    // Wait for the form to sync to today's entry, then focus.
   457→   457→    setTimeout(() => contentRef.current?.focus(), 120);
   458→   458→  };
   459→   459→
   460→   460→  const runExport = async (
   461→   461→    id: string,
   462→   462→    label: string,
   463→   463→    fn: () => Promise<void> | void
   464→   464→  ) => {
