   263→    <Card className="gap-3 rounded-2xl p-5">
   264→      <div className="flex items-center justify-between gap-2">
   265→        <CardTitle className="text-sm font-semibold">This week</CardTitle>
   266→        <span className="flex items-center gap-2 text-xs text-muted-foreground">
   267→          <span title="Average daily score this week">avg {avg}%</span>
   268→          <span aria-hidden="true">·</span>
   269→          <span>Last 7 days</span>
   270→        </span>
   271→      </div>
   272→      <div>
   273→        <div
   274→          className="relative flex h-24 items-end gap-1.5 sm:gap-2"
   275→          role="img"
   276→          aria-label="Daily scores for the last 7 days"
   277→        >
   278→          {/* average marker line */}
   279→          <div
   280→            aria-hidden="true"
   281→            className="pointer-events-none absolute inset-x-0 border-t border-dashed border-primary/30"
   282→            style={{ bottom: `${Math.max(4, avg)}%` }}

   285→            const isToday = d.date === today;
   286→            return (
   287→              <div
   288→                key={d.date}

   297→                      : d.score > 0
   298→                        ? "bg-primary/30 dark:bg-primary/35"

   307→                  >
   308→                    {d.score}
   309→                  </span>
   310→                )}
   311→              </div>
   312→            );
   313→          })}
   314→        </div>
   315→        <div className="mt-2 flex gap-1.5 sm:gap-2">
   316→          {week.map((d) => (
   317→            <span
   318→              key={d.date}
   319→              className={cn(
   320→                "flex-1 text-center text-[10px] font-semibold uppercase",
   321→                d.date === today ? "text-primary" : "text-muted-foreground/60"

   324→              {shortDayName(d.date).slice(0, 1)}
   325→            </span>

   362→function FocusRow({
   363→  todo,
   364→  onToggle,
   365→}: {
   366→  todo: Todo;
   367→  onToggle: (todo: Todo) => void;
   368→}) {
   369→  const overdue = !todo.completed && isOverdue(todo.dueDate);
   370→  const subtaskTotal = todo.subtasks.length;
   371→  const subtaskDone = todo.subtasks.filter((s) => s.completed).length;
   372→  return (
   373→    <div className="flex items-center gap-3 px-4 py-2.5">
   374→      <button
   375→        type="button"
   376→        role="checkbox"
   377→        aria-checked={todo.completed}
   378→        aria-label={`Mark "${todo.title}" ${todo.completed ? "incomplete" : "complete"}`}
   379→        onClick={() => onToggle(todo)}
   380→        className={cn(
   381→          "relative mt-0.5 flex size-6 shrink-0 items-center justify-center rounded-full border-2 transition-all duration-200 active:scale-90",
   382→          todo.completed
   383→            ? "border-primary bg-primary text-primary-foreground shadow-sm"
   384→            : "border-muted-foreground/30 hover:border-primary/60 hover:bg-primary/10"
   385→        )}
   386→      >
   387→        <Check
   388→          className={cn(
   389→            "size-3.5 transition-all duration-200",
   390→            todo.completed ? "scale-100 opacity-100" : "scale-50 opacity-0"
   391→          )}
   392→          strokeWidth={3}
   393→          aria-hidden="true"
   394→        />
   395→        <span className="absolute -inset-2" aria-hidden="true" />
   396→      </button>
   397→      <span
   398→        className={cn(
   399→          "size-2 shrink-0 rounded-full",
   400→          priorityDotStyles[todo.priority] ?? priorityDotStyles.low
   401→        )}
   402→        title={`${todo.priority} priority`}
   403→        aria-hidden="true"
   404→      />
   405→      <p
   406→        className={cn(
   407→          "min-w-0 flex-1 truncate text-sm font-medium decoration-2 transition-colors duration-300",
   408→          "line-through",
   409→          todo.completed
   410→            ? "text-muted-foreground/70 decoration-muted-foreground/50"
   411→            : "text-foreground decoration-transparent"
   412→        )}
   413→      >
   414→        {todo.title}
   415→      </p>
   416→      {todo.repeat !== "none" && (
   417→        <span
   418→          className="inline-flex shrink-0 items-center text-muted-foreground"
   419→          title={`Repeats ${todo.repeat}`}
   420→        >
   421→          <Repeat className="size-3" aria-hidden="true" />
   422→          <span className="sr-only">Repeats {todo.repeat}</span>
   423→        </span>
   424→      )}
   425→      {subtaskTotal > 0 && (
   426→        <span
   427→          className="inline-flex shrink-0 items-center gap-0.5 text-[11px] font-medium tabular-nums text-muted-foreground"
   428→          title="Checklist progress"
   429→        >
   430→          <ListChecks className="size-3" aria-hidden="true" />
   431→          <span aria-hidden="true">
   432→            {subtaskDone}/{subtaskTotal}
   433→          </span>
   434→          <span className="sr-only">
   435→            Checklist: {subtaskDone} of {subtaskTotal}{" "}
   436→            {subtaskTotal === 1 ? "step" : "steps"} done
   437→          </span>
   438→        </span>
   439→      )}
   440→      {todo.dueDate && (
   441→        <span
   442→          className={cn(
   443→            "shrink-0 text-xs font-medium tabular-nums",
   444→            overdue
   445→              ? "text-red-600 dark:text-red-400"
   446→              : "text-muted-foreground"
   447→          )}
   448→        >
   449→          {formatDueLabel(todo.dueDate)}
   450→        </span>
   451→      )}
   452→    </div>
   453→  );
   454→}

   559→  const { data, isLoading, isError, refetch } = useQuery({
   560→    queryKey: ["stats", "dashboard"],
   561→    queryFn: statsApi.dashboard,
   562→  });
   563→
   564→  const weekDays = React.useMemo(() => lastNDays(7), []);
   565→
   566→  // Toggle a focus todo with an optimistic update so the check feels instant.
   567→  const toggleTodo = useMutation({
   568→    mutationFn: (todo: Todo) =>
   569→      todosApi.update(todo.id, { completed: !todo.completed }),
   570→    onMutate: async (todo) => {
   571→      await queryClient.cancelQueries({ queryKey: ["stats", "dashboard"] });
   572→      const prev = queryClient.getQueryData<DashboardStats>([
   573→        "stats",
   574→        "dashboard",
   575→      ]);
   576→      if (prev) {
   577→        queryClient.setQueryData<DashboardStats>(["stats", "dashboard"], {
   578→          ...prev,
   579→          upcomingTodos: prev.upcomingTodos.map((t) =>
   580→            t.id === todo.id
   581→              ? {
   582→                  ...t,
   583→                  completed: !t.completed,
   584→                  completedAt: !t.completed
   585→                    ? new Date().toISOString()
   586→                    : null,
   587→                }
   588→              : t
   589→          ),
   590→        });
   591→      }
   592→      return { prev };
   593→    },
   594→    onError: (err, _todo, ctx) => {
   595→      if (ctx?.prev) {
   596→        queryClient.setQueryData(["stats", "dashboard"], ctx.prev);
   597→      }
   598→      toast.error(
   599→        err instanceof Error ? err.message : "Couldn't update the task"
   600→      );
   601→    },
   602→    onSettled: () => {
   603→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   604→      void queryClient.invalidateQueries({ queryKey: ["todos"] });
   605→    },
   606→  });
   607→
   608→  // Toggle a habit for today, optimistically flipping the chip + week dots.
   609→  const toggleHabit = useMutation({
   610→    mutationFn: (habit: Habit) => habitsApi.toggle(habit.id, todayKey()),
   611→    onMutate: async (habit) => {
   612→      await queryClient.cancelQueries({ queryKey: ["stats", "dashboard"] });
   613→      const prev = queryClient.getQueryData<DashboardStats>([
   614→        "stats",
   615→        "dashboard",
   616→      ]);
   617→      const today = todayKey();
   618→      if (prev) {
   619→        queryClient.setQueryData<DashboardStats>(["stats", "dashboard"], {
   620→          ...prev,
   621→          todayHabits: prev.todayHabits.map((h) => {
   622→            if (h.id !== habit.id) return h;
   623→            const done = !h.doneToday;
   624→            const logs = done
   625→              ? [...h.logs, { id: `tmp-${h.id}`, habitId: h.id, date: today }]
   626→              : h.logs.filter((l) => l.date !== today);
   627→            return { ...h, doneToday: done, logs };
   628→          }),
   629→        });
   630→      }
   631→      return { prev };
   632→    },
   633→    onError: (err, _habit, ctx) => {
   634→      if (ctx?.prev) {
   635→        queryClient.setQueryData(["stats", "dashboard"], ctx.prev);
   636→      }
   637→      toast.error(
   638→        err instanceof Error ? err.message : "Couldn't toggle the habit"
   639→      );
   640→    },
   641→    onSettled: () => {
   642→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   643→      void queryClient.invalidateQueries({ queryKey: ["habits"] });
   644→    },
   645→  });
   646→
   647→  // Onboarding: create three starter habits in one go.
   648→  const addStarterHabits = useMutation({
   649→    mutationFn: async () => {
   650→      await habitsApi.create({
   651→        name: "Drink 8 glasses of water",
   652→        emoji: "💧",
   653→        color: "teal",
   654→        timeOfDay: "anytime",
   655→      });
   656→      await habitsApi.create({
   657→        name: "Read 20 minutes",
   658→        emoji: "📚",
   659→        color: "amber",
   660→        timeOfDay: "evening",
   661→      });
   662→      await habitsApi.create({
   663→        name: "Move your body",
   664→        emoji: "🏃",
   665→        color: "rose",
   666→        timeOfDay: "morning",
   667→      });
   668→    },
   669→    onSuccess: () => {
   670→      toast.success("Starter habits added", {
   671→        description: "Find them under Routine and check them off daily.",
   672→      });
   673→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   674→      void queryClient.invalidateQueries({ queryKey: ["habits"] });
   675→    },
   676→    onError: (err) =>
   677→      toast.error(
   678→        err instanceof Error ? err.message : "Couldn't add starter habits"
   679→      ),
   680→  });
   681→
   682→  if (isLoading) return <DashboardSkeleton />;
   683→  if (isError || !data) {
   684→    return <DashboardError onRetry={() => void refetch()} />;
   685→  }
   686→
   687→  const { today, week, activeGoals, upcomingTodos, todayHabits, recentJournal, quote } =
   688→    data;
   689→  const isEmpty =
   690→    today.todosTotal === 0 &&
   691→    today.habitsTotal === 0 &&
   692→    activeGoals.length === 0 &&
   693→    today.routineTotal === 0;
   694→
   695→  if (isEmpty) {
   696→    return (
   697→      <div className="space-y-6">
   698→        <FadeIn>
   699→          <DashboardHeader score={today.score} />
   700→        </FadeIn>
   701→        <FadeIn delay={0.05}>
   702→          <OnboardingCard
   703→            habitsPending={addStarterHabits.isPending}
   704→            onAddTask={() => setQuickAddOpen(true)}
   705→            onAddHabits={() => addStarterHabits.mutate()}
   706→            onSetGoal={() => setView("goals")}
   707→          />
   708→        </FadeIn>
   709→        <FadeIn delay={0.1}>
   710→          <QuoteCard quote={quote} />
   711→        </FadeIn>
   712→      </div>
   713→    );
   714→  }
   715→
   716→  return (
   717→    <div className="space-y-6">
   718→      <FadeIn>
   719→        <DashboardHeader score={today.score} />
   720→      </FadeIn>
   721→
   722→      <FadeIn delay={0.04}>
   723→        <section aria-label="Quote of the day">
   724→          <QuoteCard quote={quote} />
   725→        </section>
   726→      </FadeIn>
   727→
   728→      <FadeIn delay={0.08}>
   729→        <StatCards today={today} />
   730→      </FadeIn>
   731→
   732→      <FadeIn delay={0.12}>
   733→        <WeekCard week={week} />
   734→      </FadeIn>
   735→
   736→      {today.overdueCount > 0 && (
   737→        <FadeIn delay={0.14}>
   738→          <OverdueBanner
   739→            count={today.overdueCount}
   740→            onReview={() => setView("tasks")}
   741→          />
   742→        </FadeIn>
   743→      )}
   744→
   745→      <FadeIn delay={0.16}>
   746→        <section aria-label="Today's focus">
   747→          <SectionHeading
   748→            title="Today's focus"
   749→            action={<SeeAllButton onClick={() => setView("tasks")} />}
   750→          />
   751→          {upcomingTodos.length === 0 ? (
   752→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   753→              Nothing due in the next week — enjoy the breathing room.
   754→            </p>
   755→          ) : (
   756→            <div className="divide-y divide-border/70 rounded-2xl border bg-card shadow-card">
   757→              {upcomingTodos.slice(0, 6).map((todo) => (
   758→                <FocusRow
   759→                  key={todo.id}
   760→                  todo={todo}
   761→                  onToggle={(t) => toggleTodo.mutate(t)}
   762→                />
   763→              ))}
   764→            </div>
   765→          )}
   766→        </section>
   767→      </FadeIn>
   768→
   769→      <FadeIn delay={0.2}>
   770→        <section aria-label="Habits today">
   771→          <SectionHeading
   772→            title="Habits today"
   773→            action={<SeeAllButton onClick={() => setView("routine")} />}
   774→          />
   775→          {todayHabits.length === 0 ? (
   776→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   777→              No habits yet — build momentum with one small daily win.
   778→            </p>
   779→          ) : (
   780→            <div className="-mx-1 flex gap-3 overflow-x-auto px-1 pb-2 no-scrollbar">
   781→              {todayHabits.map((habit) => (
   782→                <HabitChip
   783→                  key={habit.id}
   784→                  habit={habit}
   785→                  weekDays={weekDays}
   786→                  onToggle={(h) => toggleHabit.mutate(h)}
   787→                />
   788→              ))}
   789→            </div>
   790→          )}
   791→        </section>
   792→      </FadeIn>
   793→
   794→      <FadeIn delay={0.24}>
   795→        <section aria-label="Active goals">
   796→          <SectionHeading
   797→            title="Active goals"
   798→            action={<SeeAllButton onClick={() => setView("goals")} />}
   799→          />
   800→          {activeGoals.length === 0 ? (
   801→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   802→              No active goals — set one to give your days direction.
   803→            </p>
   804→          ) : (
   805→            <div className="grid gap-3 sm:grid-cols-2">
   806→              {activeGoals.slice(0, 4).map((goal) => {
   807→                const percent = ratio(goal.progress, goal.target);
   808→                return (
   809→                  <button
   810→                    key={goal.id}
   811→                    type="button"
   812→                    onClick={() => setView("goals")}
   813→                    aria-label={`Open goal: ${goal.title}`}
   814→                    className="press rounded-2xl border bg-card p-4 text-left shadow-card transition-all hover:-translate-y-0.5 hover:border-primary/40 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50 active:scale-[0.99]"
   815→                  >
   816→                    <div className="flex items-start justify-between gap-2">
   817→                      <p className="min-w-0 text-sm font-semibold leading-snug line-clamp-1">
   818→                        {goal.title}
   819→                      </p>
   820→                      <Badge
   821→                        variant="outline"
   822→                        className="shrink-0 rounded-full px-2 py-0 text-[10px] font-medium capitalize"
   823→                      >
   824→                        {goal.period}
   825→                      </Badge>
   826→                    </div>
   827→                    <ProgressBar value={percent} className="mt-3" />
   828→                    <p className="mt-1.5 text-xs tabular-nums text-muted-foreground">
   829→                      <span className="font-semibold text-foreground">
   830→                        {goal.progress}
   831→                      </span>
   832→                      /{goal.target}
   833→                      {goal.unit ? ` ${goal.unit}` : ""} · {percent}%
   834→                    </p>
   835→                  </button>
   836→                );
   837→              })}
   838→            </div>
   839→          )}
   840→        </section>
   841→      </FadeIn>
   842→
   843→      <FadeIn delay={0.28}>
   844→        <section aria-label="Recent journal">
   845→          <SectionHeading
   846→            title="Recent journal"
   847→            action={
   848→              <SeeAllButton
   849→                label="Open diary"
   850→                onClick={() => setView("diary")}
   851→              />
   852→            }
   853→          />
   854→          {recentJournal.length === 0 ? (
   855→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   856→              No entries yet — tonight is a good night to write.
   857→            </p>
   858→          ) : (
   859→            <div className="divide-y divide-border/70 rounded-2xl border bg-card shadow-card">
   860→              {recentJournal.map((entry) => {
   861→                const mood = MOODS.find((m) => m.value === entry.mood);
   862→                return (
   863→                  <button
   864→                    key={entry.id}
   865→                    type="button"
   866→                    onClick={() => setView("diary")}
   867→                    aria-label={`Open diary entry from ${friendlyDay(entry.date)}`}
   868→                    className="flex w-full items-start gap-3 px-4 py-3 text-left transition-colors hover:bg-muted/40"
   869→                  >
   870→                    <span
   871→                      className="mt-0.5 flex size-9 shrink-0 items-center justify-center rounded-xl bg-muted text-base"
   872→                      aria-hidden="true"
   873→                    >
   874→                      {mood ? (
   875→                        mood.emoji
   876→                      ) : (
   877→                        <BookOpen className="size-4 text-muted-foreground" />
   878→                      )}
   879→                    </span>
   880→                    <span className="min-w-0 flex-1">
   881→                      <span className="flex items-baseline justify-between gap-2">
   882→                        <span className="truncate text-sm font-semibold">
   883→                          {entry.title || "Journal entry"}
   884→                        </span>
   885→                        <span className="shrink-0 text-[11px] text-muted-foreground">
   886→                          {friendlyDay(entry.date)}
   887→                        </span>
   888→                      </span>
   889→                      <span className="mt-0.5 block line-clamp-2 text-xs leading-relaxed text-muted-foreground">
   890→                        {entry.content}
   891→                      </span>
   892→                    </span>
   893→                  </button>
   894→                );
   895→              })}
   896→            </div>
   897→          )}
   898→        </section>
   899→      </FadeIn>
   900→    </div>
   901→  );
   902→}
