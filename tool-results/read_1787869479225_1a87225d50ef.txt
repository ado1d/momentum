   111→interface PersistedFocus {
   112→  v: 1;
   113→  mode: Mode;
   114→  durations: Record<Mode, number>;
   115→  running: boolean;
   116→  endsAt: number | null;
   117→  remainingMs: number;
   118→  startedAt: number | null;
   119→  taskId: string | null;
   120→  label: string | null;
   121→  completedToday: number;
   122→  dayKey: string;
   123→  sound: boolean;
   124→}

   126→function isMode(v: unknown): v is Mode {
   127→  return v === "focus" || v === "short" || v === "long";
   128→}
   129→
   130→function clampMinutes(v: unknown, fallback: number): number {
   131→  const n = Number(v);
   132→  if (!Number.isFinite(n)) return fallback;
   133→  return Math.max(1, Math.min(240, Math.round(n)));
   134→}
   135→
   136→function readPersisted(): PersistedFocus | null {
   137→  if (typeof window === "undefined") return null;
   138→  try {
   139→    const raw = window.localStorage.getItem(FOCUS_STORAGE_KEY);
   140→    if (!raw) return null;
   141→    const d = JSON.parse(raw) as Partial<PersistedFocus> | null;
   142→    if (!d || typeof d !== "object") return null;
   143→    const mode = isMode(d.mode) ? d.mode : "focus";
   144→    const durations: Record<Mode, number> = {
   145→      focus: clampMinutes(d.durations?.focus, DEFAULT_DURATIONS.focus),
   146→      short: clampMinutes(d.durations?.short, DEFAULT_DURATIONS.short),
   147→      long: clampMinutes(d.durations?.long, DEFAULT_DURATIONS.long),
   148→    };
   149→    return {
   150→      v: 1,
   151→      mode,
   152→      durations,
   153→      running: d.running === true && typeof d.endsAt === "number",
   154→      endsAt: typeof d.endsAt === "number" ? d.endsAt : null,
   155→      remainingMs: Number.isFinite(d.remainingMs)
   156→        ? Number(d.remainingMs)
   157→        : durations[mode] * 60_000,
   158→      startedAt: typeof d.startedAt === "number" ? d.startedAt : null,
   159→      taskId: typeof d.taskId === "string" ? d.taskId : null,
   160→      label:
   161→        typeof d.label === "string" && d.label.trim().length > 0 ? d.label : null,
   162→      completedToday:
   163→        typeof d.completedToday === "number" && d.completedToday >= 0
   164→          ? Math.floor(d.completedToday)
   165→          : 0,
   166→      dayKey: typeof d.dayKey === "string" ? d.dayKey : todayKey(),
   167→      sound: typeof d.sound === "boolean" ? d.sound : true,
   168→    };
   169→  } catch {
   170→    return null;
   171→  }
   172→}

   176→/** Soft two-tone completion chime via the Web Audio API. */
   177→function playChime() {
   178→  try {
   179→    const w = window as Window & { webkitAudioContext?: typeof AudioContext };
   180→    const Ctx = window.AudioContext ?? w.webkitAudioContext;
   181→    if (!Ctx) return;
   182→    const ctx = new Ctx();
   183→    if (ctx.state === "suspended") void ctx.resume();
   184→    const tone = (freq: number, delay: number, duration: number) => {
   185→      const osc = ctx.createOscillator();
   186→      const gain = ctx.createGain();
   187→      osc.type = "sine";
   188→      const t0 = ctx.currentTime + delay;
   189→      osc.frequency.setValueAtTime(freq, t0);
   190→      gain.gain.setValueAtTime(0.0001, t0);
   191→      gain.gain.exponentialRampToValueAtTime(0.2, t0 + 0.03);
   192→      gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);
   193→      osc.connect(gain);
   194→      gain.connect(ctx.destination);
   195→      osc.start(t0);
   196→      osc.stop(t0 + duration + 0.05);
   197→    };
   198→    tone(880, 0, 0.3); // A5
   199→    tone(1174.66, 0.18, 0.5); // D6
   200→    window.setTimeout(() => {
   201→      ctx.close().catch(() => undefined);
   202→    }, 1400);
   203→  } catch {
   204→    /* Audio unavailable — chime is best-effort */
   205→  }
   206→}

   208→/** MM:SS countdown display (ceil, so a fresh timer shows exactly 25:00). */
   209→function formatClock(ms: number): string {
   210→  const totalSeconds = Math.max(0, Math.ceil(ms / 1000));
   211→  const m = Math.floor(totalSeconds / 60);
   212→  const s = totalSeconds % 60;
   213→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
   214→}

   355→export function FocusView() {
   356→  const queryClient = useQueryClient();
   357→  const setView = useUiStore((s) => s.setView);
   358→  const { resolvedTheme } = useTheme();
   359→  const reduceMotion = useReducedMotion();
   360→
   361→  const statsQuery = useQuery({
   362→    queryKey: ["focus", "stats"],
   363→    queryFn: focusApi.stats,
   364→  });
   365→  const todosQuery = useQuery({
   366→    queryKey: ["todos", "active"],
   367→    queryFn: () => todosApi.list({ status: "active" }),
   368→  });
   369→  const todos = todosQuery.data ?? [];
   370→
   371→  // ── Timer state ──────────────────────────────────────────────
   372→  const [initial] = React.useState(readPersisted);
   373→  const [mode, setMode] = React.useState<Mode>(initial?.mode ?? "focus");
   374→  const [durations, setDurations] = React.useState<Record<Mode, number>>(
   375→    initial?.durations ?? DEFAULT_DURATIONS
   376→  );
   377→  const [running, setRunning] = React.useState(false);
   378→  const [endsAt, setEndsAt] = React.useState<number | null>(null);
   379→  const [pausedRemaining, setPausedRemaining] = React.useState<number>(() => {
   380→    if (!initial) return DEFAULT_DURATIONS.focus * 60_000;
   381→    const full = initial.durations[initial.mode] * 60_000;
   382→    if (initial.running && initial.endsAt !== null) {
   383→      return Math.min(full, Math.max(0, initial.endsAt - Date.now()));
   384→    }
   385→    return Math.min(full, Math.max(0, initial.remainingMs));
   386→  });
   387→  const [startedAt, setStartedAt] = React.useState<number | null>(null);
   388→  const [taskId, setTaskId] = React.useState<string | null>(initial?.taskId ?? null);
   389→  const [label, setLabel] = React.useState<string | null>(initial?.label ?? null);
   390→  const [completedToday, setCompletedToday] = React.useState<number>(
   391→    initial && initial.dayKey === todayKey() ? initial.completedToday : 0
   392→  );
   393→  const [sound, setSound] = React.useState(initial?.sound ?? true);
   394→  const [now, setNow] = React.useState<number>(() => Date.now());
   395→
   396→  // Task picker / duration picker UI state
   397→  const [customOpen, setCustomOpen] = React.useState(false);
   398→  const [customValue, setCustomValue] = React.useState("");
   399→  const [durationOpen, setDurationOpen] = React.useState(false);
   400→  const [customMinutes, setCustomMinutes] = React.useState("");
   401→
   402→  const completingRef = React.useRef(false);
   403→  const customInputRef = React.useRef<HTMLInputElement>(null);
   404→
   405→  // ── Session logging ──────────────────────────────────────────
   406→  const logMutation = useMutation({
   407→    mutationFn: (input: FocusSessionInput) => focusApi.log(input),
   408→    onSuccess: () => {
   409→      queryClient.invalidateQueries({ queryKey: ["focus"] });
   410→      queryClient.invalidateQueries({ queryKey: ["stats"] });
   411→    },
   412→    onError: (err, input) => {
   413→      const status = err instanceof ApiError ? err.status : 0;
   414→      if (status === 404 && input.taskId) {
   415→        // The linked task was deleted — save the session with its
   416→        // title as the label instead of losing it.
   417→        const title = todos.find((t) => t.id === input.taskId)?.title;
   418→        focusApi
   419→          .log({ ...input, taskId: null, label: title ?? input.label ?? "Focus session" })
   420→          .then(() => {
   421→            queryClient.invalidateQueries({ queryKey: ["focus"] });
   422→            queryClient.invalidateQueries({ queryKey: ["stats"] });
   423→          })
   424→          .catch(() => toast.error("Couldn't save your focus session"));
   425→        return;
   426→      }
   427→      toast.error("Couldn't save your focus session");
   428→    },
   429→  });
   430→
   431→  const logSession = React.useCallback(
   432→    (
   433→      minutes: number,
   434→      started: number | null,
   435→      ended: number,
   436→      tId: string | null,
   437→      lbl: string | null
   438→    ) => {
   439→      const input: FocusSessionInput = {
   440→        minutes,
   441→        taskId: tId,
   442→        label: lbl,
   443→        endedAt: new Date(ended).toISOString(),
   444→      };
   445→      if (started !== null) input.startedAt = new Date(started).toISOString();
   446→      logMutation.mutate(input);
   447→    },
   448→    [logMutation]
   449→  );
   450→
   451→  // ── Derived values ───────────────────────────────────────────
   452→  const totalMs = durations[mode] * 60_000;
   453→  const remainingMs =
   454→    running && endsAt !== null ? Math.max(0, endsAt - now) : pausedRemaining;
   455→  const progress = totalMs > 0 ? Math.max(0, Math.min(1, remainingMs / totalMs)) : 0;
   456→  const sessionActive = running || pausedRemaining < totalMs || startedAt !== null;
   457→  const minutesLeft = Math.max(0, Math.ceil(remainingMs / 60_000));
   458→  const selectedTodo = taskId ? todos.find((t) => t.id === taskId) ?? null : null;
   459→  const subject = selectedTodo ? selectedTodo.title : label;
   460→  const stats = statsQuery.data;
   461→
   462→  const isDark = resolvedTheme === "dark";
   463→  const ringColor =
   464→    mode === "focus"
   465→      ? "var(--primary)"
   466→      : mode === "short"
   467→        ? isDark
   468→          ? "#fbbf24"
   469→          : "#f59e0b"
   470→        : isDark
   471→          ? "#2dd4bf"
   472→          : "#14b8a6";
   473→
   474→  // ── Timer engine ─────────────────────────────────────────────
   475→
   476→  /** Natural completion: chime, toast, log, dots, auto-switch to break. */
   477→  const finishNatural = () => {
   478→    const endedAtMs = endsAt ?? Date.now();
   479→    const started = startedAt;
   480→    const tId = taskId;
   481→    const lbl = label;
   482→    const currentMode = mode;
   483→    const currentDurations = durations;
   484→    if (sound) playChime();
   485→    setRunning(false);
   486→    setEndsAt(null);
   487→    setStartedAt(null);
   488→    if (currentMode === "focus") {
   489→      const minutes = Math.max(1, Math.round(totalMs / 60_000));
   490→      const nextCount = completedToday + 1;
   491→      setCompletedToday(nextCount);
   492→      toast.success("Focus session complete — take a break! 🎉");
   493→      logSession(minutes, started, endedAtMs, tId, lbl);
   494→      const next: Mode = nextCount > 0 && nextCount % 4 === 0 ? "long" : "short";
   495→      setMode(next);
   496→      setPausedRemaining(currentDurations[next] * 60_000);
   497→    } else {
   498→      toast("Break's over — ready for another round? ☕");
   499→      setMode("focus");
   500→      setPausedRemaining(currentDurations.focus * 60_000);
   501→    }
   502→    completingRef.current = false;
   503→  };
   504→
   505→  // Keep a ref to the latest completion handler so the interval
   506→  // callback never fires a stale closure (e.g. pre-task-selection).
   507→  const finishRef = React.useRef(finishNatural);
   508→  React.useEffect(() => {
   509→    finishRef.current = finishNatural;
   510→  });
   511→
   512→  React.useEffect(() => {
   513→    if (!running || endsAt === null) return;
   514→    const tick = () => {
   515→      const t = Date.now();
   516→      setNow(t);
   517→      if (t >= endsAt && !completingRef.current) {
   518→        completingRef.current = true;
   519→        finishRef.current();
   520→      }
   521→    };
   522→    tick();
   523→    const id = window.setInterval(tick, 250);
   524→    return () => window.clearInterval(id);
   525→  }, [running, endsAt]);
   526→
   527→  // ── Restore on mount ─────────────────────────────────────────
   528→  React.useEffect(() => {
   529→    if (!initial || !initial.running || initial.endsAt === null) return;
   530→    if (initial.endsAt > Date.now()) {
   531→      // Resume the running session right where it left off.
   532→      setRunning(true);
   533→      setEndsAt(initial.endsAt);
   534→      setStartedAt(
   535→        initial.startedAt ??
   536→          initial.endsAt - initial.durations[initial.mode] * 60_000
   537→      );
   538→      setNow(Date.now());
   539→      return;
   540→    }
   541→    // The session expired while the page was closed — complete it.
   542→    const m = initial.mode;
   543→    const minutes = initial.durations[m];
   544→    const ended = initial.endsAt;
   545→    const baseCount = initial.dayKey === todayKey() ? initial.completedToday : 0;
   546→    if (m === "focus") {
   547→      const nextCount = baseCount + 1;
   548→      setCompletedToday(nextCount);
   549→      toast.success("Focus session complete — take a break! 🎉");
   550→      logSession(minutes, initial.startedAt, ended, initial.taskId, initial.label);
