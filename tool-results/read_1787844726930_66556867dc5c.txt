   527→export function DashboardView() {
   528→  const setView = useUiStore((s) => s.setView);
   529→  const setQuickAddOpen = useUiStore((s) => s.setQuickAddOpen);
   530→  const queryClient = useQueryClient();
   531→
   532→  const { data, isLoading, isError, refetch } = useQuery({
   533→    queryKey: ["stats", "dashboard"],
   534→    queryFn: statsApi.dashboard,
   535→  });
   536→
   537→  const weekDays = React.useMemo(() => lastNDays(7), []);
   538→
   539→  // Toggle a focus todo with an optimistic update so the check feels instant.
   540→  const toggleTodo = useMutation({
   541→    mutationFn: (todo: Todo) =>
   542→      todosApi.update(todo.id, { completed: !todo.completed }),
   543→    onMutate: async (todo) => {
   544→      await queryClient.cancelQueries({ queryKey: ["stats", "dashboard"] });
   545→      const prev = queryClient.getQueryData<DashboardStats>([
   546→        "stats",
   547→        "dashboard",
   548→      ]);
   549→      if (prev) {
   550→        queryClient.setQueryData<DashboardStats>(["stats", "dashboard"], {
   551→          ...prev,
   552→          upcomingTodos: prev.upcomingTodos.map((t) =>
   553→            t.id === todo.id
   554→              ? {
   555→                  ...t,
   556→                  completed: !t.completed,
   557→                  completedAt: !t.completed
   558→                    ? new Date().toISOString()
   559→                    : null,
   560→                }
   561→              : t
   562→          ),
   563→        });
   564→      }
   565→      return { prev };
   566→    },
   567→    onError: (err, _todo, ctx) => {
   568→      if (ctx?.prev) {
   569→        queryClient.setQueryData(["stats", "dashboard"], ctx.prev);
   570→      }
   571→      toast.error(
   572→        err instanceof Error ? err.message : "Couldn't update the task"
   573→      );
   574→    },
   575→    onSettled: () => {
   576→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   577→      void queryClient.invalidateQueries({ queryKey: ["todos"] });
   578→    },
   579→  });
   580→
   581→  // Toggle a habit for today, optimistically flipping the chip + week dots.
   582→  const toggleHabit = useMutation({
   583→    mutationFn: (habit: Habit) => habitsApi.toggle(habit.id, todayKey()),
   584→    onMutate: async (habit) => {
   585→      await queryClient.cancelQueries({ queryKey: ["stats", "dashboard"] });
   586→      const prev = queryClient.getQueryData<DashboardStats>([
   587→        "stats",
   588→        "dashboard",
   589→      ]);
   590→      const today = todayKey();
   591→      if (prev) {
   592→        queryClient.setQueryData<DashboardStats>(["stats", "dashboard"], {
   593→          ...prev,
   594→          todayHabits: prev.todayHabits.map((h) => {
   595→            if (h.id !== habit.id) return h;
   596→            const done = !h.doneToday;
   597→            const logs = done
   598→              ? [...h.logs, { id: `tmp-${h.id}`, habitId: h.id, date: today }]
   599→              : h.logs.filter((l) => l.date !== today);
   600→            return { ...h, doneToday: done, logs };
   601→          }),
   602→        });
   603→      }
   604→      return { prev };
   605→    },
   606→    onError: (err, _habit, ctx) => {
   607→      if (ctx?.prev) {
   608→        queryClient.setQueryData(["stats", "dashboard"], ctx.prev);
   609→      }
   610→      toast.error(
   611→        err instanceof Error ? err.message : "Couldn't toggle the habit"
   612→      );
   613→    },
   614→    onSettled: () => {
   615→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   616→      void queryClient.invalidateQueries({ queryKey: ["habits"] });
   617→    },
   618→  });
   619→
   620→  // Onboarding: create three starter habits in one go.
   621→  const addStarterHabits = useMutation({
   622→    mutationFn: async () => {
   623→      await habitsApi.create({
   624→        name: "Drink 8 glasses of water",
   625→        emoji: "💧",
   626→        color: "teal",
   627→        timeOfDay: "anytime",
   628→      });
   629→      await habitsApi.create({
   630→        name: "Read 20 minutes",
   631→        emoji: "📚",
   632→        color: "amber",
   633→        timeOfDay: "evening",
   634→      });
   635→      await habitsApi.create({
   636→        name: "Move your body",
   637→        emoji: "🏃",
   638→        color: "rose",
   639→        timeOfDay: "morning",
   640→      });
   641→    },
   642→    onSuccess: () => {
   643→      toast.success("Starter habits added", {
   644→        description: "Find them under Routine and check them off daily.",
   645→      });
   646→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   647→      void queryClient.invalidateQueries({ queryKey: ["habits"] });
   648→    },
   649→    onError: (err) =>
   650→      toast.error(
   651→        err instanceof Error ? err.message : "Couldn't add starter habits"
   652→      ),
   653→  });
   654→
   655→  if (isLoading) return <DashboardSkeleton />;
   656→  if (isError || !data) {
   657→    return <DashboardError onRetry={() => void refetch()} />;
   658→  }
   659→
   660→  const { today, week, activeGoals, upcomingTodos, todayHabits, recentJournal, quote } =
   661→    data;
   662→  const isEmpty =
   663→    today.todosTotal === 0 &&
   664→    today.habitsTotal === 0 &&
   665→    activeGoals.length === 0 &&
   666→    today.routineTotal === 0;
   667→
   668→  if (isEmpty) {
   669→    return (
   670→      <div className="space-y-6">
   671→        <FadeIn>
   672→          <DashboardHeader score={today.score} />
   673→        </FadeIn>
   674→        <FadeIn delay={0.05}>
   675→          <OnboardingCard
   676→            habitsPending={addStarterHabits.isPending}
   677→            onAddTask={() => setQuickAddOpen(true)}
   678→            onAddHabits={() => addStarterHabits.mutate()}
   679→            onSetGoal={() => setView("goals")}
   680→          />
   681→        </FadeIn>
   682→        <FadeIn delay={0.1}>
   683→          <QuoteCard quote={quote} />
   684→        </FadeIn>
   685→      </div>
   686→    );
   687→  }
   688→
   689→  return (
   690→    <div className="space-y-6">
   691→      <FadeIn>
   692→        <DashboardHeader score={today.score} />
   693→      </FadeIn>
   694→
   695→      <FadeIn delay={0.04}>
   696→        <section aria-label="Quote of the day">
   697→          <QuoteCard quote={quote} />
   698→        </section>
   699→      </FadeIn>
   700→
   701→      <FadeIn delay={0.08}>
   702→        <StatCards today={today} />
   703→      </FadeIn>
   704→
   705→      <FadeIn delay={0.12}>
   706→        <WeekCard week={week} />
   707→      </FadeIn>
   708→
   709→      {today.overdueCount > 0 && (
   710→        <FadeIn delay={0.14}>
   711→          <OverdueBanner
   712→            count={today.overdueCount}
   713→            onReview={() => setView("tasks")}
   714→          />
   715→        </FadeIn>
   716→      )}
   717→
   718→      <FadeIn delay={0.16}>
   719→        <section aria-label="Today's focus">
   720→          <SectionHeading
   721→            title="Today's focus"
   722→            action={<SeeAllButton onClick={() => setView("tasks")} />}
   723→          />
   724→          {upcomingTodos.length === 0 ? (
   725→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   726→              Nothing due in the next week — enjoy the breathing room.
   727→            </p>
   728→          ) : (
   729→            <div className="divide-y divide-border/70 rounded-2xl border bg-card shadow-sm">
   730→              {upcomingTodos.slice(0, 6).map((todo) => (
   731→                <FocusRow
   732→                  key={todo.id}
   733→                  todo={todo}
   734→                  onToggle={(t) => toggleTodo.mutate(t)}
   735→                />
   736→              ))}
   737→            </div>
   738→          )}
   739→        </section>
   740→      </FadeIn>
   741→
   742→      <FadeIn delay={0.2}>
   743→        <section aria-label="Habits today">
   744→          <SectionHeading
   745→            title="Habits today"
   746→            action={<SeeAllButton onClick={() => setView("routine")} />}
   747→          />
   748→          {todayHabits.length === 0 ? (
   749→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   750→              No habits yet — build momentum with one small daily win.
   751→            </p>
   752→          ) : (
   753→            <div className="-mx-1 flex gap-3 overflow-x-auto px-1 pb-2 no-scrollbar">
   754→              {todayHabits.map((habit) => (
   755→                <HabitChip
   756→                  key={habit.id}
   757→                  habit={habit}
   758→                  weekDays={weekDays}
   759→                  onToggle={(h) => toggleHabit.mutate(h)}
   760→                />
   761→              ))}
   762→            </div>
   763→          )}
   764→        </section>
   765→      </FadeIn>
   766→
   767→      <FadeIn delay={0.24}>
   768→        <section aria-label="Active goals">
   769→          <SectionHeading
   770→            title="Active goals"
   771→            action={<SeeAllButton onClick={() => setView("goals")} />}
   772→          />
   773→          {activeGoals.length === 0 ? (
   774→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   775→              No active goals — set one to give your days direction.
   776→            </p>
   777→          ) : (
   778→            <div className="grid gap-3 sm:grid-cols-2">
   779→              {activeGoals.slice(0, 4).map((goal) => {
   780→                const percent = ratio(goal.progress, goal.target);
   781→                return (
   782→                  <button
   783→                    key={goal.id}
   784→                    type="button"
   785→                    onClick={() => setView("goals")}
   786→                    aria-label={`Open goal: ${goal.title}`}
   787→                    className="rounded-2xl border bg-card p-4 text-left shadow-sm transition-all hover:-translate-y-0.5 hover:border-primary/40 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50 active:scale-[0.99]"
   788→                  >
   789→                    <div className="flex items-start justify-between gap-2">
   790→                      <p className="min-w-0 text-sm font-semibold leading-snug line-clamp-1">
   791→                        {goal.title}
   792→                      </p>
   793→                      <Badge
   794→                        variant="outline"
   795→                        className="shrink-0 rounded-full px-2 py-0 text-[10px] font-medium capitalize"
   796→                      >
   797→                        {goal.period}
   798→                      </Badge>
   799→                    </div>
   800→                    <ProgressBar value={percent} className="mt-3" />
   801→                    <p className="mt-1.5 text-xs tabular-nums text-muted-foreground">
   802→                      <span className="font-semibold text-foreground">
   803→                        {goal.progress}
   804→                      </span>
   805→                      /{goal.target}
   806→                      {goal.unit ? ` ${goal.unit}` : ""} · {percent}%
   807→                    </p>
   808→                  </button>
   809→                );
   810→              })}
   811→            </div>
   812→          )}
   813→        </section>
   814→      </FadeIn>
   815→
   816→      <FadeIn delay={0.28}>
   817→        <section aria-label="Recent journal">
   818→          <SectionHeading
   819→            title="Recent journal"
   820→            action={
   821→              <SeeAllButton
   822→                label="Open diary"
   823→                onClick={() => setView("diary")}
   824→              />
   825→            }
   826→          />
   827→          {recentJournal.length === 0 ? (
   828→            <p className="rounded-2xl border border-dashed bg-muted/30 px-4 py-6 text-center text-sm text-balance text-muted-foreground">
   829→              No entries yet — tonight is a good night to write.
   830→            </p>
   831→          ) : (
   832→            <div className="divide-y divide-border/70 rounded-2xl border bg-card shadow-sm">
   833→              {recentJournal.map((entry) => {
   834→                const mood = MOODS.find((m) => m.value === entry.mood);
   835→                return (
   836→                  <button
   837→                    key={entry.id}
   838→                    type="button"
   839→                    onClick={() => setView("diary")}
   840→                    aria-label={`Open diary entry from ${friendlyDay(entry.date)}`}
   841→                    className="flex w-full items-start gap-3 px-4 py-3 text-left transition-colors hover:bg-muted/40"
   842→                  >
   843→                    <span
   844→                      className="mt-0.5 flex size-9 shrink-0 items-center justify-center rounded-xl bg-muted text-base"
   845→                      aria-hidden="true"
   846→                    >
   847→                      {mood ? (
   848→                        mood.emoji
   849→                      ) : (
   850→                        <BookOpen className="size-4 text-muted-foreground" />
   851→                      )}
   852→                    </span>
   853→                    <span className="min-w-0 flex-1">
   854→                      <span className="flex items-baseline justify-between gap-2">
   855→                        <span className="truncate text-sm font-semibold">
   856→                          {entry.title || "Journal entry"}
   857→                        </span>
   858→                        <span className="shrink-0 text-[11px] text-muted-foreground">
   859→                          {friendlyDay(entry.date)}
   860→                        </span>
   861→                      </span>
   862→                      <span className="mt-0.5 block line-clamp-2 text-xs leading-relaxed text-muted-foreground">
   863→                        {entry.content}
   864→                      </span>
   865→                    </span>
   866→                  </button>
   867→                );
   868→              })}
   869→            </div>
   870→          )}
   871→        </section>
   872→      </FadeIn>
   873→    </div>
   874→  );
