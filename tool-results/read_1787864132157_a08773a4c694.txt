    60→/** 125 → "2h 5m", 45 → "45m", 120 → "2h" */
    61→function formatMinutes(total: number): string {
    62→  if (total <= 0) return "0m";
    63→  const h = Math.floor(total / 60);
    64→  const m = total % 60;
    65→  if (h === 0) return `${m}m`;
    66→  if (m === 0) return `${h}h`;
    67→  return `${h}h ${m}m`;
    68→}

    70→/** "Aug 24 – 30, 2026" (dialog header). */
    71→function rangeLabel(startKey: string, endKey: string): string {
    72→  const s = keyToDate(startKey);
    73→  const e = keyToDate(endKey);
    74→  if (!isValid(s) || !isValid(e)) return `${startKey} – ${endKey}`;
    75→  if (s.getFullYear() !== e.getFullYear()) {
    76→    return `${format(s, "MMM d, yyyy")} – ${format(e, "MMM d, yyyy")}`;
    77→  }
    78→  if (s.getMonth() === e.getMonth()) {
    79→    return `${format(s, "MMM d")} – ${format(e, "d, yyyy")}`;
    80→  }
    81→  return `${format(s, "MMM d")} – ${format(e, "MMM d, yyyy")}`;
    82→}
    83→
    84→/** "Aug 24–30, 2026" (compact, for export titles / filenames). */
    85→function compactRangeLabel(startKey: string, endKey: string): string {
    86→  const s = keyToDate(startKey);
    87→  const e = keyToDate(endKey);
    88→  if (!isValid(s) || !isValid(e)) return `${startKey}–${endKey}`;
    89→  if (s.getFullYear() === e.getFullYear() && s.getMonth() === e.getMonth()) {
    90→    return `${format(s, "MMM d")}–${format(e, "d, yyyy")}`;
    91→  }
    92→  return rangeLabel(startKey, endKey);
    93→}

   294→function ScoreDelta({ avg, prev }: { avg: number; prev: number }) {
   295→  const delta = avg - prev;
   296→  if (delta === 0) {
   297→    return (
   298→      <span className="inline-flex items-center gap-1 rounded-full bg-muted px-2 py-0.5 text-xs font-medium text-muted-foreground">
   299→        Same as last week
   300→      </span>
   301→    );
   302→  }
   303→  const up = delta > 0;
   304→  return (
   305→    <span
   306→      className={cn(
   307→        "inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-xs font-semibold tabular-nums",
   308→        up
   309→          ? "bg-emerald-500/15 text-emerald-700 dark:text-emerald-300"
   310→          : "bg-rose-500/15 text-rose-700 dark:text-rose-300"
   311→      )}
   312→    >
   313→      {up ? (
   314→        <TrendingUp className="size-3" aria-hidden="true" />
   315→      ) : (
   316→        <TrendingDown className="size-3" aria-hidden="true" />
   317→      )}
   318→      {up ? "+" : "−"}
   319→      {Math.abs(delta)} vs last week {prev}
   320→    </span>
   321→  );
   322→}

   491→export function ReviewDialog({ open, onOpenChange }: ReviewDialogProps) {
   492→  const [anchor, setAnchor] = React.useState(todayKey());
   493→
   494→  // Always start on the current week when the dialog opens.
   495→  React.useEffect(() => {
   496→    if (open) setAnchor(todayKey());
   497→  }, [open]);
   498→
   499→  const prevAnchor = addDaysToKey(anchor, -7);
   500→  const nextAnchor = addDaysToKey(anchor, 7);
   501→  const nextDisabled = nextAnchor > todayKey(); // no future weeks
   502→
   503→  const { data, isLoading, isError, refetch } = useQuery({
   504→    queryKey: ["review", anchor],
   505→    queryFn: () => reviewApi.get(anchor),
   506→    enabled: open,
   507→    staleTime: 30_000,
   508→    placeholderData: keepPreviousData,
   509→  });
   510→
   511→  const handleExportPdf = () => {
   512→    if (!data) return;
