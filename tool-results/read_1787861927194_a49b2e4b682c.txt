   153→function TodayBanner({ habits }: { habits: Habit[] }) {
   154→  const total = habits.length;
   155→  const done = habits.filter((h) => h.doneToday).length;
   156→  const pct = total > 0 ? Math.round((done / total) * 100) : 0;
   157→  const allDone = total > 0 && done === total;
   158→  const remaining = total - done;
   159→
   160→  return (
   161→    <Card className="overflow-hidden rounded-2xl border-primary/20 bg-gradient-to-br from-primary/10 via-card to-card shadow-card">
   162→      <CardContent className="flex items-center gap-4 p-4 sm:p-5">
   163→        <ProgressRing
   164→          value={pct}
   165→          size={64}
   166→          strokeWidth={7}
   167→          label={String(done)}
   168→          sublabel={`of ${total}`}
   169→          className="shrink-0"
   170→        />
   171→        <div className="min-w-0 flex-1">
   172→          <p className="text-sm font-semibold sm:text-base">
   173→            <span className="tabular-nums">{done}</span> of{" "}
   174→            <span className="tabular-nums">{total}</span> habits done today
   175→          </p>
   176→          <ProgressBar value={pct} className="mt-2 h-2.5" />
   177→          <p className="mt-2 text-xs text-muted-foreground">
   178→            {allDone
   179→              ? "Perfect day! 🎉"
   180→              : remaining === 1
   181→                ? "Just one more — you are almost there."
   182→                : `${remaining} to go. Keep the momentum!`}
   183→          </p>
   184→        </div>
   185→      </CardContent>
   186→    </Card>
   187→  );
   188→}

   305→
   306→interface HabitFormDialogProps {
   307→  open: boolean;
   308→  onOpenChange: (open: boolean) => void;
   309→  habit: Habit | null;
   310→  submitting: boolean;
   311→  onSubmit: (values: HabitFormValues) => void;
   312→}
   313→
   314→function HabitFormDialog({ open, onOpenChange, habit, submitting, onSubmit }: HabitFormDialogProps) {
   315→  const [values, setValues] = React.useState<HabitFormValues>(EMPTY_HABIT);
   316→
   317→  React.useEffect(() => {
   318→    if (open) {
   319→      setValues(
   320→        habit
   321→          ? {
   322→              name: habit.name,
   323→              emoji: habit.emoji,
   324→              color: habit.color,
   325→              timeOfDay: habit.timeOfDay,
   326→              reminderTime: habit.reminderTime ?? "",
   327→            }
   328→          : EMPTY_HABIT
   329→      );
   330→    }
   331→  }, [open, habit]);
   332→
   333→  const set = <K extends keyof HabitFormValues>(key: K, value: HabitFormValues[K]) =>
   334→    setValues((prev) => ({ ...prev, [key]: value }));
   335→
   336→  const submit = (e: React.FormEvent) => {
   337→    e.preventDefault();
   338→    if (!values.name.trim()) {
   339→      toast.error("Give your habit a name");
   340→      return;
   341→    }
   342→    onSubmit({
   343→      ...values,
   344→      name: values.name.trim(),
   345→      emoji: values.emoji.trim() || "✨",
   346→      reminderTime: values.reminderTime.trim(),
   347→    });
   348→  };
   349→
   350→  return (
   351→    <Dialog open={open} onOpenChange={onOpenChange}>
   352→      <DialogContent className="max-h-[88dvh] overflow-y-auto rounded-2xl sm:max-w-md">
   353→        <DialogHeader>
   354→          <DialogTitle>{habit ? "Edit habit" : "New habit"}</DialogTitle>
   355→          <DialogDescription>
   356→            {habit
   357→              ? "Update the details of this habit."
   358→              : "Pick something small enough to do every day."}
   359→          </DialogDescription>
   360→        </DialogHeader>
   361→
   362→        <form onSubmit={submit} className="space-y-5">
   363→          <div className="space-y-2">
   364→            <Label htmlFor="habit-name">Name</Label>
   365→            <Input
   366→              id="habit-name"
   367→              value={values.name}
   368→              onChange={(e) => set("name", e.target.value)}
   369→              placeholder="e.g. Drink more water"
   370→              maxLength={80}
   371→              autoFocus
   372→            />
   373→          </div>
   374→
   375→          <div className="space-y-2">
   376→            <Label>Icon</Label>
   377→            <div className="grid grid-cols-6 gap-1.5 sm:grid-cols-8">
   378→              {HABIT_EMOJIS.map((emoji) => (
   379→                <button
   380→                  key={emoji}
   381→                  type="button"
   382→                  onClick={() => set("emoji", emoji)}
   383→                  aria-label={`Icon ${emoji}`}
   384→                  aria-pressed={values.emoji === emoji}
   385→                  className={cn(
   386→                    "flex h-11 items-center justify-center rounded-xl border text-lg transition-all active:scale-90",
   387→                    "focus-visible:outline-none focus-visible:ring-[3px] focus-visible:ring-ring/50",
