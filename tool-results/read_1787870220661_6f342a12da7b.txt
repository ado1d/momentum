    98→    98→/** "Edited 2h ago" style relative timestamps */
    99→    99→function relativeTime(iso: string): string {
   100→   100→  const ms = Date.now() - new Date(iso).getTime();
   101→   101→  if (Number.isNaN(ms)) return "some time ago";
   102→   102→  const mins = Math.floor(ms / 60_000);
   103→   103→  if (mins < 1) return "just now";
   104→   104→  if (mins < 60) return `${mins}m ago`;
   105→   105→  const hours = Math.floor(mins / 60);
   106→   106→  if (hours < 24) return `${hours}h ago`;
   107→   107→  const days = Math.floor(hours / 24);
   108→   108→  if (days < 7) return `${days}d ago`;
   109→   109→  const weeks = Math.floor(days / 7);
   110→   110→  if (weeks < 5) return `${weeks}w ago`;
   111→   111→  return new Date(iso).toLocaleDateString(undefined, {
   112→   112→    month: "short",
   113→   113→    day: "numeric",
   114→   114→  });
   115→   115→}

   496→   496→  const { data: notes = [], isLoading, isError, refetch } = useQuery({
   497→   497→    queryKey: ["notes"],
   498→   498→    queryFn: notesApi.list,
   499→   499→  });
   500→   500→
   501→   501→  const [search, setSearch] = React.useState("");
   502→   502→  const [activeTag, setActiveTag] = React.useState<string>("all");
   503→   503→
   504→   504→  const [dialogOpen, setDialogOpen] = React.useState(false);
   505→   505→  const [editingNote, setEditingNote] = React.useState<Note | null>(null);
   506→   506→  const [noteToDelete, setNoteToDelete] = React.useState<Note | null>(null);
   507→   507→  const [exporting, setExporting] = React.useState<string | null>(null);
   508→   508→
   509→   509→  const invalidate = (keys: string[]) => {
   510→   510→    for (const key of keys) void queryClient.invalidateQueries({ queryKey: [key] });
   511→   511→  };
   512→   512→
   513→   513→  // ── Derived data ──
   514→   514→
   515→   515→  const tags = React.useMemo(
   516→   516→    () =>
   517→   517→      Array.from(
   518→   518→        new Set(notes.map((n) => n.tag).filter((t): t is string => !!t))
   519→   519→      ).sort((a, b) => a.localeCompare(b)),
   520→   520→    [notes]
   521→   521→  );
   522→   522→
   523→   523→  const query = search.trim().toLowerCase();
   524→   524→  const filtered = notes.filter((n) => {
   525→   525→    if (activeTag !== "all" && n.tag !== activeTag) return false;
   526→   526→    if (!query) return true;
   527→   527→    return (
   528→   528→      n.title.toLowerCase().includes(query) ||
   529→   529→      n.content.toLowerCase().includes(query) ||
   530→   530→      (n.tag ?? "").toLowerCase().includes(query)
   531→   531→    );
   532→   532→  });
   533→   533→
   534→   534→  // ── Mutations ──
   535→   535→
   536→   536→  const saveNote = useMutation({
   537→   537→    mutationFn: (vars: { id?: string; input: NoteInput }) =>
   538→   538→      vars.id ? notesApi.update(vars.id, vars.input) : notesApi.create(vars.input),
   539→   539→    onSuccess: (_result, vars) => {
   540→   540→      invalidate(["notes", "stats"]);
   541→   541→      toast.success(vars.id ? "Note updated" : "Note created");
   542→   542→      setDialogOpen(false);
   543→   543→    },
   544→   544→    onError: (e) => toast.error(e.message || "Could not save note"),
   545→   545→  });
   546→   546→
   547→   547→  const patchNote = useMutation({
   548→   548→    mutationFn: (vars: { note: Note; patch: NoteInput }) =>
   549→   549→      notesApi.update(vars.note.id, vars.patch),
   550→   550→    onMutate: async ({ note, patch }) => {
   551→   551→      await queryClient.cancelQueries({ queryKey: ["notes"] });
   552→   552→      const previous = queryClient.getQueryData<Note[]>(["notes"]);
   553→   553→      queryClient.setQueryData<Note[]>(["notes"], (old) =>
   554→   554→        (old ?? []).map((n) => (n.id === note.id ? { ...n, ...patch } : n))
   555→   555→      );
   556→   556→      return { previous };
   557→   557→    },
   558→   558→    onError: (error, _vars, context) => {
   559→   559→      if (context?.previous) queryClient.setQueryData(["notes"], context.previous);
   560→   560→      toast.error(error.message || "Could not update note");
   561→   561→    },
   562→   562→    onSettled: () => invalidate(["notes", "stats"]),
   563→   563→  });
   564→   564→
   565→   565→  const removeNote = useMutation({
   566→   566→    mutationFn: (id: string) => notesApi.remove(id),
   567→   567→    onSuccess: () => {
   568→   568→      invalidate(["notes", "stats"]);
   569→   569→      toast.success("Note deleted");
   570→   570→      setNoteToDelete(null);
   571→   571→    },
   572→   572→    onError: (e) => toast.error(e.message || "Could not delete note"),
   573→   573→  });
   574→   574→
   575→   575→  // ── Handlers ──
   576→   576→
   577→   577→  const openNewNote = () => {
   578→   578→    setEditingNote(null);
   579→   579→    setDialogOpen(true);
   580→   580→  };
   581→   581→
   582→   582→  const openEditNote = (note: Note) => {
   583→   583→    setEditingNote(note);
   584→   584→    setDialogOpen(true);
   585→   585→  };
   586→   586→
   587→   587→  const submitNote = (values: NoteFormValues) => {
   588→   588→    saveNote.mutate({
   589→   589→      id: editingNote?.id,
   590→   590→      input: {
   591→   591→        title: values.title.trim() || "Untitled note",
   592→   592→        content: values.content,
   593→   593→        tag: values.tag.trim().toLowerCase() || null,
   594→   594→        color: values.color,
   595→   595→        pinned: values.pinned,
   596→   596→      },
   597→   597→    });
   598→   598→  };
   599→   599→
   600→   600→  const togglePin = (note: Note) => {
   601→   601→    patchNote.mutate({
   602→   602→      note,
   603→   603→      patch: { pinned: !note.pinned },
   604→   604→    });
   605→   605→  };
   606→   606→
   607→   607→  const runExport = async (
   608→   608→    id: string,
   609→   609→    label: string,
   610→   610→    fn: () => Promise<void> | void
   611→   611→  ) => {
   612→   612→    setExporting(id);
   613→   613→    const toastId = toast.loading(`Preparing ${label}…`);
   614→   614→    try {
   615→   615→      await fn();
   616→   616→      toast.success(`${label} ready`, { id: toastId });
   617→   617→    } catch (e) {
   618→   618→      toast.error(
   619→   619→        e instanceof Error ? e.message : `${label} failed`,
   620→   620→        { id: toastId }
   621→   621→      );
   622→   622→    } finally {
   623→   623→      setExporting(null);
   624→   624→    }
   625→   625→  };
   626→   626→
   627→   627→  const exportMarkdown = () =>
   628→   628→    runExport("md", "Markdown export", async () => {
   629→   629→      const md = await exportApi.markdown("notes");
   630→   630→      downloadMarkdown(md, "momentum-notes.md");
   631→   631→    });
   632→   632→
   633→   633→  const exportPdf = () =>
   634→   634→    runExport("pdf", "PDF export", async () => {
   635→   635→      if (notes.length === 0) {
   636→   636→        toast.error("Nothing to export yet");
   637→   637→        return;
   638→   638→      }
   639→   639→      const html = notes
   640→   640→        .map(
   641→   641→          (n) => `
   642→   642→        <div>
   643→   643→          <div style="font-size:17px; font-weight:700; margin:18px 0 4px;">${esc(n.title || "Untitled note")}${n.pinned ? " 📌" : ""}</div>
   644→   644→          ${n.tag ? `<div style="font-size:11px; color:#777; margin-bottom:6px;"># ${esc(n.tag)}</div>` : ""}
   645→   645→          ${miniMarkdownToHtml(n.content)}
   646→   646→          <hr style="border:none; border-top:1px solid #e3e3e3; margin:16px 0 4px;" />
   647→   647→        </div>`
   648→   648→        )
   649→   649→        .join("");
   650→   650→      printHtml("Notes", html);
   651→   651→    });
   652→   652→
   653→   653→  const exportBusy = exporting !== null;
   654→   654→
   655→   655→  // ── Render ──
   656→   656→
   657→   657→  return (
   658→   658→    <div>
   659→   659→      <ViewHeader
   660→   660→        title="Notes"
   661→   661→        subtitle="Quick capture & ideas"
   662→   662→        actions={
   663→   663→          <>
   664→   664→            <DropdownMenu>
   665→   665→              <DropdownMenuTrigger asChild>
   666→   666→                <Button
   667→   667→                  variant="outline"
   668→   668→                  size="icon"
   669→   669→                  className="size-11 rounded-xl"
   670→   670→                  aria-label="Export notes"
   671→   671→                  disabled={exportBusy}
   672→   672→                >
   673→   673→                  {exportBusy ? (
   674→   674→                    <Loader2 className="animate-spin" aria-hidden="true" />
   675→   675→                  ) : (
   676→   676→                    <Download aria-hidden="true" />
   677→   677→                  )}
   678→   678→                </Button>
   679→   679→              </DropdownMenuTrigger>
   680→   680→              <DropdownMenuContent align="end" className="w-56">
   681→   681→                <DropdownMenuItem onSelect={exportMarkdown}>
   682→   682→                  <FileText aria-hidden="true" /> Export notes (Markdown)
   683→   683→                </DropdownMenuItem>
   684→   684→                <DropdownMenuItem onSelect={exportPdf}>
   685→   685→                  <Printer aria-hidden="true" /> Export notes (PDF)
   686→   686→                </DropdownMenuItem>
   687→   687→              </DropdownMenuContent>
   688→   688→            </DropdownMenu>
   689→   689→            <Button className="h-11 rounded-xl" onClick={openNewNote}>
   690→   690→              <Plus aria-hidden="true" /> New note
   691→   691→            </Button>
   692→   692→          </>
   693→   693→        }
   694→   694→      />
   695→   695→
   696→   696→      {isLoading ? (
   697→   697→        <>
   698→   698→          <div className="mb-4 space-y-3">
   699→   699→            <Skeleton className="h-11 w-full rounded-xl" />
   700→   700→            <Skeleton className="h-9 w-2/3 rounded-full" />
   701→   701→          </div>
   702→   702→          <NotesSkeleton />
   703→   703→        </>
   704→   704→      ) : isError ? (
   705→   705→        <QueryError onRetry={() => void refetch()} />
   706→   706→      ) : (
   707→   707→        <div className="space-y-4">
   708→   708→          {notes.length > 0 && (
   709→   709→            <div className="space-y-3">
   710→   710→              <div className="relative">
   711→   711→                <Search
   712→   712→                  className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   713→   713→                  aria-hidden="true"
   714→   714→                />
   715→   715→                <Input
   716→   716→                  value={search}
   717→   717→                  onChange={(e) => setSearch(e.target.value)}
   718→   718→                  placeholder="Search notes by title, content or tag…"
   719→   719→                  aria-label="Search notes"
   720→   720→                  className="h-11 rounded-xl pl-9 pr-10"
   721→   721→                />
   722→   722→                {search && (
   723→   723→                  <button
   724→   724→                    type="button"
   725→   725→                    onClick={() => setSearch("")}
   726→   726→                    className="absolute right-1.5 top-1/2 flex size-8 -translate-y-1/2 items-center justify-center rounded-lg text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"
   727→   727→                    aria-label="Clear search"
   728→   728→                  >
   729→   729→                    <X className="size-4" aria-hidden="true" />
   730→   730→                  </button>
   731→   731→                )}
   732→   732→              </div>
   733→   733→
   734→   734→              {tags.length > 0 && (
   735→   735→                <div
   736→   736→                  className="fade-edges -mx-3 flex gap-2 overflow-x-auto px-3 pb-1"
   737→   737→                  role="group"
   738→   738→                  aria-label="Filter by tag"
   739→   739→                >
   740→   740→                  <button
   741→   741→                    type="button"
   742→   742→                    onClick={() => setActiveTag("all")}
   743→   743→                    aria-pressed={activeTag === "all"}
   744→   744→                    className={cn(
   745→   745→                      "shrink-0 rounded-full border px-3.5 py-1.5 text-xs font-medium transition-colors",
   746→   746→                      activeTag === "all"
   747→   747→                        ? "border-primary bg-primary text-primary-foreground"
   748→   748→                        : "border-border bg-card text-muted-foreground hover:bg-muted hover:text-foreground"
   749→   749→                    )}
   750→   750→                  >
   751→   751→                    All ({notes.length})
   752→   752→                  </button>
   753→   753→                  {tags.map((tag) => {
   754→   754→                    const count = notes.filter((n) => n.tag === tag).length;
   755→   755→                    return (
   756→   756→                      <button
   757→   757→                        key={tag}
   758→   758→                        type="button"
   759→   759→                        onClick={() => setActiveTag(activeTag === tag ? "all" : tag)}
   760→   760→                        aria-pressed={activeTag === tag}
   761→   761→                        className={cn(
   762→   762→                          "shrink-0 rounded-full border px-3.5 py-1.5 text-xs font-medium transition-colors",
   763→   763→                          activeTag === tag
   764→   764→                            ? "border-primary bg-primary text-primary-foreground"
   765→   765→                            : "border-border bg-card text-muted-foreground hover:bg-muted hover:text-foreground"
   766→   766→                        )}
   767→   767→                      >
   768→   768→                        # {tag} ({count})
   769→   769→                      </button>
   770→   770→                    );
   771→   771→                  })}
   772→   772→                </div>
   773→   773→              )}
   774→   774→            </div>
   775→   775→          )}
   776→   776→
   777→   777→          {notes.length === 0 ? (
   778→   778→            <EmptyState
   779→   779→              icon={StickyNote}
   780→   780→              title="No notes yet"
   781→   781→              description="Capture ideas, snippets and lists before they escape. They'll be right here, pinned or searchable."
   782→   782→              actionLabel="New note"
   783→   783→              onAction={openNewNote}
   784→   784→            />
   785→   785→          ) : filtered.length === 0 ? (
   786→   786→            <EmptyState
   787→   787→              icon={Search}
   788→   788→              title="No matching notes"
   789→   789→              description={
   790→   790→                query
   791→   791→                  ? `Nothing matches “${search.trim()}”. Try a different word or clear the filters.`
   792→   792→                  : "No notes carry this tag yet."
   793→   793→              }
   794→   794→              actionLabel="Clear filters"
   795→   795→              onAction={() => {
   796→   796→                setSearch("");
   797→   797→                setActiveTag("all");
   798→   798→              }}
   799→   799→            />
   800→   800→          ) : (
   801→   801→            <div className="stagger-list columns-1 gap-4 sm:columns-2">
   802→   802→              {filtered.map((note) => (
   803→   803→                <NoteCard
   804→   804→                  key={note.id}
   805→   805→                  note={note}
   806→   806→                  onOpen={openEditNote}
   807→   807→                  onTogglePin={togglePin}
   808→   808→                  onDelete={setNoteToDelete}
   809→   809→                  pinPending={patchNote.isPending}
   810→   810→                />
   811→   811→              ))}
   812→   812→            </div>
   813→   813→          )}
   814→   814→        </div>
   815→   815→      )}
   816→   816→
   817→   817→      <NoteDialog
   818→   818→        open={dialogOpen}
   819→   819→        onOpenChange={setDialogOpen}
   820→   820→        note={editingNote}
   821→   821→        existingTags={tags}
   822→   822→        submitting={saveNote.isPending}
   823→   823→        onSubmit={submitNote}
   824→   824→      />
   825→   825→
   826→   826→      <AlertDialog
   827→   827→        open={noteToDelete !== null}
   828→   828→        onOpenChange={(open) => {
   829→   829→          if (!open) setNoteToDelete(null);
   830→   830→        }}
   831→   831→      >
   832→   832→        <AlertDialogContent className="rounded-2xl">
   833→   833→          <AlertDialogHeader>
   834→   834→            <AlertDialogTitle>Delete note?</AlertDialogTitle>
   835→   835→            <AlertDialogDescription>
   836→   836→              &ldquo;{noteToDelete?.title || "Untitled note"}&rdquo; will be permanently
   837→   837→              removed. This cannot be undone.
   838→   838→            </AlertDialogDescription>
   839→   839→          </AlertDialogHeader>
   840→   840→          <AlertDialogFooter>
   841→   841→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   842→   842→            <AlertDialogAction
   843→   843→              className="bg-destructive text-white hover:bg-destructive/90"
   844→   844→              onClick={() => noteToDelete && removeNote.mutate(noteToDelete.id)}
   845→   845→            >
   846→   846→              Delete
   847→   847→            </AlertDialogAction>
   848→   848→          </AlertDialogFooter>
   849→   849→        </AlertDialogContent>
   850→   850→      </AlertDialog>
   851→   851→    </div>
   852→   852→  );
   853→   853→}
