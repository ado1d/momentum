    98→/** "Edited 2h ago" style relative timestamps */
    99→function relativeTime(iso: string): string {
   100→  const ms = Date.now() - new Date(iso).getTime();
   101→  if (Number.isNaN(ms)) return "some time ago";
   102→  const mins = Math.floor(ms / 60_000);
   103→  if (mins < 1) return "just now";
   104→  if (mins < 60) return `${mins}m ago`;
   105→  const hours = Math.floor(mins / 60);
   106→  if (hours < 24) return `${hours}h ago`;
   107→  const days = Math.floor(hours / 24);
   108→  if (days < 7) return `${days}d ago`;
   109→  const weeks = Math.floor(days / 7);
   110→  if (weeks < 5) return `${weeks}w ago`;
   111→  return new Date(iso).toLocaleDateString(undefined, {
   112→    month: "short",
   113→    day: "numeric",
   114→  });
   115→}

   487→export function NotesView() {
   488→  const queryClient = useQueryClient();
   489→
   490→  const { data: notes = [], isLoading, isError, refetch } = useQuery({
   491→    queryKey: ["notes"],
   492→    queryFn: notesApi.list,
   493→  });
   494→
   495→  const [search, setSearch] = React.useState("");
   496→  const [activeTag, setActiveTag] = React.useState<string>("all");
   497→
   498→  const [dialogOpen, setDialogOpen] = React.useState(false);
   499→  const [editingNote, setEditingNote] = React.useState<Note | null>(null);
   500→  const [noteToDelete, setNoteToDelete] = React.useState<Note | null>(null);
   501→  const [exporting, setExporting] = React.useState<string | null>(null);
   502→
   503→  const invalidate = (keys: string[]) => {
   504→    for (const key of keys) void queryClient.invalidateQueries({ queryKey: [key] });
   505→  };
   506→
   507→  // ── Derived data ──
   508→
   509→  const tags = React.useMemo(
   510→    () =>
   511→      Array.from(
   512→        new Set(notes.map((n) => n.tag).filter((t): t is string => !!t))
   513→      ).sort((a, b) => a.localeCompare(b)),
   514→    [notes]
   515→  );
   516→
   517→  const query = search.trim().toLowerCase();
   518→  const filtered = notes.filter((n) => {
   519→    if (activeTag !== "all" && n.tag !== activeTag) return false;
   520→    if (!query) return true;
   521→    return (
   522→      n.title.toLowerCase().includes(query) ||
   523→      n.content.toLowerCase().includes(query) ||
   524→      (n.tag ?? "").toLowerCase().includes(query)
   525→    );
   526→  });
   527→
   528→  // ── Mutations ──
   529→
   530→  const saveNote = useMutation({
   531→    mutationFn: (vars: { id?: string; input: NoteInput }) =>
   532→      vars.id ? notesApi.update(vars.id, vars.input) : notesApi.create(vars.input),
   533→    onSuccess: (_result, vars) => {
   534→      invalidate(["notes", "stats"]);
   535→      toast.success(vars.id ? "Note updated" : "Note created");
   536→      setDialogOpen(false);
   537→    },
   538→    onError: (e) => toast.error(e.message || "Could not save note"),
   539→  });
   540→
   541→  const patchNote = useMutation({
   542→    mutationFn: (vars: { note: Note; patch: NoteInput }) =>
   543→      notesApi.update(vars.note.id, vars.patch),
   544→    onMutate: async ({ note, patch }) => {
   545→      await queryClient.cancelQueries({ queryKey: ["notes"] });
   546→      const previous = queryClient.getQueryData<Note[]>(["notes"]);
   547→      queryClient.setQueryData<Note[]>(["notes"], (old) =>
   548→        (old ?? []).map((n) => (n.id === note.id ? { ...n, ...patch } : n))
   549→      );
   550→      return { previous };
   551→    },
   552→    onError: (error, _vars, context) => {
   553→      if (context?.previous) queryClient.setQueryData(["notes"], context.previous);
   554→      toast.error(error.message || "Could not update note");
   555→    },
   556→    onSettled: () => invalidate(["notes", "stats"]),
   557→  });
   558→
   559→  const removeNote = useMutation({
   560→    mutationFn: (id: string) => notesApi.remove(id),
   561→    onSuccess: () => {
   562→      invalidate(["notes", "stats"]);
   563→      toast.success("Note deleted");
   564→      setNoteToDelete(null);
   565→    },
   566→    onError: (e) => toast.error(e.message || "Could not delete note"),
   567→  });
   568→
   569→  // ── Handlers ──
   570→
   571→  const openNewNote = () => {
   572→    setEditingNote(null);
   573→    setDialogOpen(true);
   574→  };
   575→
   576→  const openEditNote = (note: Note) => {
   577→    setEditingNote(note);
   578→    setDialogOpen(true);
   579→  };
   580→
   581→  const submitNote = (values: NoteFormValues) => {
   582→    saveNote.mutate({
   583→      id: editingNote?.id,
   584→      input: {
   585→        title: values.title.trim() || "Untitled note",
   586→        content: values.content,
   587→        tag: values.tag.trim().toLowerCase() || null,
   588→        color: values.color,
   589→        pinned: values.pinned,
   590→      },
   591→    });
   592→  };
   593→
   594→  const togglePin = (note: Note) => {
   595→    patchNote.mutate({
   596→      note,
   597→      patch: { pinned: !note.pinned },
   598→    });
   599→  };
   600→
   601→  const runExport = async (
   602→    id: string,
   603→    label: string,
   604→    fn: () => Promise<void> | void
   605→  ) => {
   606→    setExporting(id);
   607→    const toastId = toast.loading(`Preparing ${label}…`);
   608→    try {
   609→      await fn();
   610→      toast.success(`${label} ready`, { id: toastId });
   611→    } catch (e) {
   612→      toast.error(
   613→        e instanceof Error ? e.message : `${label} failed`,
   614→        { id: toastId }
   615→      );
   616→    } finally {
   617→      setExporting(null);
   618→    }
   619→  };
   620→
   621→  const exportMarkdown = () =>
   622→    runExport("md", "Markdown export", async () => {
   623→      const md = await exportApi.markdown("notes");
   624→      downloadMarkdown(md, "momentum-notes.md");
   625→    });
   626→
   627→  const exportPdf = () =>
   628→    runExport("pdf", "PDF export", async () => {
   629→      if (notes.length === 0) {
   630→        toast.error("Nothing to export yet");
   631→        return;
   632→      }
   633→      const html = notes
   634→        .map(
   635→          (n) => `
   636→        <div>
   637→          <div style="font-size:17px; font-weight:700; margin:18px 0 4px;">${esc(n.title || "Untitled note")}${n.pinned ? " 📌" : ""}</div>
   638→          ${n.tag ? `<div style="font-size:11px; color:#777; margin-bottom:6px;"># ${esc(n.tag)}</div>` : ""}
   639→          ${miniMarkdownToHtml(n.content)}
   640→          <hr style="border:none; border-top:1px solid #e3e3e3; margin:16px 0 4px;" />
   641→        </div>`
   642→        )
   643→        .join("");
   644→      printHtml("Notes", html);
   645→    });
   646→
   647→  const exportBusy = exporting !== null;
   648→
   649→  // ── Render ──
   650→
   651→  return (
   652→    <div>
   653→      <ViewHeader
   654→        title="Notes"
   655→        subtitle="Quick capture & ideas"
   656→        actions={
   657→          <>
   658→            <DropdownMenu>
   659→              <DropdownMenuTrigger asChild>
   660→                <Button
   661→                  variant="outline"
   662→                  size="icon"
   663→                  className="size-11 rounded-xl"
   664→                  aria-label="Export notes"
   665→                  disabled={exportBusy}
   666→                >
   667→                  {exportBusy ? (
   668→                    <Loader2 className="animate-spin" aria-hidden="true" />
   669→                  ) : (
   670→                    <Download aria-hidden="true" />
   671→                  )}
   672→                </Button>
   673→              </DropdownMenuTrigger>
   674→              <DropdownMenuContent align="end" className="w-56">
   675→                <DropdownMenuItem onSelect={exportMarkdown}>
   676→                  <FileText aria-hidden="true" /> Export notes (Markdown)
   677→                </DropdownMenuItem>
   678→                <DropdownMenuItem onSelect={exportPdf}>
   679→                  <Printer aria-hidden="true" /> Export notes (PDF)
   680→                </DropdownMenuItem>
   681→              </DropdownMenuContent>
   682→            </DropdownMenu>
   683→            <Button className="h-11 rounded-xl" onClick={openNewNote}>
   684→              <Plus aria-hidden="true" /> New note
   685→            </Button>
   686→          </>
   687→        }
   688→      />
   689→
   690→      {isLoading ? (
   691→        <>
   692→          <div className="mb-4 space-y-3">
   693→            <Skeleton className="h-11 w-full rounded-xl" />
   694→            <Skeleton className="h-9 w-2/3 rounded-full" />
   695→          </div>
   696→          <NotesSkeleton />
   697→        </>
   698→      ) : isError ? (
   699→        <QueryError onRetry={() => void refetch()} />
   700→      ) : (
   701→        <div className="space-y-4">
   702→          {notes.length > 0 && (
   703→            <div className="space-y-3">
   704→              <div className="relative">
   705→                <Search
   706→                  className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   707→                  aria-hidden="true"
   708→                />
   709→                <Input
   710→                  value={search}
   711→                  onChange={(e) => setSearch(e.target.value)}
   712→                  placeholder="Search notes by title, content or tag…"
   713→                  aria-label="Search notes"
   714→                  className="h-11 rounded-xl pl-9 pr-10"
   715→                />
   716→                {search && (
   717→                  <button
   718→                    type="button"
   719→                    onClick={() => setSearch("")}
   720→                    className="absolute right-1.5 top-1/2 flex size-8 -translate-y-1/2 items-center justify-center rounded-lg text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"
   721→                    aria-label="Clear search"
   722→                  >
   723→                    <X className="size-4" aria-hidden="true" />
   724→                  </button>
   725→                )}
   726→              </div>
   727→
   728→              {tags.length > 0 && (
   729→                <div
   730→                  className="flex gap-2 overflow-x-auto pb-1"
   731→                  role="group"
   732→                  aria-label="Filter by tag"
   733→                >
   734→                  <button
   735→                    type="button"
   736→                    onClick={() => setActiveTag("all")}
   737→                    aria-pressed={activeTag === "all"}
   738→                    className={cn(
   739→                      "shrink-0 rounded-full border px-3.5 py-1.5 text-xs font-medium transition-colors",
   740→                      activeTag === "all"
   741→                        ? "border-primary bg-primary text-primary-foreground"
   742→                        : "border-border bg-card text-muted-foreground hover:bg-muted hover:text-foreground"
   743→                    )}
   744→                  >
   745→                    All ({notes.length})
   746→                  </button>
   747→                  {tags.map((tag) => {
   748→                    const count = notes.filter((n) => n.tag === tag).length;
   749→                    return (
   750→                      <button
   751→                        key={tag}
   752→                        type="button"
   753→                        onClick={() => setActiveTag(activeTag === tag ? "all" : tag)}
   754→                        aria-pressed={activeTag === tag}
   755→                        className={cn(
   756→                          "shrink-0 rounded-full border px-3.5 py-1.5 text-xs font-medium transition-colors",
   757→                          activeTag === tag
   758→                            ? "border-primary bg-primary text-primary-foreground"
   759→                            : "border-border bg-card text-muted-foreground hover:bg-muted hover:text-foreground"
   760→                        )}
   761→                      >
   762→                        # {tag} ({count})
   763→                      </button>
   764→                    );
   765→                  })}
   766→                </div>
   767→              )}
   768→            </div>
   769→          )}
   770→
   771→          {notes.length === 0 ? (
   772→            <EmptyState
   773→              icon={StickyNote}
   774→              title="No notes yet"
   775→              description="Capture ideas, snippets and lists before they escape. They'll be right here, pinned or searchable."
   776→              actionLabel="New note"
   777→              onAction={openNewNote}
   778→            />
   779→          ) : filtered.length === 0 ? (
   780→            <EmptyState
   781→              icon={Search}
   782→              title="No matching notes"
   783→              description={
   784→                query
   785→                  ? `Nothing matches “${search.trim()}”. Try a different word or clear the filters.`
   786→                  : "No notes carry this tag yet."
   787→              }
   788→              actionLabel="Clear filters"
   789→              onAction={() => {
   790→                setSearch("");
   791→                setActiveTag("all");
   792→              }}
   793→            />
   794→          ) : (
   795→            <div className="stagger-list columns-1 gap-4 sm:columns-2">
   796→              {filtered.map((note) => (
   797→                <NoteCard
   798→                  key={note.id}
   799→                  note={note}
   800→                  onOpen={openEditNote}
   801→                  onTogglePin={togglePin}
   802→                  onDelete={setNoteToDelete}
   803→                  pinPending={patchNote.isPending}
   804→                />
   805→              ))}
   806→            </div>
   807→          )}
   808→        </div>
   809→      )}
   810→
   811→      <NoteDialog
   812→        open={dialogOpen}
   813→        onOpenChange={setDialogOpen}
   814→        note={editingNote}
   815→        existingTags={tags}
   816→        submitting={saveNote.isPending}
   817→        onSubmit={submitNote}
   818→      />
   819→
   820→      <AlertDialog
   821→        open={noteToDelete !== null}
   822→        onOpenChange={(open) => {
   823→          if (!open) setNoteToDelete(null);
   824→        }}
   825→      >
   826→        <AlertDialogContent className="rounded-2xl">
   827→          <AlertDialogHeader>
   828→            <AlertDialogTitle>Delete note?</AlertDialogTitle>
   829→            <AlertDialogDescription>
   830→              &ldquo;{noteToDelete?.title || "Untitled note"}&rdquo; will be permanently
   831→              removed. This cannot be undone.
   832→            </AlertDialogDescription>
   833→          </AlertDialogHeader>
   834→          <AlertDialogFooter>
   835→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   836→            <AlertDialogAction
   837→              className="bg-destructive text-white hover:bg-destructive/90"
   838→              onClick={() => noteToDelete && removeNote.mutate(noteToDelete.id)}
   839→            >
   840→              Delete
   841→            </AlertDialogAction>
   842→          </AlertDialogFooter>
   843→        </AlertDialogContent>
   844→      </AlertDialog>
   845→    </div>
   846→  );
   847→}
