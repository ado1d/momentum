   151→/** Local calendar-day key of a due/completed ISO timestamp (null when unset). */
   152→function isoDayKey(iso: string | null): string | null {
   153→  if (!iso) return null;
   154→  const d = new Date(iso);
   155→  return Number.isNaN(d.getTime()) ? null : dateToKey(d);
   156→}
   157→
   158→function dueDayKey(todo: Todo): string | null {
   159→  return isoDayKey(todo.dueDate);
   160→}
   161→
   162→/** Turn a local day key (+ optional "HH:MM") into a real ISO instant. */
   163→function localDateKeyToIso(key: string, time = "00:00"): string {
   164→  const [y, mo, d] = key.split("-").map(Number);
   165→  const [h, mi] = time.split(":").map(Number);
   166→  return new Date(y, (mo ?? 1) - 1, d ?? 1, h ?? 0, mi ?? 0).toISOString();
   167→}

   169→function toTimeInputValue(d: Date): string {
   170→  const pad = (n: number) => String(n).padStart(2, "0");
   171→  return `${pad(d.getHours())}:${pad(d.getMinutes())}`;
   172→}
   173→
   174→/** ISO instant -> "YYYY-MM-DDTHH:MM" (local) for datetime-local inputs. */
   175→function isoToLocalInputValue(iso: string): string {
   176→  const d = new Date(iso);
   177→  if (Number.isNaN(d.getTime())) return "";
   178→  const pad = (n: number) => String(n).padStart(2, "0");
   179→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
   180→}

   182→function matchesTab(todo: Todo, tab: TabValue, today: string): boolean {
   183→  const key = dueDayKey(todo);
   184→  switch (tab) {
   185→    case "all":
   186→      return true;
   187→    case "today":
   188→      return todo.completed
   189→        ? isoDayKey(todo.completedAt) === today
   190→        : key === today;
   191→    case "upcoming":
   192→      return !todo.completed && !!key && key > today;
   193→    case "overdue":
   194→      return !todo.completed && !!key && key < today;
   195→    case "completed":
   196→      return todo.completed;
   197→  }
   198→}

   400→function EditTaskDialog({
   401→  todo,
   402→  saving,
   403→  onCancel,
   404→  onSave,
   405→}: {
   406→  todo: Todo;
   407→  saving: boolean;
   408→  onCancel: () => void;
   409→  onSave: (patch: Partial<TodoInput>) => void;
   410→}) {
   411→  const [title, setTitle] = React.useState(todo.title);
   412→  const [notes, setNotes] = React.useState(todo.notes ?? "");
   413→  const [priority, setPriority] = React.useState<Priority>(todo.priority);
   414→  const [category, setCategory] = React.useState(todo.category);
   415→  const [repeat, setRepeat] = React.useState<RepeatKind>(todo.repeat);
   416→  const [dueDate, setDueDate] = React.useState(() => {
   417→    if (!todo.dueDate) return "";
   418→    const d = new Date(todo.dueDate);
   419→    return Number.isNaN(d.getTime()) ? "" : dateToKey(d);
   420→  });
   421→  const [dueTime, setDueTime] = React.useState(() => {
   422→    if (!todo.dueDate || !hasTime(todo.dueDate)) return "";
   423→    const d = new Date(todo.dueDate);
   424→    return Number.isNaN(d.getTime()) ? "" : toTimeInputValue(d);
   425→  });
   426→  const [reminder, setReminder] = React.useState(() =>
   427→    todo.reminderAt ? isoToLocalInputValue(todo.reminderAt) : ""
   428→  );
   429→
   430→  const save = () => {
   431→    const trimmed = title.trim();
   432→    if (!trimmed) {
   433→      toast.error("Give the task a title");
   434→      return;
   435→    }
   436→    const dueIso = dueDate
   437→      ? localDateKeyToIso(dueDate, dueTime || "00:00")
   438→      : null;
   439→    const reminderIso = reminder ? new Date(reminder).toISOString() : null;
   440→    onSave({
   441→      title: trimmed,
   442→      notes: notes.trim() || null,
   443→      priority,
   444→      category,
   445→      dueDate: dueIso,
   446→      reminderAt: reminderIso,
   447→      repeat,
   448→    });
   449→  };

   451→  const labelClass = "text-xs font-medium text-muted-foreground";
   452→
   453→  return (
   454→    <Dialog
   455→      open
   456→      onOpenChange={(open) => {
   457→        if (!open) onCancel();
   458→      }}
   459→    >
   460→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   461→        <DialogHeader>
   462→          <DialogTitle>Edit task</DialogTitle>
   463→          <DialogDescription>
   464→            Update the details of your task.
   465→          </DialogDescription>
   466→        </DialogHeader>
   467→
   468→        <div className="space-y-4">
   469→          <div className="space-y-1.5">
   470→            <label htmlFor="edit-title" className={labelClass}>
   471→              Title
   472→            </label>
   473→            <Input
   474→              id="edit-title"
   475→              value={title}
   476→              autoFocus
   477→              onChange={(e) => setTitle(e.target.value)}
   478→            />
   479→          </div>
   480→
   481→          <div className="space-y-1.5">
   482→            <label htmlFor="edit-notes" className={labelClass}>
   483→              Notes
   484→            </label>
   485→            <Textarea
   486→              id="edit-notes"
   487→              rows={3}
   488→              placeholder="Add notes (optional)"
   489→              value={notes}
   490→              onChange={(e) => setNotes(e.target.value)}
   491→            />
   492→          </div>
   493→
   494→          <div className="grid grid-cols-2 gap-3">
   495→            <div className="space-y-1.5">
   496→              <label htmlFor="edit-priority" className={labelClass}>
   497→                Priority

   508→                    <SelectItem key={p.value} value={p.value}>
