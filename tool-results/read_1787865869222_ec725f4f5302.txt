    42→/** "Aug 27" style short label for a YYYY-MM-DD key. */
    43→function shortDate(key: string): string {
    44→  const d = keyToDate(key);
    45→  return isValid(d) ? format(d, "MMM d") : key;
    46→}
    47→
    48→/** 65 → "1h 5m", 45 → "45m", 120 → "2h" */
    49→function formatMinutes(total: number): string {
    50→  if (total <= 0) return "0m";
    51→  const h = Math.floor(total / 60);
    52→  const m = total % 60;
    53→  if (h === 0) return `${m}m`;
    54→  if (m === 0) return `${h}h`;
    55→  return `${h}h ${m}m`;
    56→}
    57→
    58→function formatCompact(n: number): string {
    59→  return Number.isInteger(n) ? String(n) : n.toFixed(1);
    60→}

    81→/** Catmull-Rom → cubic bezier smoothing, clamped to the 0..40 plot band. */
    82→function smoothPath(pts: { x: number; y: number }[]): string {
    83→  if (pts.length === 0) return "";
    84→  if (pts.length === 1) return `M ${pts[0].x.toFixed(2)} ${pts[0].y.toFixed(2)}`;
    85→  const clamp = (v: number) => Math.min(40, Math.max(0, v));
    86→  let d = `M ${pts[0].x.toFixed(2)} ${pts[0].y.toFixed(2)}`;
    87→  for (let i = 0; i < pts.length - 1; i += 1) {
    88→    const p0 = pts[i - 1] ?? pts[i];
    89→    const p1 = pts[i];
    90→    const p2 = pts[i + 1];
    91→    const p3 = pts[i + 2] ?? p2;
    92→    const c1x = p1.x + (p2.x - p0.x) / 6;
    93→    const c1y = clamp(p1.y + (p2.y - p0.y) / 6);
    94→    const c2x = p2.x - (p3.x - p1.x) / 6;
    95→    const c2y = clamp(p2.y - (p3.y - p1.y) / 6);
    96→    d += ` C ${c1x.toFixed(2)} ${c1y.toFixed(2)}, ${c2x.toFixed(2)} ${c2y.toFixed(2)}, ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`;
    97→  }
    98→  return d;
    99→}

   260→function HeatmapCard({ heatmap }: { heatmap: DayStat[] }) {
   261→  const today = todayKey();
   262→
   263→  // Pad the first column so day 1 of the data lands on its Monday-based row.
   264→  const weeks = React.useMemo(() => {
   265→    if (heatmap.length === 0) return [] as (DayStat | null)[][];
   266→    const pad = weekdayOfKey(heatmap[0].date) - 1; // Mon=1 → 0 padding
   267→    const cells: (DayStat | null)[] = [
   268→      ...Array.from({ length: pad }, () => null),
   269→      ...heatmap,
   270→    ];
   271→    const out: (DayStat | null)[][] = [];
   272→    for (let i = 0; i < cells.length; i += 7) out.push(cells.slice(i, i + 7));
   273→    return out;
   274→  }, [heatmap]);
   275→
   276→  // Tiny month label above a column whenever the month changes.
   277→  const monthLabels = React.useMemo(() => {
   278→    let last = -1;
   279→    return weeks.map((week) => {
   280→      const first = week.find((c): c is DayStat => c !== null);
   281→      if (!first) return "";
   282→      const d = keyToDate(first.date);
   283→      if (!isValid(d)) return "";
   284→      const m = d.getFullYear() * 12 + d.getMonth();
   285→      if (m !== last) {
   286→        last = m;
   287→        return format(d, "MMM");
   288→      }
   289→      return "";
   290→    });
   291→  }, [weeks]);
   292→
   293→  const { longestStreak, activeDays, avgScore } = React.useMemo(() => {
   294→    let streak = 0;
   295→    let best = 0;
   296→    let active = 0;
   297→    let sum = 0;
   298→    for (const d of heatmap) {
   299→      if (d.score > 0) {
   300→        streak += 1;
   301→        active += 1;
   302→        if (streak > best) best = streak;
   303→      } else {
   304→        streak = 0;
   305→      }
   306→      sum += d.score;
   307→    }
   308→    return {
   309→      longestStreak: best,
   310→      activeDays: active,
   311→      avgScore: heatmap.length ? Math.round(sum / heatmap.length) : 0,
   312→    };
   313→  }, [heatmap]);
   314→
   315→  const allEmpty = heatmap.length > 0 && heatmap.every((d) => d.score <= 0);
   316→  const totalDays = heatmap.length;
   317→
   318→  return (
   319→    <Card className="gap-4 rounded-2xl p-4 sm:p-5">
   320→      <CardHead
   321→        title="Activity"
   322→        subtitle="Last 12 weeks"
   323→        action={
   324→          <div
   325→            className="flex items-center gap-1 text-[10px] font-medium text-muted-foreground"
   326→            aria-hidden="true"
   327→          >
   328→            Less
   329→            {[
   330→              "bg-muted",
   331→              "bg-emerald-500/25",
   332→              "bg-emerald-500/45",
   333→              "bg-emerald-500/70",
   334→              "bg-emerald-500",
   335→            ].map((c) => (
   336→              <span key={c} className={cn("size-2.5 rounded-[3px]", c)} />
   337→            ))}
   338→            More
   339→          </div>
   340→        }
   341→      />
   342→
   343→      <div className="flex flex-col gap-5 sm:flex-row sm:items-center sm:justify-between">
   344→        <div
   345→          className="no-scrollbar overflow-x-auto"
   346→          role="img"
   347→          aria-label={`Activity heatmap for the last 12 weeks — ${activeDays} of ${totalDays} days active, average score ${avgScore}.`}
   348→        >
   349→          <div className="flex w-max gap-1">
   350→            {/* Weekday hints (M / W / F) */}
   351→            <div
   352→              className="mr-1 flex flex-col gap-1 pt-4 text-[9px] font-medium leading-none text-muted-foreground/60"
   353→              aria-hidden="true"
   354→            >
   355→              {WEEKDAY_LABELS.map((label, i) => (
   356→                <span key={i} className="flex h-[11px] items-center sm:h-3">
   357→                  {label}
   358→                </span>
   359→              ))}
   360→            </div>
   361→
   362→            {weeks.map((week, wi) => (
   363→              <div key={wi} className="flex flex-col gap-1">
   364→                <span className="flex h-3 items-center text-[9px] font-medium leading-none text-muted-foreground/70">
   365→                  {monthLabels[wi]}
   366→                </span>
   367→                {week.map((cell, ci) =>
   368→                  cell ? (
   369→                    <span
   370→                      key={cell.date}
   371→                      title={`${shortDate(cell.date)} · score ${cell.score} · ${cell.todosCompleted} ${cell.todosCompleted === 1 ? "task" : "tasks"}, ${cell.habitsCompleted} ${cell.habitsCompleted === 1 ? "habit" : "habits"}, ${cell.routineCompleted} routine`}
   372→                      className={cn(
   373→                        "size-[11px] rounded-[3px] sm:size-3",
   374→                        heatLevelClass(cell.score),
   375→                        cell.date === today &&
   376→                          "ring-2 ring-primary ring-offset-1 ring-offset-card"
   377→                      )}
   378→                    />
   379→                  ) : (
   380→                    <span
   381→                      key={`pad-${wi}-${ci}`}
   382→                      className="size-[11px] sm:size-3"
   383→                      aria-hidden="true"
   384→                    />
   385→                  )
   386→                )}
   387→              </div>
   388→            ))}
   389→          </div>
   390→        </div>
   391→
   392→        <div className="flex flex-wrap items-center gap-2 sm:max-w-[13rem] sm:flex-col sm:items-stretch sm:gap-2.5">
   393→          <StatChip
   394→            icon={Flame}
   395→            iconClassName="text-orange-500"
   396→            label="Longest streak"
   397→            value={`${longestStreak} ${longestStreak === 1 ? "day" : "days"}`}
   398→          />
   399→          <StatChip
   400→            icon={CalendarCheck}
   401→            iconClassName="text-teal-600 dark:text-teal-400"
   402→            label="Active days"
   403→            value={`${activeDays} / ${totalDays}`}
   404→          />
   405→          <StatChip
   406→            icon={Activity}
   407→            iconClassName="text-emerald-600 dark:text-emerald-400"
   408→            label="Avg score"
   409→            value={String(avgScore)}
   410→          />
   411→        </div>
   412→      </div>
   413→
   414→      {allEmpty && (
   415→        <p className="text-xs text-muted-foreground">
   416→          Start checking things off to fill your heatmap.
   417→        </p>
   418→      )}
   419→    </Card>
   420→  );
   421→}

   423→// ── Tasks trend (pure inline SVG line/area chart) ────────────
   424→
   425→function TasksTrendCard({ trend }: { trend: TrendPoint[] }) {
   426→  const n = trend.length;
   427→
   428→  const { max, total, best, avg } = React.useMemo(() => {
   429→    const counts = trend.map((p) => p.count);
   430→    const t = counts.reduce((a, b) => a + b, 0);
   431→    // Round odd maxima up so the mid gridline/label stays an integer.
   432→    const m = Math.max(1, ...counts);
   433→    return {
   434→      max: m >= 3 && m % 2 === 1 ? m + 1 : m,
   435→      total: t,
   436→      best: counts.length ? Math.max(...counts) : 0,
   437→      avg: counts.length ? t / counts.length : 0,
   438→    };
   439→  }, [trend]);
   440→
   441→  // Points live in a 100×40 viewBox; the SVG stretches with
   442→  // preserveAspectRatio="none" and strokes use vector-effect so they
   443→  // stay crisp. Text (axis labels) is rendered as HTML so it never distorts.
   444→  const points = React.useMemo(
   445→    () =>
   446→      trend.map((p, i) => ({
   447→        ...p,
   448→        x: ((i + 0.5) * 100) / n,
   449→        y: 40 - (p.count / max) * 40,
   450→        yPct: (1 - p.count / max) * 100,
   451→      })),
   452→    [trend, n, max]
   453→  );
   454→
   455→  const linePath = React.useMemo(() => smoothPath(points), [points]);
   456→  const areaPath = React.useMemo(() => {
   457→    if (points.length < 2) return "";
   458→    const last = points[points.length - 1];
   459→    const first = points[0];
   460→    return `${linePath} L ${last.x.toFixed(2)} 40 L ${first.x.toFixed(2)} 40 Z`;
   461→  }, [linePath, points]);
   462→
   463→  const xLabelIdx = React.useMemo(() => {
   464→    if (n === 0) return [] as number[];
   465→    const raw = [0, 0.25, 0.5, 0.75, 1].map((f) => Math.round((n - 1) * f));
   466→    return [...new Set(raw)];
   467→  }, [n]);
   468→
   469→  const mid = max / 2;
   470→  const showMid = max >= 2; // max=1 has no integer midpoint
   471→
   472→  return (
   473→    <Card className="gap-4 rounded-2xl p-4 sm:p-5">
   474→      <CardHead title="Tasks completed" subtitle="Last 30 days" />
   475→
   476→      <div
   477→        className="flex gap-2"
   478→        role="img"
   479→        aria-label={`Tasks completed per day over the last ${n} days — total ${total}, best day ${best}, average ${avg.toFixed(1)} per day.`}
   480→      >
   481→        {/* Y-axis labels (HTML so they stay crisp) */}
   482→        <div className="relative w-7 shrink-0 text-right text-[9px] font-medium leading-none tabular-nums text-muted-foreground">
   483→          <span className="absolute inset-x-0 top-0 -translate-y-1/2">
   484→            {formatCompact(max)}
   485→          </span>
   486→          {showMid && (
   487→            <span className="absolute inset-x-0 top-1/2 -translate-y-1/2">
   488→              {formatCompact(mid)}
   489→            </span>
   490→          )}
   491→          <span className="absolute inset-x-0 top-full -translate-y-1/2">0</span>
   492→        </div>
   493→
   494→        {/* Plot area */}
   495→        <div className="relative h-36 min-w-0 flex-1">
   496→          <svg
   497→            className="absolute inset-0 h-full w-full text-primary"
   498→            viewBox="0 0 100 40"
   499→            preserveAspectRatio="none"
   500→            aria-hidden="true"
   501→          >
   502→            <defs>
   503→              <linearGradient id="insights-trend-fill" x1="0" y1="0" x2="0" y2="1">
   504→                <stop offset="0%" stopColor="currentColor" stopOpacity="0.16" />
   505→                <stop offset="100%" stopColor="currentColor" stopOpacity="0.02" />
   506→              </linearGradient>
   507→            </defs>
   508→            {/* Gridlines */}
   509→            <line
   510→              x1="0"
   511→              y1="0"
   512→              x2="100"
   513→              y2="0"
   514→              strokeWidth="1"
   515→              vectorEffect="non-scaling-stroke"
   516→              className="stroke-muted-foreground/15"
   517→            />
   518→            {showMid && (
   519→              <line
   520→                x1="0"
   521→                y1="20"
   522→                x2="100"
   523→                y2="20"
   524→                strokeWidth="1"
   525→                vectorEffect="non-scaling-stroke"
   526→                className="stroke-muted-foreground/10"
   527→              />
   528→            )}
   529→            <line
   530→              x1="0"
   531→              y1="40"
   532→              x2="100"
   533→              y2="40"
   534→              strokeWidth="1"
   535→              vectorEffect="non-scaling-stroke"
   536→              className="stroke-muted-foreground/15"
   537→            />
   538→            {/* Area + line */}
   539→            {areaPath && <path d={areaPath} fill="url(#insights-trend-fill)" />}
   540→            {linePath && (
   541→              <path
   542→                d={linePath}
   543→                fill="none"
   544→                strokeWidth="2"
   545→                strokeLinecap="round"
   546→                strokeLinejoin="round"
   547→                vectorEffect="non-scaling-stroke"
   548→                className="stroke-primary"
   549→              />
   550→            )}
   551→          </svg>
   552→
   553→          {/* Hover zones: one per day — tooltip + guideline + dot */}
   554→          <div
   555→            className="absolute inset-0 grid"
   556→            style={{ gridTemplateColumns: `repeat(${n}, 1fr)` }}
   557→            aria-hidden="true"
   558→          >
   559→            {points.map((p) => (
   560→              <div

   689→// ── Mood distribution ────────────────────────────────────────
   690→
   691→function MoodCard({ distribution }: { distribution: InsightsData["moodDistribution"] }) {
   692→  const total = distribution.reduce((a, d) => a + d.count, 0);
   693→
   694→  // Render in a fixed best→rough order, only moods that were recorded.
   695→  const rows = MOODS.map((m) => {
   696→    const entry = distribution.find((d) => d.mood === m.value);
   697→    return entry ? { ...m, count: entry.count } : null;
   698→  }).filter((r): r is { value: Mood; label: string; emoji: string; count: number } => r !== null);
   699→
   700→  const pctOf = (count: number) => (total > 0 ? Math.round((count / total) * 100) : 0);
   701→
   702→  return (
   703→    <Card className="gap-4 rounded-2xl p-4 sm:p-5">
   704→      <CardHead title="Mood over time" subtitle="From your diary entries" />
   705→
   706→      <div
   707→        className="flex h-3 w-full overflow-hidden rounded-full bg-muted"
   708→        role="img"
   709→        aria-label={`Mood distribution across ${total} diary ${total === 1 ? "entry" : "entries"}: ${rows
   710→          .map((r) => `${r.label} ${pctOf(r.count)}%`)
   711→          .join(", ")}.`}
   712→      >
   713→        {rows.map((r) => (
   714→          <div
   715→            key={r.value}
   716→            className={moodBarStyles[r.value]}
   717→            style={{ width: `${pctOf(r.count)}%` }}
   718→            title={`${r.label} · ${r.count} ${r.count === 1 ? "entry" : "entries"} · ${pctOf(r.count)}%`}
   719→          />
   720→        ))}
   721→      </div>
   722→
   723→      <div className="grid grid-cols-1 gap-x-6 gap-y-2 sm:grid-cols-2">
   724→        {rows.map((r) => (
   725→          <div key={r.value} className="flex items-center gap-2 text-sm">
   726→            <span
   727→              className={cn("size-2.5 shrink-0 rounded-full", moodBarStyles[r.value])}
   728→              aria-hidden="true"
   729→            />
   730→            <span className="text-base leading-none" aria-hidden="true">
   731→              {r.emoji}
   732→            </span>
   733→            <span className="font-medium">{r.label}</span>
   734→            <span className="ml-auto text-xs tabular-nums text-muted-foreground">
   735→              {r.count} · {pctOf(r.count)}%
   736→            </span>
   737→          </div>
   738→        ))}
   739→      </div>
   740→    </Card>
   741→  );
   742→}
