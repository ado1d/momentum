   123→   123→/** Mirrors the server-side status rules for optimistic updates. */
   124→   124→function applyStatusRules(goal: Goal, patch: GoalPatch): Goal {
   125→   125→  const target = patch.target ?? goal.target;
   126→   126→  const progress =
   127→   127→    patch.progress !== undefined
   128→   128→      ? Math.min(Math.max(patch.progress, 0), target)
   129→   129→      : Math.min(goal.progress, target);
   130→   130→  let status: GoalStatus = goal.status;
   131→   131→  if (progress >= target) status = "completed";
   132→   132→  else if (patch.status !== undefined) status = patch.status as GoalStatus;
   133→   133→  else if (goal.status === "completed") status = "active";
   134→   134→  return { ...goal, ...patch, target, progress, status } as Goal;
   135→   135→}
   136→   136→
