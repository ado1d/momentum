   147→   147→const MAX_BACKUP_BYTES = 20 * 1024 * 1024; // sanity cap: 20 MB
   148→   148→
   149→   149→// A file counts as a plausible backup when at least one of these is an array.
   150→   150→const BACKUP_ARRAY_KEYS = [
   151→   151→  "todos",
   152→   152→  "subtasks",
   153→   153→  "habits",
   154→   154→  "routineTasks",
   155→   155→  "notes",
   156→   156→  "journal",
   157→   157→  "goals",
   158→   158→] as const;
   159→   159→
   160→   160→interface ParsedBackup {
   161→   161→  fileName: string;
   162→   162→  sizeLabel: string;
   163→   163→  exportedAt: string | null; // pre-formatted friendly label
   164→   164→  data: Record<string, unknown>;
   165→   165→  preview: { label: string; count: number }[];
   166→   166→}
   167→   167→
   168→   168→function isRecord(value: unknown): value is Record<string, unknown> {
   169→   169→  return typeof value === "object" && value !== null && !Array.isArray(value);
   170→   170→}
   171→   171→
   172→   172→function arrayCount(data: Record<string, unknown>, key: string): number {
   173→   173→  const value = data[key];
   174→   174→  return Array.isArray(value) ? value.length : 0;
   175→   175→}
   176→   176→
   177→   177→function formatBytes(bytes: number): string {
   178→   178→  if (bytes < 1024) return `${bytes} B`;
   179→   179→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   180→   180→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   181→   181→}
   182→   182→
   183→   183→function friendlyTimestamp(value: unknown): string | null {
   184→   184→  if (typeof value !== "string" || value.length === 0) return null;
   185→   185→  const date = new Date(value);
   186→   186→  if (Number.isNaN(date.getTime())) return null;
   187→   187→  return date.toLocaleString(undefined, { dateStyle: "medium", timeStyle: "short" });
   188→   188→}
   189→   189→
   190→   190→function pluralize(count: number, one: string, many: string): string {
   191→   191→  return `${count} ${count === 1 ? one : many}`;
   192→   192→}
   193→   193→
   194→   194→function formatImportCounts(counts: ImportCounts): string {
   195→   195→  const parts = [
   196→   196→    pluralize(counts.todos, "task", "tasks"),
   197→   197→    pluralize(counts.habits, "habit", "habits"),
   198→   198→  ];
   199→   199→  if (counts.routineTasks > 0) {
   200→   200→    parts.push(pluralize(counts.routineTasks, "routine task", "routine tasks"));
   201→   201→  }
   202→   202→  parts.push(
   203→   203→    pluralize(counts.notes, "note", "notes"),
   204→   204→    pluralize(counts.journal, "journal entry", "journal entries"),
   205→   205→    pluralize(counts.goals, "goal", "goals"),
   206→   206→  );
   207→   207→  parts.push(`${counts.skipped} skipped`);
   208→   208→  return parts.join(" · ");
   209→   209→}

   211→   211→function ImportSection() {
   212→   212→  const queryClient = useQueryClient();
   213→   213→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   214→   214→
   215→   215→  const [parsed, setParsed] = React.useState<ParsedBackup | null>(null);
   216→   216→  const [mode, setMode] = React.useState<"merge" | "replace">("merge");
   217→   217→  const [replaceConfirmed, setReplaceConfirmed] = React.useState(false);
   218→   218→  const [dragging, setDragging] = React.useState(false);
   219→   219→  const [result, setResult] = React.useState<ImportResult | null>(null);
   220→   220→
   221→   221→  const importBackup = useMutation({
   222→   222→    mutationFn: () => {
   223→   223→      if (!parsed) throw new Error("Choose a backup file first");
   224→   224→      return importApi.restore(parsed.data, mode);
   225→   225→    },
   226→   226→    onSuccess: (res) => {
   227→   227→      toast.success(res.message);
   228→   228→      setResult(res);
   229→   229→      // Reset the pick + confirmation; the success panel stays visible
   230→   230→      // until another file is chosen or it is dismissed.
   231→   231→      setParsed(null);
   232→   232→      setMode("merge");
   233→   233→      setReplaceConfirmed(false);
   234→   234→      // An import can touch every feature (todos, subtasks, habits, routine,
   235→   235→      // notes, journal, goals, settings — plus derived stats / insights /
   236→   236→      // focus / review / search), so invalidate ALL queries — the robust
   237→   237→      // and simplest option after a full restore.
   238→   238→      void queryClient.invalidateQueries();
   239→   239→    },
   240→   240→    onError: (error) => {
   241→   241→      toast.error(error.message || "Import failed — please try again");
   242→   242→    },
   243→   243→  });
   244→   244→
   245→   245→  const isImporting = importBackup.isPending;
   246→   246→
   247→   247→  const openPicker = () => {
   248→   248→    if (!isImporting) fileInputRef.current?.click();
   249→   249→  };
   250→   250→
   251→   251→  const handleFile = (file: File | undefined | null) => {
   252→   252→    if (!file || isImporting) return;
   253→   253→    setResult(null); // choosing a new file clears the previous outcome
   254→   254→    if (file.size > MAX_BACKUP_BYTES) {
   255→   255→      toast.error("File too large — backups up to 20 MB are supported");
   256→   256→      return;
   257→   257→    }
   258→   258→    const reader = new FileReader();
   259→   259→    reader.onerror = () => toast.error("Could not read the file — try again");
   260→   260→    reader.onload = () => {
   261→   261→      let json: unknown;
   262→   262→      try {
   263→   263→        json = JSON.parse(String(reader.result ?? ""));
   264→   264→      } catch {
   265→   265→        toast.error("Invalid JSON file");
   266→   266→        return;
   267→   267→      }
   268→   268→      if (
   269→   269→        !isRecord(json) ||
   270→   270→        !BACKUP_ARRAY_KEYS.some((key) => Array.isArray(json[key]))
   271→   271→      ) {
   272→   272→        toast.error("Not a Momentum backup file");
   273→   273→        return;
   274→   274→      }
   275→   275→      setParsed({
   276→   276→        fileName: file.name,
   277→   277→        sizeLabel: formatBytes(file.size),
   278→   278→        exportedAt: friendlyTimestamp(json.exportedAt),
   279→   279→        data: json,
   280→   280→        preview: [
   281→   281→          { label: "tasks", count: arrayCount(json, "todos") },
   282→   282→          { label: "subtasks", count: arrayCount(json, "subtasks") },
   283→   283→          { label: "habits", count: arrayCount(json, "habits") },
   284→   284→          { label: "routine", count: arrayCount(json, "routineTasks") },
   285→   285→          { label: "notes", count: arrayCount(json, "notes") },
   286→   286→          { label: "journal", count: arrayCount(json, "journal") },
   287→   287→          { label: "goals", count: arrayCount(json, "goals") },
   288→   288→        ].filter((p) => p.count > 0),
   289→   289→      });
   290→   290→      setMode("merge");
   291→   291→      setReplaceConfirmed(false);
   292→   292→    };
   293→   293→    reader.readAsText(file);
   294→   294→  };
   295→   295→
   296→   296→  const clearFile = () => {
   297→   297→    setParsed(null);
   298→   298→    setMode("merge");
   299→   299→    setReplaceConfirmed(false);
   300→   300→  };
   301→   301→
   302→   302→  const canImport =
   303→   303→    parsed !== null && !isImporting && (mode === "merge" || replaceConfirmed);
   304→   304→
   305→   305→  return (
   306→   306→    <Card className="rounded-2xl py-0 shadow-card">
   307→   307→      <CardHeader className="p-4 pb-2 sm:p-6 sm:pb-3">
   308→   308→        <CardTitle className="flex items-center gap-2 text-base">
   309→   309→          <Upload className="size-4.5 text-primary" aria-hidden="true" />
   310→   310→          Import &amp; restore
   311→   311→        </CardTitle>
   312→   312→        <CardDescription>
   313→   313→          Restore a Momentum JSON backup — merge it in, or replace everything.
   314→   314→        </CardDescription>
   315→   315→      </CardHeader>
   316→   316→      <CardContent className="space-y-4 p-4 pt-2 sm:p-6 sm:pt-2">
   317→   317→        {/* Drop / pick zone */}
   318→   318→        <div
   319→   319→          role="button"
   320→   320→          tabIndex={0}
   321→   321→          aria-label="Choose a backup file, or drop one here"
   322→   322→          aria-disabled={isImporting}
   323→   323→          onClick={openPicker}
   324→   324→          onKeyDown={(e) => {
   325→   325→            if (e.key === "Enter" || e.key === " ") {
   326→   326→              e.preventDefault();
   327→   327→              openPicker();
   328→   328→            }
   329→   329→          }}
   330→   330→          onDragOver={(e) => {
   331→   331→            e.preventDefault();
   332→   332→            setDragging(true);
   333→   333→          }}
   334→   334→          onDragLeave={() => setDragging(false)}
   335→   335→          onDrop={(e) => {
   336→   336→            e.preventDefault();
   337→   337→            setDragging(false);
   338→   338→            handleFile(e.dataTransfer.files?.[0]);
   339→   339→          }}
   340→   340→          className={cn(
   341→   341→            "flex cursor-pointer flex-col items-center gap-1.5 rounded-2xl border-2 border-dashed px-4 py-7 text-center transition-colors",
   342→   342→            "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50 hover:border-primary/50 hover:bg-primary/5",
   343→   343→            dragging && "border-primary bg-primary/5",
   344→   344→            isImporting && "pointer-events-none opacity-60"
   345→   345→          )}
   346→   346→        >
   347→   347→          <FileJson className="size-8 text-primary" aria-hidden="true" />
   348→   348→          <p className="mt-1 text-sm font-semibold">
   349→   349→            {parsed ? "Choose a different backup" : "Drag & drop a backup file"}
   350→   350→          </p>
   351→   351→          <p className="text-xs text-muted-foreground">or</p>
   352→   352→          <span className="inline-flex h-9 items-center rounded-xl bg-primary px-4 text-sm font-semibold text-primary-foreground shadow-sm">
   353→   353→            Choose backup file
   354→   354→          </span>
   355→   355→          <p className="mt-1.5 text-[11px] text-muted-foreground/80">
   356→   356→            JSON backups from “Backup (JSON)” · up to 20 MB
   357→   357→          </p>
   358→   358→          <input

   388→   388→              <Button
