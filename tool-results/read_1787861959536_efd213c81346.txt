    85→/** Longest run of consecutive day keys in a list of entries */
    86→function longestStreakOf(entries: JournalEntry[]): number {
    87→  const keys = Array.from(new Set(entries.map((e) => e.date))).sort();
    88→  let best = 0;
    89→  let run = 0;
    90→  let prev: string | null = null;
    91→  for (const k of keys) {
    92→    run = prev !== null && addDaysToKey(prev, 1) === k ? run + 1 : 1;
    93→    if (run > best) best = run;
    94→    prev = k;
    95→  }
    96→  return best;
    97→}

    99→function entryPreview(entry: JournalEntry): string {
   100→  if (entry.title?.trim()) return entry.title.trim();
   101→  const plain = entry.content.replace(/[#>*_`~\-]/g, "").replace(/\s+/g, " ").trim();
   102→  return plain.length > 60 ? `${plain.slice(0, 60)}…` : plain;
   103→}

   294→export function DiaryView() {
   295→  const queryClient = useQueryClient();
   296→  const today = todayKey();
   297→
   298→  const { data: entries = [], isLoading, isError, refetch } = useQuery({
   299→    queryKey: ["journal"],
   300→    queryFn: () => journalApi.list({ limit: 366 }),
   301→  });
   302→
   303→  // Shared settings query (powers the calendar's week start)
   304→  const { data: settings } = useQuery({
   305→    queryKey: ["settings"],
   306→    queryFn: settingsApi.get,
   307→  });
   308→
   309→  // ── Editor state ──
   310→  const [date, setDate] = React.useState(today);
   311→  const [title, setTitle] = React.useState("");
   312→  const [content, setContent] = React.useState("");
   313→  const [mood, setMood] = React.useState<Mood | null>(null);
   314→  const [energy, setEnergy] = React.useState(3);
   315→  const [gratitude, setGratitude] = React.useState("");
   316→  const [justSaved, setJustSaved] = React.useState(false);
   317→  const [calOpen, setCalOpen] = React.useState(false);
   318→
   319→  const [expandedId, setExpandedId] = React.useState<string | null>(null);
   320→  const [entryToDelete, setEntryToDelete] = React.useState<JournalEntry | null>(null);
   321→  const [exporting, setExporting] = React.useState<string | null>(null);
   322→
   323→  const contentRef = React.useRef<HTMLTextAreaElement>(null);
   324→
   325→  const entry = entries.find((e) => e.date === date) ?? null;
   326→
   327→  // Load the selected day's entry into the form whenever the day
   328→  // changes or a different server version arrives (e.g. after save).
   329→  const syncedSig = React.useRef("");
   330→  React.useEffect(() => {
   331→    const sig = `${date}:${entry?.updatedAt ?? "-"}`;
   332→    if (syncedSig.current !== sig) {
   333→      syncedSig.current = sig;
   334→      setTitle(entry?.title ?? "");
   335→      setContent(entry?.content ?? "");
   336→      setMood(entry?.mood ?? null);
   337→      setEnergy(entry?.energy ?? 3);
   338→      setGratitude(entry?.gratitude ?? "");
   339→    }
   340→  }, [date, entry]);
   341→
   342→  // Auto-hide the "Saved" chip
   343→  React.useEffect(() => {
   344→    if (!justSaved) return;
   345→    const id = setTimeout(() => setJustSaved(false), 3000);
   346→    return () => clearTimeout(id);
   347→  }, [justSaved]);
   348→
   349→  // ── Mutations ──
   350→
   351→  const saveEntry = useMutation({
   352→    mutationFn: () =>
   353→      journalApi.upsert({
   354→        date,
   355→        title: title.trim() || null,
   356→        content,
   357→        mood,
   358→        energy,
   359→        gratitude: gratitude.trim() || null,
   360→      }),
   361→    onSuccess: () => {
   362→      void queryClient.invalidateQueries({ queryKey: ["journal"] });
   363→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   364→      setJustSaved(true);
   365→      toast.success(entry ? "Diary updated" : "Diary saved");
   366→    },
   367→    onError: (e) => toast.error(e.message || "Could not save entry"),
   368→  });
   369→
   370→  const removeEntry = useMutation({
   371→    mutationFn: (id: string) => journalApi.remove(id),
   372→    onSuccess: () => {
   373→      void queryClient.invalidateQueries({ queryKey: ["journal"] });
   374→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   375→      toast.success("Entry deleted");
   376→      setEntryToDelete(null);
   377→    },
   378→    onError: (e) => toast.error(e.message || "Could not delete entry"),
   379→  });
   380→
   381→  // ── Derived data ──
   382→
   383→  const isToday = date === today;
   384→  const nextKey = addDaysToKey(date, 1);
   385→  const nextDisabled = nextKey > today;
   386→
   387→  const thisMonthPrefix = today.slice(0, 7);
   388→  const thisMonthCount = entries.filter((e) => e.date.startsWith(thisMonthPrefix)).length;
   389→  const longestStreak = longestStreakOf(entries);
   390→
   391→  const timeline = React.useMemo(() => {
   392→    const groups: { month: string; items: JournalEntry[] }[] = [];
   393→    for (const e of [...entries].sort((a, b) => b.date.localeCompare(a.date))) {
   394→      const month = monthLabel(e.date);
   395→      const last = groups[groups.length - 1];
   396→      if (last && last.month === month) last.items.push(e);
   397→      else groups.push({ month, items: [e] });
   398→    }
   399→    return groups;
   400→  }, [entries]);
   401→
   402→  // ── Handlers ──
   403→
   404→  const save = () => {
   405→    if (
   406→      !entry &&
   407→      !title.trim() &&
   408→      !content.trim() &&
   409→      !gratitude.trim() &&
   410→      mood === null
   411→    ) {
   412→      toast.error("Write something first — even one line counts");
   413→      contentRef.current?.focus();
   414→      return;
   415→    }
   416→    saveEntry.mutate();
   417→  };
   418→
   419→  const openDay = (e: JournalEntry) => {
   420→    setDate(e.date);
   421→    setExpandedId(null);
   422→    window.scrollTo({ top: 0, behavior: "smooth" });
   423→  };
   424→
   425→  const focusTodayEditor = () => {
   426→    setDate(today);
   427→    window.scrollTo({ top: 0, behavior: "smooth" });
   428→    // Wait for the form to sync to today's entry, then focus.
   429→    setTimeout(() => contentRef.current?.focus(), 120);
   430→  };
   431→
   432→  const runExport = async (
   433→    id: string,
   434→    label: string,
   435→    fn: () => Promise<void> | void
   436→  ) => {
   437→    setExporting(id);
   438→    const toastId = toast.loading(`Preparing ${label}…`);
   439→    try {
   440→      await fn();
   441→      toast.success(`${label} ready`, { id: toastId });
   442→    } catch (e) {
   443→      toast.error(e instanceof Error ? e.message : `${label} failed`, { id: toastId });
   444→    } finally {
   445→      setExporting(null);
   446→    }
   447→  };
   448→
   449→  const exportMarkdown = () =>
   450→    runExport("md", "Markdown export", async () => {
   451→      const md = await exportApi.markdown("journal");
   452→      downloadMarkdown(md, "momentum-diary.md");
   453→    });
   454→
   455→  const exportPdf = () =>
   456→    runExport("pdf", "PDF export", async () => {
   457→      if (entries.length === 0) {
   458→        toast.error("Nothing to export yet");
   459→        return;
   460→      }
   461→      const sorted = [...entries].sort((a, b) => a.date.localeCompare(b.date));
   462→      const html = sorted
   463→        .map(
   464→          (e) => `
   465→        <div>
   466→          <div style="font-size:17px; font-weight:700; margin:18px 0 4px;">${esc(formatKeyLabel(e.date))}${moodEmojiOf(e.mood) ? ` ${moodEmojiOf(e.mood)}` : ""}${e.title ? ` — ${esc(e.title)}` : ""}</div>
   467→          ${e.energy !== null ? `<div style="font-size:11px; color:#777;">Energy ${e.energy}/5</div>` : ""}
   468→          ${e.gratitude ? `<div style="font-size:13px; color:#0f766e; margin:6px 0;">♥ Grateful for: ${esc(e.gratitude)}</div>` : ""}
   469→          ${miniMarkdownToHtml(e.content)}
   470→          <hr style="border:none; border-top:1px solid #e3e3e3; margin:16px 0 4px;" />
   471→        </div>`
   472→        )
   473→        .join("");
   474→      printHtml("Daily Diary", html);
   475→    });
   476→
   477→  const exportBusy = exporting !== null;
   478→
   479→  // ── Render ──
   480→
   481→  return (
   482→    <div>
   483→      <ViewHeader
   484→        title="Daily Diary"
   485→        subtitle="Reflect, learn, grow"
   486→        actions={
   487→          <DropdownMenu>
   488→            <DropdownMenuTrigger asChild>
   489→              <Button
   490→                variant="outline"
   491→                size="icon"
   492→                className="size-11 rounded-xl"
   493→                aria-label="Export diary"
   494→                disabled={exportBusy}
   495→              >
   496→                {exportBusy ? (
   497→                  <Loader2 className="animate-spin" aria-hidden="true" />
   498→                ) : (
   499→                  <Download aria-hidden="true" />
   500→                )}
   501→              </Button>
   502→            </DropdownMenuTrigger>
   503→            <DropdownMenuContent align="end" className="w-56">
   504→              <DropdownMenuItem onSelect={exportMarkdown}>
   505→                <FileText aria-hidden="true" /> Export diary (Markdown)
   506→              </DropdownMenuItem>
   507→              <DropdownMenuItem onSelect={exportPdf}>
   508→                <Printer aria-hidden="true" /> Export diary (PDF)
   509→              </DropdownMenuItem>
   510→            </DropdownMenuContent>
   511→          </DropdownMenu>
   512→        }
   513→      />
   514→
   515→      {isLoading ? (
   516→        <DiarySkeleton />
   517→      ) : isError ? (
   518→        <QueryError onRetry={() => void refetch()} />
   519→      ) : (
   520→        <div className="space-y-6">
   521→          {entries.length > 0 && (
   522→            <div className="grid grid-cols-3 gap-2">
   523→              <StatChip value={entries.length} label="Entries" title="Total diary entries" />
   524→              <StatChip value={thisMonthCount} label="This month" title="Entries written this calendar month" />
   525→              <StatChip
   526→                value={longestStreak}
   527→                label="Best streak"
   528→                title="Longest run of consecutive days with an entry"
   529→              />
   530→            </div>
   531→          )}
   532→
   533→          {/* ── Entry editor ── */}
   534→          <Card className="rounded-2xl py-0 shadow-card">
   535→            <CardContent className="p-4 sm:p-6">
   536→              {/* Date navigation */}
   537→              <div className="flex items-center gap-1.5">
   538→                <Button
   539→                  variant="outline"
   540→                  size="icon"
   541→                  className="size-11 shrink-0 rounded-xl"
   542→                  onClick={() => setDate(addDaysToKey(date, -1))}
   543→                  aria-label="Previous day"
   544→                >
   545→                  <ChevronLeft aria-hidden="true" />
   546→                </Button>
   547→                <Popover open={calOpen} onOpenChange={setCalOpen}>
   548→                  <PopoverTrigger asChild>
   549→                    <button
   550→                      type="button"
   551→                      className="flex min-h-11 min-w-0 flex-1 items-center justify-center gap-2 rounded-xl px-2 py-2 text-center transition-colors hover:bg-muted focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/60"
   552→                      aria-label={`Choose a date — currently ${isToday ? "today" : formatKeyLabel(date)}`}
   553→                    >
   554→                      <span className="min-w-0 truncate text-sm font-semibold sm:text-base">
   555→                        <span className="sm:hidden">
   556→                          {isToday ? "Today · " : ""}
   557→                          {formatKeyLabel(date)}
   558→                        </span>
   559→                        <span className="hidden sm:inline">
   560→                          {isToday ? "Today · " : ""}
   561→                          {formatKeyLong(date)}
   562→                        </span>
   563→                      </span>
   564→                      <CalendarIcon
   565→                        className="size-4 shrink-0 text-muted-foreground"
