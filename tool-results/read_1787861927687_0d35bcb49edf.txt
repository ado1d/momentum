   123→/** Mirrors the server-side status rules for optimistic updates. */
   124→function applyStatusRules(goal: Goal, patch: GoalPatch): Goal {
   125→  const target = patch.target ?? goal.target;
   126→  const progress =
   127→    patch.progress !== undefined
   128→      ? Math.min(Math.max(patch.progress, 0), target)
   129→      : Math.min(goal.progress, target);
   130→  let status: GoalStatus = goal.status;
   131→  if (progress >= target) status = "completed";
   132→  else if (patch.status !== undefined) status = patch.status as GoalStatus;
   133→  else if (goal.status === "completed") status = "active";
   134→  return { ...goal, ...patch, target, progress, status } as Goal;
   135→}
   136→
   137→type GoalPatch = Partial<GoalInput> & { progress?: number; status?: string };

   598→          <DialogFooter className="pt-1">
   599→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   600→              Cancel
   601→            </Button>
   602→            <Button type="submit" disabled={submitting}>
   603→              {goal ? "Save changes" : "Create goal"}
   604→            </Button>
   605→          </DialogFooter>
   606→        </form>
   607→      </DialogContent>
   608→    </Dialog>
   609→  );
   610→}
   611→
   612→// ── Main view ────────────────────────────────────────────────
   613→
   614→type PeriodTab = "all" | GoalPeriod;
   615→type StatusFilter = GoalStatus | "all";
   616→
   617→export function GoalsView() {
   618→  const today = todayKey();
   619→  const queryClient = useQueryClient();
   620→
   621→  const goalsQuery = useQuery({
   622→    queryKey: ["goals"],
   623→    queryFn: () => goalsApi.list({ status: "all" }),
   624→  });
   625→  const goals = goalsQuery.data ?? [];
   626→
   627→  const [periodTab, setPeriodTab] = React.useState<PeriodTab>("all");
   628→  const [statusFilter, setStatusFilter] = React.useState<StatusFilter>("active");
   629→
   630→  const [dialogOpen, setDialogOpen] = React.useState(false);
   631→  const [editingGoal, setEditingGoal] = React.useState<Goal | null>(null);
   632→  const [goalToDelete, setGoalToDelete] = React.useState<Goal | null>(null);
   633→  const [resetConfirm, setResetConfirm] = React.useState<GoalPeriod | null>(null);
   634→
   635→  const invalidate = (keys: string[]) => {
   636→    for (const key of keys) void queryClient.invalidateQueries({ queryKey: [key] });
   637→  };
   638→
   639→  // ── Mutations ──
   640→
   641→  const incrementGoal = useMutation({
   642→    mutationFn: ({ goal, delta }: { goal: Goal; delta: number }) =>
   643→      goalsApi.increment(goal.id, delta),
   644→    onMutate: async ({ goal, delta }) => {
   645→      await queryClient.cancelQueries({ queryKey: ["goals"] });
   646→      const previous = queryClient.getQueryData<Goal[]>(["goals"]);
   647→      queryClient.setQueryData<Goal[]>(["goals"], (old) =>
   648→        (old ?? []).map((g) =>
   649→          g.id === goal.id
   650→            ? applyStatusRules(g, {
   651→                progress: Math.min(Math.max(g.progress + delta, 0), g.target),
   652→              })
   653→            : g
   654→        )
   655→      );
   656→      return { previous };
   657→    },
   658→    onError: (error, _vars, context) => {
   659→      if (context?.previous) queryClient.setQueryData(["goals"], context.previous);
   660→      toast.error(error.message || "Could not update progress");
   661→    },
   662→    onSuccess: (updated, { goal }) => {
   663→      if (updated.progress >= updated.target && goal.progress < goal.target) {
   664→        toast.success("🎉 Goal completed!");
   665→      }
   666→    },
   667→    onSettled: () => invalidate(["goals", "stats"]),
   668→  });
   669→
   670→  const patchGoal = useMutation({
   671→    mutationFn: (vars: { goal: Goal; patch: GoalPatch; successToast?: string }) =>
   672→      goalsApi.update(vars.goal.id, vars.patch),
   673→    onMutate: async ({ goal, patch }) => {
   674→      await queryClient.cancelQueries({ queryKey: ["goals"] });
   675→      const previous = queryClient.getQueryData<Goal[]>(["goals"]);
   676→      queryClient.setQueryData<Goal[]>(["goals"], (old) =>
   677→        (old ?? []).map((g) => (g.id === goal.id ? applyStatusRules(g, patch) : g))
   678→      );
   679→      return { previous };
   680→    },
   681→    onError: (error, _vars, context) => {
   682→      if (context?.previous) queryClient.setQueryData(["goals"], context.previous);
   683→      toast.error(error.message || "Could not update goal");
   684→    },
   685→    onSuccess: (_updated, vars) => {
   686→      if (vars.successToast) toast.success(vars.successToast);
   687→    },
   688→    onSettled: () => invalidate(["goals", "stats"]),
   689→  });
   690→
   691→  const saveGoal = useMutation({
   692→    mutationFn: (vars: { id?: string; input: GoalInput }) =>
   693→      vars.id ? goalsApi.update(vars.id, vars.input) : goalsApi.create(vars.input),
   694→    onSuccess: (_result, vars) => {
   695→      invalidate(["goals", "stats"]);
   696→      toast.success(vars.id ? "Goal updated" : "Goal created");
   697→      setDialogOpen(false);
   698→    },
   699→    onError: (e) => toast.error(e.message || "Could not save goal"),
   700→  });
   701→
   702→  const removeGoal = useMutation({
   703→    mutationFn: (id: string) => goalsApi.remove(id),
   704→    onSuccess: () => {
   705→      invalidate(["goals", "stats"]);
   706→      setGoalToDelete(null);
   707→      toast.success("Goal deleted");
   708→    },
   709→    onError: (e) => toast.error(e.message || "Could not delete goal"),
   710→  });
   711→
   712→  const resetPeriodProgress = useMutation({
   713→    mutationFn: async (period: GoalPeriod) => {
   714→      const targets = goals.filter(
   715→        (g) => g.period === period && g.status !== "archived" && g.progress > 0
   716→      );
   717→      await Promise.all(targets.map((g) => goalsApi.update(g.id, { progress: 0 })));
   718→      return targets.length;
   719→    },
   720→    onMutate: async (period) => {
   721→      await queryClient.cancelQueries({ queryKey: ["goals"] });
   722→      const previous = queryClient.getQueryData<Goal[]>(["goals"]);
   723→      queryClient.setQueryData<Goal[]>(["goals"], (old) =>
   724→        (old ?? []).map((g) =>
   725→          g.period === period && g.status !== "archived"
   726→            ? { ...g, progress: 0, status: "active" }
   727→            : g
   728→        )
   729→      );
   730→      return { previous };
   731→    },
   732→    onError: (error, _period, context) => {
   733→      if (context?.previous) queryClient.setQueryData(["goals"], context.previous);
   734→      toast.error(error.message || "Could not reset progress");
   735→    },
   736→    onSuccess: (count, period) => {
   737→      invalidate(["goals", "stats"]);
   738→      toast.success(
   739→        count === 0
   740→          ? `No ${PERIOD_LABEL[period].toLowerCase()} goals to reset`
   741→          : `${PERIOD_LABEL[period]} progress reset — ${count} goal${count === 1 ? "" : "s"} back to zero`
   742→      );
   743→    },
   744→  });
   745→
   746→  const addSampleGoals = useMutation({
   747→    mutationFn: async () => {
   748→      await goalsApi.create({
   749→        title: "Read 10 pages daily",
   750→        category: "learning",
   751→        period: "daily",
   752→        target: 10,
   753→        unit: "pages",
   754→        startDate: todayKey(),
   755→      });
   756→      await goalsApi.create({
   757→        title: "Complete 1 online course module",
   758→        category: "learning",
   759→        period: "weekly",
   760→        target: 3,
   761→        unit: "modules",
   762→        startDate: todayKey(),
   763→      });
   764→    },
   765→    onSuccess: () => {
   766→      invalidate(["goals", "stats"]);
   767→      toast.success("Sample goals added 🎯");
   768→    },
   769→    onError: (e) => toast.error(e.message || "Could not add sample goals"),
   770→  });
   771→
   772→  // ── Derived data ──
   773→
   774→  const activeCount = goals.filter((g) => g.status === "active").length;
   775→  const completedCount = goals.filter((g) => g.status === "completed").length;
   776→  const weekStart = weekStartKey(today);
   777→  const completedThisWeek = goals.filter(
   778→    (g) => g.status === "completed" && dateToKey(new Date(g.updatedAt)) >= weekStart
   779→  ).length;
   780→
   781→  const filtered = goals.filter(
   782→    (g) =>
   783→      (periodTab === "all" || g.period === periodTab) &&
   784→      (statusFilter === "all" || g.status === statusFilter)
   785→  );
   786→
   787→  const openNewGoal = () => {
   788→    setEditingGoal(null);
   789→    setDialogOpen(true);
   790→  };
   791→  const openEditGoal = (goal: Goal) => {
   792→    setEditingGoal(goal);
   793→    setDialogOpen(true);
   794→  };
   795→
   796→  const submitGoal = (values: GoalFormValues) => {
   797→    saveGoal.mutate({
   798→      id: editingGoal?.id,
   799→      input: {
   800→        title: values.title,
   801→        description: values.description.trim() || null,
   802→        category: values.category,
   803→        period: values.period,
   804→        target: Number.parseInt(values.target, 10),
   805→        unit: values.unit.trim() || null,
   806→        startDate: values.startDate || todayKey(),
   807→        endDate: values.endDate || null,
   808→      },
   809→    });
   810→  };
   811→
   812→  const changingId =
   813→    incrementGoal.isPending ? incrementGoal.variables?.goal.id : undefined;
   814→
   815→  return (
   816→    <div>
   817→      <ViewHeader
   818→        title="Goals"
   819→        subtitle="Daily learning goals, weekly targets, monthly ambitions"
   820→        actions={
   821→          <>
   822→            <DropdownMenu>
   823→              <DropdownMenuTrigger asChild>
   824→                <Button
   825→                  variant="outline"
   826→                  size="sm"
   827→                  className="h-11 rounded-xl"
   828→                  aria-label="Reset progress"
   829→                >
   830→                  <RotateCcw aria-hidden="true" />
   831→                  Reset
   832→                </Button>
   833→              </DropdownMenuTrigger>
   834→              <DropdownMenuContent align="end" className="w-52">
   835→                <DropdownMenuLabel>Set progress back to zero</DropdownMenuLabel>
   836→                {GOAL_PERIODS.map((p) => (
   837→                  <DropdownMenuItem key={p.value} onSelect={() => setResetConfirm(p.value)}>
   838→                    <RotateCcw aria-hidden="true" /> Reset {p.label.toLowerCase()} progress
   839→                  </DropdownMenuItem>
   840→                ))}
   841→              </DropdownMenuContent>
   842→            </DropdownMenu>
   843→            <Button size="sm" className="h-11 rounded-xl" onClick={openNewGoal}>
   844→              <Plus aria-hidden="true" />
   845→              New goal
   846→            </Button>
   847→          </>
   848→        }
   849→      />
   850→
   851→      {goalsQuery.isLoading ? (
   852→        <GoalsSkeleton />
   853→      ) : goalsQuery.isError ? (
   854→        <QueryError onRetry={() => goalsQuery.refetch()} />
   855→      ) : (
   856→        <div className="space-y-4">
   857→          {/* Summary strip */}
   858→          <div className="grid grid-cols-3 gap-2 sm:gap-3">
   859→            <StatChip label="Active" value={activeCount} hint="Goals currently in progress" />
   860→            <StatChip label="Completed" value={completedCount} hint="Completed all time" />
   861→            <StatChip
   862→              label="This week"
   863→              value={completedThisWeek}
   864→              hint="Completed this calendar week"
   865→            />
   866→          </div>
   867→
   868→          {/* Filters */}
   869→          <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   870→            <Tabs
   871→              value={periodTab}
   872→              onValueChange={(v) => setPeriodTab(v as PeriodTab)}
   873→            >
   874→              <TabsList className="h-11 rounded-xl p-1">
   875→                <TabsTrigger value="all" className="px-3 text-sm">
   876→                  All
   877→                </TabsTrigger>
   878→                {GOAL_PERIODS.map((p) => (
   879→                  <TabsTrigger key={p.value} value={p.value} className="px-3 text-sm">
   880→                    {p.label}
   881→                  </TabsTrigger>
   882→                ))}
   883→              </TabsList>
   884→            </Tabs>
   885→            <div
   886→              className="flex items-center gap-1.5"
   887→              role="group"
   888→              aria-label="Filter goals by status"
   889→            >
   890→              {STATUS_FILTERS.map((status) => (
   891→                <button
   892→                  key={status}
   893→                  type="button"
   894→                  onClick={() =>
   895→                    setStatusFilter((current) => (current === status ? "all" : status))
   896→                  }
   897→                  aria-pressed={statusFilter === status}
   898→                  className={cn(
   899→                    "h-11 rounded-full border px-3.5 text-xs font-medium capitalize transition-all active:scale-95",
   900→                    "focus-visible:outline-none focus-visible:ring-[3px] focus-visible:ring-ring/50",
   901→                    statusFilter === status
   902→                      ? "border-primary bg-primary/10 text-primary shadow-sm"
   903→                      : "border-border bg-card text-muted-foreground hover:bg-muted/50"
   904→                  )}
   905→                >
   906→                  {status}
   907→                </button>
   908→              ))}
   909→            </div>
   910→          </div>
   911→
   912→          {/* Goal list */}
   913→          {goals.length === 0 ? (
   914→            <EmptyState
   915→              icon={Target}
   916→              title="No goals yet"
   917→              description="Set a daily, weekly, or monthly target and watch the progress add up."
   918→              actionLabel="Set a goal"
   919→              onAction={openNewGoal}
   920→              secondaryLabel={addSampleGoals.isPending ? "Adding…" : "Add sample learning goals"}
   921→              onSecondary={() => addSampleGoals.mutate()}
   922→            />
   923→          ) : filtered.length === 0 ? (
   924→            <div className="rounded-2xl border border-dashed bg-muted/30 px-6 py-10 text-center">
   925→              <p className="text-sm font-medium">No goals match these filters</p>
   926→              <p className="mt-1 text-xs text-muted-foreground">
   927→                Try a different period or status.
   928→              </p>
   929→              <Button
   930→                variant="outline"
   931→                size="sm"
   932→                className="mt-4"
   933→                onClick={() => {
   934→                  setPeriodTab("all");
   935→                  setStatusFilter("all");
   936→                }}
   937→              >
   938→                Clear filters
   939→              </Button>
   940→            </div>
   941→          ) : (
   942→            <ul className="stagger-list space-y-3">
   943→              {filtered.map((goal) => (
   944→                <GoalCard
   945→                  key={goal.id}
   946→                  goal={goal}
   947→                  today={today}
   948→                  changing={changingId === goal.id}
   949→                  onIncrement={(g, delta) => incrementGoal.mutate({ goal: g, delta })}
   950→                  onEdit={openEditGoal}
   951→                  onMarkComplete={(g) =>
   952→                    patchGoal.mutate({
   953→                      goal: g,
   954→                      patch: { progress: g.target },
   955→                      successToast: "🎉 Goal completed!",
   956→                    })
   957→                  }
   958→                  onSetStatus={(g, status) =>
   959→                    patchGoal.mutate({
   960→                      goal: g,
   961→                      patch: { status },
   962→                      successToast: status === "archived" ? "Goal archived" : "Goal restored",
   963→                    })
   964→                  }
   965→                  onDelete={setGoalToDelete}
   966→                />
   967→              ))}
   968→            </ul>
   969→          )}
   970→        </div>
   971→      )}
   972→
   973→      {/* ── Dialogs ── */}
   974→      <GoalFormDialog
   975→        open={dialogOpen}
   976→        onOpenChange={setDialogOpen}
   977→        goal={editingGoal}
   978→        submitting={saveGoal.isPending}
   979→        onSubmit={submitGoal}
   980→      />
   981→
   982→      <AlertDialog
   983→        open={resetConfirm !== null}
   984→        onOpenChange={(open) => {
   985→          if (!open) setResetConfirm(null);
   986→        }}
   987→      >
   988→        <AlertDialogContent className="rounded-2xl">
   989→          <AlertDialogHeader>
   990→            <AlertDialogTitle>
   991→              Reset {resetConfirm ? PERIOD_LABEL[resetConfirm].toLowerCase() : ""} progress?
   992→            </AlertDialogTitle>
   993→            <AlertDialogDescription>
   994→              Progress for every {resetConfirm ? PERIOD_LABEL[resetConfirm].toLowerCase() : ""}{" "}
   995→              goal goes back to zero, including completed ones. Nothing is deleted and the
   996→              targets stay the same.
   997→            </AlertDialogDescription>
   998→          </AlertDialogHeader>
   999→          <AlertDialogFooter>
  1000→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1001→            <AlertDialogAction
  1002→              onClick={() => resetConfirm && resetPeriodProgress.mutate(resetConfirm)}
  1003→            >
  1004→              Reset progress
  1005→            </AlertDialogAction>
  1006→          </AlertDialogFooter>
  1007→        </AlertDialogContent>
  1008→      </AlertDialog>
  1009→
  1010→      <AlertDialog
  1011→        open={goalToDelete !== null}
  1012→        onOpenChange={(open) => {
  1013→          if (!open) setGoalToDelete(null);
  1014→        }}
  1015→      >
  1016→        <AlertDialogContent className="rounded-2xl">
  1017→          <AlertDialogHeader>
  1018→            <AlertDialogTitle>Delete goal?</AlertDialogTitle>
  1019→            <AlertDialogDescription>
  1020→              &ldquo;{goalToDelete?.title}&rdquo; and its progress will be permanently removed.
  1021→              This cannot be undone.
  1022→            </AlertDialogDescription>
  1023→          </AlertDialogHeader>
  1024→          <AlertDialogFooter>
  1025→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1026→            <AlertDialogAction
  1027→              className="bg-destructive text-white hover:bg-destructive/90"
  1028→              onClick={() => goalToDelete && removeGoal.mutate(goalToDelete.id)}
  1029→            >
  1030→              Delete
  1031→            </AlertDialogAction>
  1032→          </AlertDialogFooter>
  1033→        </AlertDialogContent>
  1034→      </AlertDialog>
  1035→    </div>
  1036→  );
  1037→}
