   169→function TodayBanner({ habits }: { habits: Habit[] }) {
   170→  const total = habits.length;
   171→  const done = habits.filter((h) => h.doneToday).length;
   172→  const pct = total > 0 ? Math.round((done / total) * 100) : 0;
   173→  const allDone = total > 0 && done === total;
   174→  const remaining = total - done;
   175→
   176→  return (
   177→    <Card className="overflow-hidden rounded-2xl border-primary/20 bg-gradient-to-br from-primary/10 via-card to-card shadow-card">
   178→      <CardContent className="flex items-center gap-4 p-4 sm:p-5">
   179→        <ProgressRing
   180→          value={pct}
   181→          size={64}
   182→          strokeWidth={7}
   183→          label={String(done)}
   184→          sublabel={`of ${total}`}
   185→          className="shrink-0 glow-ring"
   186→        />
   187→        <div className="min-w-0 flex-1">
   188→          <p className="text-sm font-semibold sm:text-base">
   189→            <span className="tabular-nums">{done}</span> of{" "}
   190→            <span className="tabular-nums">{total}</span> habits done today
   191→          </p>
   192→          <ProgressBar value={pct} className="mt-2 h-2.5" />
   193→          <p className="mt-2 text-xs text-muted-foreground">
   194→            {allDone ? (
   195→              <span className="gradient-text text-sm font-semibold">
   196→                Perfect day! 🎉
   197→              </span>
   198→            ) : remaining === 1 ? (
   199→              "Just one more — you are almost there."
   200→            ) : (
   201→              `${remaining} to go. Keep the momentum!`
   202→            )}
   203→          </p>
   204→        </div>
   205→      </CardContent>
   206→    </Card>
   207→  );
   208→}

   313→interface HabitFormValues {
   314→  name: string;
   315→  emoji: string;
   316→  color: HabitColor;
   317→  timeOfDay: TimeOfDay;
   318→  reminderTime: string;

   320→
   321→const EMPTY_HABIT: HabitFormValues = {
   322→  name: "",

   326→  reminderTime: "",

   328→
   329→interface HabitFormDialogProps {
   330→  open: boolean;
   331→  onOpenChange: (open: boolean) => void;
   332→  habit: Habit | null;
   333→  submitting: boolean;
   334→  onSubmit: (values: HabitFormValues) => void;
   335→}
   336→
   337→function HabitFormDialog({ open, onOpenChange, habit, submitting, onSubmit }: HabitFormDialogProps) {
   338→  const [values, setValues] = React.useState<HabitFormValues>(EMPTY_HABIT);
   339→
   340→  React.useEffect(() => {
   341→    if (open) {
   342→      setValues(
   343→        habit
   344→          ? {
   345→              name: habit.name,
   346→              emoji: habit.emoji,
   347→              color: habit.color,
   348→              timeOfDay: habit.timeOfDay,
   349→              reminderTime: habit.reminderTime ?? "",
   350→            }
   351→          : EMPTY_HABIT
   352→      );
   353→    }
   354→  }, [open, habit]);
   355→
   356→  const set = <K extends keyof HabitFormValues>(key: K, value: HabitFormValues[K]) =>
   357→    setValues((prev) => ({ ...prev, [key]: value }));
   358→
   359→  const submit = (e: React.FormEvent) => {
   360→    e.preventDefault();
   361→    if (!values.name.trim()) {
   362→      toast.error("Give your habit a name");
   363→      return;
   364→    }
   365→    onSubmit({
   366→      ...values,
   367→      name: values.name.trim(),
   368→      emoji: values.emoji.trim() || "✨",
   369→      reminderTime: values.reminderTime.trim(),
   370→    });
   371→  };
   372→
   373→  return (
   374→    <Dialog open={open} onOpenChange={onOpenChange}>
   375→      <DialogContent className="max-h-[88dvh] overflow-y-auto rounded-2xl sm:max-w-md">
   376→        <DialogHeader>
   377→          <DialogTitle>{habit ? "Edit habit" : "New habit"}</DialogTitle>
   378→          <DialogDescription>
   379→            {habit
   380→              ? "Update the details of this habit."
   381→              : "Pick something small enough to do every day."}
   382→          </DialogDescription>
   383→        </DialogHeader>
   384→
   385→        <form onSubmit={submit} className="space-y-5">
   386→          <div className="space-y-2">
   387→            <Label htmlFor="habit-name">Name</Label>
   388→            <Input
   389→              id="habit-name"
   390→              value={values.name}
   391→              onChange={(e) => set("name", e.target.value)}
   392→              placeholder="e.g. Drink more water"
   393→              maxLength={80}
   394→              autoFocus
   395→            />
   396→          </div>
   397→
   398→          <div className="space-y-2">
   399→            <Label>Icon</Label>
   400→            <div className="grid grid-cols-6 gap-1.5 sm:grid-cols-8">
   401→              {HABIT_EMOJIS.map((emoji) => (
   402→                <button
   403→                  key={emoji}
   404→                  type="button"
   405→                  onClick={() => set("emoji", emoji)}
   406→                  aria-label={`Icon ${emoji}`}
   407→                  aria-pressed={values.emoji === emoji}
   408→                  className={cn(
   409→                    "flex h-11 items-center justify-center rounded-xl border text-lg transition-all active:scale-90",
   410→                    "focus-visible:outline-none focus-visible:ring-[3px] focus-visible:ring-ring/50",
   411→                    values.emoji === emoji
   412→                      ? "border-primary bg-primary/10 shadow-sm"

   414→                  )}
   415→                >
   416→                  {emoji}
