   104→/** Longest run of consecutive day keys in a list of entries */
   105→function longestStreakOf(entries: JournalEntry[]): number {
   106→  const keys = Array.from(new Set(entries.map((e) => e.date))).sort();
   107→  let best = 0;
   108→  let run = 0;
   109→  let prev: string | null = null;
   110→  for (const k of keys) {
   111→    run = prev !== null && addDaysToKey(prev, 1) === k ? run + 1 : 1;
   112→    if (run > best) best = run;
   113→    prev = k;
   114→  }
   115→  return best;
   116→}

   118→function entryPreview(entry: JournalEntry): string {
   119→  if (entry.title?.trim()) return entry.title.trim();
   120→  const plain = entry.content.replace(/[#>*_`~\-]/g, "").replace(/\s+/g, " ").trim();
   121→  return plain.length > 60 ? `${plain.slice(0, 60)}…` : plain;
   122→}

   322→export function DiaryView() {
   323→  const queryClient = useQueryClient();
   324→  const today = todayKey();
   325→
   326→  const { data: entries = [], isLoading, isError, refetch } = useQuery({
   327→    queryKey: ["journal"],
   328→    queryFn: () => journalApi.list({ limit: 366 }),
   329→  });
   330→
   331→  // Shared settings query (powers the calendar's week start)
   332→  const { data: settings } = useQuery({
   333→    queryKey: ["settings"],
   334→    queryFn: settingsApi.get,
   335→  });
   336→
   337→  // ── Editor state ──
   338→  const [date, setDate] = React.useState(today);
   339→  const [title, setTitle] = React.useState("");
   340→  const [content, setContent] = React.useState("");
   341→  const [mood, setMood] = React.useState<Mood | null>(null);
   342→  const [energy, setEnergy] = React.useState(3);
   343→  const [gratitude, setGratitude] = React.useState("");
   344→  const [justSaved, setJustSaved] = React.useState(false);
   345→  const [calOpen, setCalOpen] = React.useState(false);
   346→
   347→  const [expandedId, setExpandedId] = React.useState<string | null>(null);
   348→  const [entryToDelete, setEntryToDelete] = React.useState<JournalEntry | null>(null);
   349→  const [exporting, setExporting] = React.useState<string | null>(null);
   350→
   351→  const contentRef = React.useRef<HTMLTextAreaElement>(null);
   352→
   353→  const entry = entries.find((e) => e.date === date) ?? null;
   354→
   355→  // Load the selected day's entry into the form whenever the day
   356→  // changes or a different server version arrives (e.g. after save).
   357→  const syncedSig = React.useRef("");
   358→  React.useEffect(() => {
   359→    const sig = `${date}:${entry?.updatedAt ?? "-"}`;
   360→    if (syncedSig.current !== sig) {
   361→      syncedSig.current = sig;
   362→      setTitle(entry?.title ?? "");
   363→      setContent(entry?.content ?? "");
   364→      setMood(entry?.mood ?? null);
   365→      setEnergy(entry?.energy ?? 3);
   366→      setGratitude(entry?.gratitude ?? "");
   367→    }
   368→  }, [date, entry]);
   369→
   370→  // Auto-hide the "Saved" chip
   371→  React.useEffect(() => {
   372→    if (!justSaved) return;
   373→    const id = setTimeout(() => setJustSaved(false), 3000);
   374→    return () => clearTimeout(id);
   375→  }, [justSaved]);
   376→
   377→  // ── Mutations ──
   378→
   379→  const saveEntry = useMutation({
   380→    mutationFn: () =>
   381→      journalApi.upsert({
   382→        date,
   383→        title: title.trim() || null,
   384→        content,
   385→        mood,
   386→        energy,
   387→        gratitude: gratitude.trim() || null,
   388→      }),
   389→    onSuccess: () => {
   390→      void queryClient.invalidateQueries({ queryKey: ["journal"] });
   391→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   392→      setJustSaved(true);
   393→      toast.success(entry ? "Diary updated" : "Diary saved");
   394→    },
   395→    onError: (e) => toast.error(e.message || "Could not save entry"),
   396→  });
   397→
   398→  const removeEntry = useMutation({
   399→    mutationFn: (id: string) => journalApi.remove(id),
   400→    onSuccess: () => {
   401→      void queryClient.invalidateQueries({ queryKey: ["journal"] });
   402→      void queryClient.invalidateQueries({ queryKey: ["stats"] });
   403→      toast.success("Entry deleted");
   404→      setEntryToDelete(null);
   405→    },
   406→    onError: (e) => toast.error(e.message || "Could not delete entry"),
   407→  });
   408→
   409→  // ── Derived data ──
   410→
   411→  const isToday = date === today;
   412→  const nextKey = addDaysToKey(date, 1);
   413→  const nextDisabled = nextKey > today;
   414→
   415→  const thisMonthPrefix = today.slice(0, 7);
   416→  const thisMonthCount = entries.filter((e) => e.date.startsWith(thisMonthPrefix)).length;
   417→  const longestStreak = longestStreakOf(entries);
   418→
   419→  const timeline = React.useMemo(() => {
   420→    const groups: { month: string; items: JournalEntry[] }[] = [];
   421→    for (const e of [...entries].sort((a, b) => b.date.localeCompare(a.date))) {
   422→      const month = monthLabel(e.date);
   423→      const last = groups[groups.length - 1];
   424→      if (last && last.month === month) last.items.push(e);
   425→      else groups.push({ month, items: [e] });
   426→    }
   427→    return groups;
   428→  }, [entries]);
   429→
   430→  // ── Handlers ──
   431→
   432→  const save = () => {
   433→    if (
   434→      !entry &&
   435→      !title.trim() &&
   436→      !content.trim() &&
   437→      !gratitude.trim() &&
   438→      mood === null
   439→    ) {
   440→      toast.error("Write something first — even one line counts");
   441→      contentRef.current?.focus();
   442→      return;
   443→    }
   444→    saveEntry.mutate();
   445→  };
   446→
   447→  const openDay = (e: JournalEntry) => {
   448→    setDate(e.date);
   449→    setExpandedId(null);
   450→    window.scrollTo({ top: 0, behavior: "smooth" });
   451→  };
   452→
   453→  const focusTodayEditor = () => {
   454→    setDate(today);
   455→    window.scrollTo({ top: 0, behavior: "smooth" });
   456→    // Wait for the form to sync to today's entry, then focus.
   457→    setTimeout(() => contentRef.current?.focus(), 120);
   458→  };
   459→
   460→  const runExport = async (
   461→    id: string,
   462→    label: string,
   463→    fn: () => Promise<void> | void
   464→  ) => {
   465→    setExporting(id);
   466→    const toastId = toast.loading(`Preparing ${label}…`);
   467→    try {
   468→      await fn();
   469→      toast.success(`${label} ready`, { id: toastId });
   470→    } catch (e) {
   471→      toast.error(e instanceof Error ? e.message : `${label} failed`, { id: toastId });
   472→    } finally {
   473→      setExporting(null);
   474→    }
   475→  };
   476→
   477→  const exportMarkdown = () =>
   478→    runExport("md", "Markdown export", async () => {
   479→      const md = await exportApi.markdown("journal");
   480→      downloadMarkdown(md, "momentum-diary.md");
   481→    });
   482→
   483→  const exportPdf = () =>
   484→    runExport("pdf", "PDF export", async () => {
   485→      if (entries.length === 0) {
   486→        toast.error("Nothing to export yet");
   487→        return;
   488→      }
   489→      const sorted = [...entries].sort((a, b) => a.date.localeCompare(b.date));
   490→      const html = sorted
   491→        .map(
   492→          (e) => `
   493→        <div>
   494→          <div style="font-size:17px; font-weight:700; margin:18px 0 4px;">${esc(formatKeyLabel(e.date))}${moodEmojiOf(e.mood) ? ` ${moodEmojiOf(e.mood)}` : ""}${e.title ? ` — ${esc(e.title)}` : ""}</div>
   495→          ${e.energy !== null ? `<div style="font-size:11px; color:#777;">Energy ${e.energy}/5</div>` : ""}
   496→          ${e.gratitude ? `<div style="font-size:13px; color:#0f766e; margin:6px 0;">♥ Grateful for: ${esc(e.gratitude)}</div>` : ""}
   497→          ${miniMarkdownToHtml(e.content)}
   498→          <hr style="border:none; border-top:1px solid #e3e3e3; margin:16px 0 4px;" />
   499→        </div>`
   500→        )
   501→        .join("");
   502→      printHtml("Daily Diary", html);
   503→    });
   504→
   505→  const exportBusy = exporting !== null;
   506→
   507→  // ── Render ──
   508→
   509→  return (
   510→    <div>
   511→      <ViewHeader
   512→        title="Daily Diary"
   513→        subtitle="Reflect, learn, grow"
   514→        actions={
   515→          <DropdownMenu>
   516→            <DropdownMenuTrigger asChild>
   517→              <Button
   518→                variant="outline"
   519→                size="icon"
   520→                className="size-11 rounded-xl"
   521→                aria-label="Export diary"
   522→                disabled={exportBusy}
   523→              >
   524→                {exportBusy ? (
   525→                  <Loader2 className="animate-spin" aria-hidden="true" />
   526→                ) : (
   527→                  <Download aria-hidden="true" />
   528→                )}
   529→              </Button>
   530→            </DropdownMenuTrigger>
   531→            <DropdownMenuContent align="end" className="w-56">
   532→              <DropdownMenuItem onSelect={exportMarkdown}>
   533→                <FileText aria-hidden="true" /> Export diary (Markdown)
   534→              </DropdownMenuItem>
   535→              <DropdownMenuItem onSelect={exportPdf}>
   536→                <Printer aria-hidden="true" /> Export diary (PDF)
   537→              </DropdownMenuItem>
   538→            </DropdownMenuContent>
   539→          </DropdownMenu>
   540→        }
   541→      />
   542→
   543→      {isLoading ? (
   544→        <DiarySkeleton />
   545→      ) : isError ? (
   546→        <QueryError onRetry={() => void refetch()} />
   547→      ) : (
   548→        <div className="space-y-6">
   549→          {entries.length > 0 && (
   550→            <div className="grid grid-cols-3 gap-2">
   551→              <StatChip value={entries.length} label="Entries" title="Total diary entries" />
   552→              <StatChip value={thisMonthCount} label="This month" title="Entries written this calendar month" />
   553→              <StatChip
   554→                value={longestStreak}
   555→                label="Best streak"
   556→                title="Longest run of consecutive days with an entry"
   557→              />
   558→            </div>
   559→          )}
   560→
   561→          {/* ── Entry editor ── */}
