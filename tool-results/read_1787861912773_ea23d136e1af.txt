    39→/** "Aug 27" style short label for a YYYY-MM-DD key. */
    40→function shortDate(key: string): string {
    41→  const d = keyToDate(key);
    42→  return isValid(d) ? format(d, "MMM d") : key;
    43→}
    44→
    45→/** 65 → "1h 5m", 45 → "45m", 120 → "2h" */
    46→function formatMinutes(total: number): string {
    47→  if (total <= 0) return "0m";
    48→  const h = Math.floor(total / 60);
    49→  const m = total % 60;
    50→  if (h === 0) return `${m}m`;
    51→  if (m === 0) return `${h}h`;
    52→  return `${h}h ${m}m`;
    53→}
    54→
    55→function formatCompact(n: number): string {
    56→  return Number.isInteger(n) ? String(n) : n.toFixed(1);
    57→}

    78→/** Catmull-Rom → cubic bezier smoothing, clamped to the 0..40 plot band. */
    79→function smoothPath(pts: { x: number; y: number }[]): string {
    80→  if (pts.length === 0) return "";
    81→  if (pts.length === 1) return `M ${pts[0].x.toFixed(2)} ${pts[0].y.toFixed(2)}`;
    82→  const clamp = (v: number) => Math.min(40, Math.max(0, v));
    83→  let d = `M ${pts[0].x.toFixed(2)} ${pts[0].y.toFixed(2)}`;
    84→  for (let i = 0; i < pts.length - 1; i += 1) {
    85→    const p0 = pts[i - 1] ?? pts[i];
    86→    const p1 = pts[i];
    87→    const p2 = pts[i + 1];
    88→    const p3 = pts[i + 2] ?? p2;
    89→    const c1x = p1.x + (p2.x - p0.x) / 6;
    90→    const c1y = clamp(p1.y + (p2.y - p0.y) / 6);
    91→    const c2x = p2.x - (p3.x - p1.x) / 6;
    92→    const c2y = clamp(p2.y - (p3.y - p1.y) / 6);
    93→    d += ` C ${c1x.toFixed(2)} ${c1y.toFixed(2)}, ${c2x.toFixed(2)} ${c2y.toFixed(2)}, ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`;
    94→  }
    95→  return d;
    96→}

   257→function HeatmapCard({ heatmap }: { heatmap: DayStat[] }) {
   258→  const today = todayKey();
   259→
   260→  // Pad the first column so day 1 of the data lands on its Monday-based row.
   261→  const weeks = React.useMemo(() => {
   262→    if (heatmap.length === 0) return [] as (DayStat | null)[][];
   263→    const pad = weekdayOfKey(heatmap[0].date) - 1; // Mon=1 → 0 padding
   264→    const cells: (DayStat | null)[] = [
   265→      ...Array.from({ length: pad }, () => null),
   266→      ...heatmap,
   267→    ];
   268→    const out: (DayStat | null)[][] = [];
   269→    for (let i = 0; i < cells.length; i += 7) out.push(cells.slice(i, i + 7));
   270→    return out;
   271→  }, [heatmap]);
   272→
   273→  // Tiny month label above a column whenever the month changes.
   274→  const monthLabels = React.useMemo(() => {
   275→    let last = -1;
   276→    return weeks.map((week) => {
   277→      const first = week.find((c): c is DayStat => c !== null);
   278→      if (!first) return "";
   279→      const d = keyToDate(first.date);
   280→      if (!isValid(d)) return "";
   281→      const m = d.getFullYear() * 12 + d.getMonth();
   282→      if (m !== last) {
   283→        last = m;
   284→        return format(d, "MMM");
   285→      }
   286→      return "";
   287→    });
   288→  }, [weeks]);
   289→
   290→  const { longestStreak, activeDays, avgScore } = React.useMemo(() => {
   291→    let streak = 0;
   292→    let best = 0;
   293→    let active = 0;
   294→    let sum = 0;
   295→    for (const d of heatmap) {
   296→      if (d.score > 0) {
   297→        streak += 1;
   298→        active += 1;
   299→        if (streak > best) best = streak;
   300→      } else {
   301→        streak = 0;
   302→      }
   303→      sum += d.score;
   304→    }
   305→    return {
   306→      longestStreak: best,
   307→      activeDays: active,
   308→      avgScore: heatmap.length ? Math.round(sum / heatmap.length) : 0,
   309→    };
   310→  }, [heatmap]);
   311→
   312→  const allEmpty = heatmap.length > 0 && heatmap.every((d) => d.score <= 0);
   313→  const totalDays = heatmap.length;
   314→
   315→  return (
   316→    <Card className="gap-4 rounded-2xl p-4 sm:p-5">
   317→      <CardHead
   318→        title="Activity"
   319→        subtitle="Last 12 weeks"
   320→        action={
   321→          <div
   322→            className="flex items-center gap-1 text-[10px] font-medium text-muted-foreground"
   323→            aria-hidden="true"
   324→          >
   325→            Less
   326→            {[
   327→              "bg-muted",
   328→              "bg-emerald-500/25",
   329→              "bg-emerald-500/45",
   330→              "bg-emerald-500/70",
   331→              "bg-emerald-500",
   332→            ].map((c) => (
   333→              <span key={c} className={cn("size-2.5 rounded-[3px]", c)} />
   334→            ))}
   335→            More
   336→          </div>
   337→        }
   338→      />
   339→
   340→      <div className="flex flex-col gap-5 sm:flex-row sm:items-center sm:justify-between">
   341→        <div
   342→          className="no-scrollbar overflow-x-auto"
   343→          role="img"
   344→          aria-label={`Activity heatmap for the last 12 weeks — ${activeDays} of ${totalDays} days active, average score ${avgScore}.`}
   345→        >
   346→          <div className="flex w-max gap-1">
   347→            {/* Weekday hints (M / W / F) */}
   348→            <div
   349→              className="mr-1 flex flex-col gap-1 pt-4 text-[9px] font-medium leading-none text-muted-foreground/60"
   350→              aria-hidden="true"
   351→            >
   352→              {WEEKDAY_LABELS.map((label, i) => (
   353→                <span key={i} className="flex h-[11px] items-center sm:h-3">
   354→                  {label}
   355→                </span>
   356→              ))}
   357→            </div>
   358→
   359→            {weeks.map((week, wi) => (
   360→              <div key={wi} className="flex flex-col gap-1">
   361→                <span className="flex h-3 items-center text-[9px] font-medium leading-none text-muted-foreground/70">
   362→                  {monthLabels[wi]}
   363→                </span>
   364→                {week.map((cell, ci) =>
   365→                  cell ? (
   366→                    <span
   367→                      key={cell.date}
   368→                      title={`${shortDate(cell.date)} · score ${cell.score} · ${cell.todosCompleted} ${cell.todosCompleted === 1 ? "task" : "tasks"}, ${cell.habitsCompleted} ${cell.habitsCompleted === 1 ? "habit" : "habits"}, ${cell.routineCompleted} routine`}
   369→                      className={cn(
   370→                        "size-[11px] rounded-[3px] sm:size-3",
   371→                        heatLevelClass(cell.score),
   372→                        cell.date === today &&
   373→                          "ring-2 ring-primary ring-offset-1 ring-offset-card"
   374→                      )}
   375→                    />
   376→                  ) : (
   377→                    <span
   378→                      key={`pad-${wi}-${ci}`}
   379→                      className="size-[11px] sm:size-3"
   380→                      aria-hidden="true"
   381→                    />
   382→                  )
   383→                )}
   384→              </div>
   385→            ))}
   386→          </div>
   387→        </div>
   388→
   389→        <div className="flex flex-wrap items-center gap-2 sm:max-w-[13rem] sm:flex-col sm:items-stretch sm:gap-2.5">
   390→          <StatChip
   391→            icon={Flame}
   392→            iconClassName="text-orange-500"
   393→            label="Longest streak"
   394→            value={`${longestStreak} ${longestStreak === 1 ? "day" : "days"}`}
   395→          />
   396→          <StatChip
   397→            icon={CalendarCheck}
   398→            iconClassName="text-teal-600 dark:text-teal-400"
   399→            label="Active days"
   400→            value={`${activeDays} / ${totalDays}`}
   401→          />
   402→          <StatChip
   403→            icon={Activity}
   404→            iconClassName="text-emerald-600 dark:text-emerald-400"
   405→            label="Avg score"
   406→            value={String(avgScore)}
   407→          />
   408→        </div>
   409→      </div>
   410→
   411→      {allEmpty && (
   412→        <p className="text-xs text-muted-foreground">
   413→          Start checking things off to fill your heatmap.
   414→        </p>
   415→      )}
   416→    </Card>
   417→  );
   418→}

   420→// ── Tasks trend (pure inline SVG line/area chart) ────────────

   422→function TasksTrendCard({ trend }: { trend: TrendPoint[] }) {
   423→  const n = trend.length;
   424→
   425→  const { max, total, best, avg } = React.useMemo(() => {
   426→    const counts = trend.map((p) => p.count);
   427→    const t = counts.reduce((a, b) => a + b, 0);
   428→    // Round odd maxima up so the mid gridline/label stays an integer.
   429→    const m = Math.max(1, ...counts);
   430→    return {
   431→      max: m >= 3 && m % 2 === 1 ? m + 1 : m,
   432→      total: t,
   433→      best: counts.length ? Math.max(...counts) : 0,
   434→      avg: counts.length ? t / counts.length : 0,
   435→    };
   436→  }, [trend]);
   437→
   438→  // Points live in a 100×40 viewBox; the SVG stretches with
   439→  // preserveAspectRatio="none" and strokes use vector-effect so they
   440→  // stay crisp. Text (axis labels) is rendered as HTML so it never distorts.
   441→  const points = React.useMemo(
   442→    () =>
   443→      trend.map((p, i) => ({
   444→        ...p,
   445→        x: ((i + 0.5) * 100) / n,
   446→        y: 40 - (p.count / max) * 40,
   447→        yPct: (1 - p.count / max) * 100,
   448→      })),
   449→    [trend, n, max]
   450→  );
   451→
   452→  const linePath = React.useMemo(() => smoothPath(points), [points]);
   453→  const areaPath = React.useMemo(() => {
   454→    if (points.length < 2) return "";
   455→    const last = points[points.length - 1];
   456→    const first = points[0];
   457→    return `${linePath} L ${last.x.toFixed(2)} 40 L ${first.x.toFixed(2)} 40 Z`;
   458→  }, [linePath, points]);
   459→
   460→  const xLabelIdx = React.useMemo(() => {
   461→    if (n === 0) return [] as number[];
   462→    const raw = [0, 0.25, 0.5, 0.75, 1].map((f) => Math.round((n - 1) * f));
   463→    return [...new Set(raw)];
   464→  }, [n]);
   465→
   466→  const mid = max / 2;
   467→  const showMid = max >= 2; // max=1 has no integer midpoint
   468→
   469→  return (
   470→    <Card className="gap-4 rounded-2xl p-4 sm:p-5">
   471→      <CardHead title="Tasks completed" subtitle="Last 30 days" />
   472→
   473→      <div
   474→        className="flex gap-2"
   475→        role="img"
   476→        aria-label={`Tasks completed per day over the last ${n} days — total ${total}, best day ${best}, average ${avg.toFixed(1)} per day.`}
   477→      >
   478→        {/* Y-axis labels (HTML so they stay crisp) */}
   479→        <div className="relative w-7 shrink-0 text-right text-[9px] font-medium leading-none tabular-nums text-muted-foreground">
   480→          <span className="absolute inset-x-0 top-0 -translate-y-1/2">
   481→            {formatCompact(max)}
   482→          </span>
   483→          {showMid && (
   484→            <span className="absolute inset-x-0 top-1/2 -translate-y-1/2">
   485→              {formatCompact(mid)}
   486→            </span>
   487→          )}
   488→          <span className="absolute inset-x-0 top-full -translate-y-1/2">0</span>
   489→        </div>
   490→
   491→        {/* Plot area */}
   492→        <div className="relative h-36 min-w-0 flex-1">
   493→          <svg
   494→            className="absolute inset-0 h-full w-full text-primary"
   495→            viewBox="0 0 100 40"
   496→            preserveAspectRatio="none"
   497→            aria-hidden="true"
   498→          >
   499→            <defs>
   500→              <linearGradient id="insights-trend-fill" x1="0" y1="0" x2="0" y2="1">
   501→                <stop offset="0%" stopColor="currentColor" stopOpacity="0.16" />
   502→                <stop offset="100%" stopColor="currentColor" stopOpacity="0.02" />
   503→              </linearGradient>
   504→            </defs>
   505→            {/* Gridlines */}
   506→            <line
   507→              x1="0"
   508→              y1="0"
   509→              x2="100"
   510→              y2="0"
   511→              strokeWidth="1"
   512→              vectorEffect="non-scaling-stroke"
   513→              className="stroke-muted-foreground/15"
   514→            />
   515→            {showMid && (
   516→              <line
   517→                x1="0"
   518→                y1="20"
   519→                x2="100"
   520→                y2="20"
   521→                strokeWidth="1"
   522→                vectorEffect="non-scaling-stroke"
   523→                className="stroke-muted-foreground/10"
   524→              />
   525→            )}
   526→            <line
   527→              x1="0"
   528→              y1="40"
   529→              x2="100"
   530→              y2="40"
   531→              strokeWidth="1"
   532→              vectorEffect="non-scaling-stroke"
   533→              className="stroke-muted-foreground/15"
   534→            />
   535→            {/* Area + line */}
   536→            {areaPath && <path d={areaPath} fill="url(#insights-trend-fill)" />}
   537→            {linePath && (
   538→              <path
   539→                d={linePath}
   540→                fill="none"
   541→                strokeWidth="2"
   542→                strokeLinecap="round"
   543→                strokeLinejoin="round"
   544→                vectorEffect="non-scaling-stroke"
   545→                className="stroke-primary"
   546→              />
   547→            )}
   548→          </svg>
   549→
   550→          {/* Hover zones: one per day — tooltip + guideline + dot */}
   551→          <div
   552→            className="absolute inset-0 grid"
   553→            style={{ gridTemplateColumns: `repeat(${n}, 1fr)` }}
   554→            aria-hidden="true"
   555→          >
   556→            {points.map((p) => (
   557→              <div
   558→                key={p.date}
   559→                className="group relative"

   686→// ── Mood distribution ────────────────────────────────────────
   687→
   688→function MoodCard({ distribution }: { distribution: InsightsData["moodDistribution"] }) {
   689→  const total = distribution.reduce((a, d) => a + d.count, 0);
   690→
   691→  // Render in a fixed best→rough order, only moods that were recorded.
   692→  const rows = MOODS.map((m) => {
   693→    const entry = distribution.find((d) => d.mood === m.value);
   694→    return entry ? { ...m, count: entry.count } : null;
   695→  }).filter((r): r is { value: Mood; label: string; emoji: string; count: number } => r !== null);
   696→
   697→  const pctOf = (count: number) => (total > 0 ? Math.round((count / total) * 100) : 0);
   698→
   699→  return (
   700→    <Card className="gap-4 rounded-2xl p-4 sm:p-5">
   701→      <CardHead title="Mood over time" subtitle="From your diary entries" />
   702→
   703→      <div
   704→        className="flex h-3 w-full overflow-hidden rounded-full bg-muted"
   705→        role="img"
   706→        aria-label={`Mood distribution across ${total} diary ${total === 1 ? "entry" : "entries"}: ${rows
   707→          .map((r) => `${r.label} ${pctOf(r.count)}%`)
   708→          .join(", ")}.`}
   709→      >
   710→        {rows.map((r) => (
   711→          <div
   712→            key={r.value}
   713→            className={moodBarStyles[r.value]}
   714→            style={{ width: `${pctOf(r.count)}%` }}
   715→            title={`${r.label} · ${r.count} ${r.count === 1 ? "entry" : "entries"} · ${pctOf(r.count)}%`}
   716→          />
   717→        ))}
   718→      </div>
   719→
   720→      <div className="grid grid-cols-1 gap-x-6 gap-y-2 sm:grid-cols-2">
   721→        {rows.map((r) => (
   722→          <div key={r.value} className="flex items-center gap-2 text-sm">
   723→            <span
   724→              className={cn("size-2.5 shrink-0 rounded-full", moodBarStyles[r.value])}
   725→              aria-hidden="true"
   726→            />
   727→            <span className="text-base leading-none" aria-hidden="true">
   728→              {r.emoji}
   729→            </span>
   730→            <span className="font-medium">{r.label}</span>
   731→            <span className="ml-auto text-xs tabular-nums text-muted-foreground">
   732→              {r.count} · {pctOf(r.count)}%
   733→            </span>
   734→          </div>
   735→        ))}
   736→      </div>
   737→    </Card>
   738→  );
   739→}

   743→function FocusCard({ focus }: { focus: InsightsData["focus"] }) {
   744→  const delta = focus.weekMinutes - focus.lastWeekMinutes;
   745→  const noFocus = focus.weekMinutes === 0;
   746→
   747→  return (
   748→    <Card className="gap-4 rounded-2xl p-4 sm:p-5">
   749→      <CardHead title="Focus time" subtitle="Your recent deep-work sessions" />
   750→
   751→      <div className="flex flex-wrap items-end justify-between gap-x-4 gap-y-2">
   752→        <div>
   753→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   754→            This week
   755→          </p>
   756→          <p className="mt-1 text-3xl font-bold leading-none tabular-nums">
   757→            {formatMinutes(focus.weekMinutes)}
   758→          </p>

   761→          <p className="max-w-[16rem] text-xs text-muted-foreground">
   762→            Use the Focus timer to track deep work.
   763→          </p>
   764→        ) : delta === 0 ? (
   765→          <span className="inline-flex items-center gap-1.5 rounded-full bg-muted px-2.5 py-1 text-xs font-medium text-muted-foreground">
   766→            Same as last week
   767→          </span>
   768→        ) : (
   769→          <span
   770→            className={cn(
   771→              "inline-flex items-center gap-1 rounded-full px-2.5 py-1 text-xs font-semibold tabular-nums",
   772→              delta > 0
   773→                ? "bg-emerald-500/15 text-emerald-700 dark:text-emerald-300"
   774→                : "bg-red-500/15 text-red-700 dark:text-red-300"
   775→            )}
   776→          >
   777→            {delta > 0 ? (
   778→              <TrendingUp className="size-3.5" aria-hidden="true" />
   779→            ) : (
   780→              <TrendingDown className="size-3.5" aria-hidden="true" />
   781→            )}
   782→            {delta > 0 ? "+" : "-"}
   783→            {formatMinutes(Math.abs(delta))} vs last week
   784→          </span>
   785→        )}
   786→      </div>
   787→
   788→      <Separator />
   789→
   790→      <div className="grid grid-cols-2 gap-3">

   794→            {formatMinutes(focus.todayMinutes)}
   795→          </p>
   796→        </div>
   797→        <div className="rounded-xl bg-muted/40 px-3.5 py-3">
   798→          <p className="text-xs font-medium text-muted-foreground">Avg session</p>
   799→          <p className="mt-1 text-xl font-bold leading-none tabular-nums">
   800→            {formatMinutes(focus.avgSessionMinutes)}
   801→          </p>
   802→        </div>
   803→      </div>
   804→    </Card>
   805→  );
   806→}
