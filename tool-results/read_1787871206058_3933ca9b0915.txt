   121→/** Mirrors the server-side status rules for optimistic updates. */
   122→function applyStatusRules(goal: Goal, patch: GoalPatch): Goal {
   123→  const target = patch.target ?? goal.target;
   124→  const progress =
   125→    patch.progress !== undefined
   126→      ? Math.min(Math.max(patch.progress, 0), target)
   127→      : Math.min(goal.progress, target);
   128→  let status: GoalStatus = goal.status;
   129→  if (progress >= target) status = "completed";
   130→  else if (patch.status !== undefined) status = patch.status as GoalStatus;
   131→  else if (goal.status === "completed") status = "active";
   132→  return { ...goal, ...patch, target, progress, status } as Goal;
   133→}
   134→
   135→type GoalPatch = Partial<GoalInput> & { progress?: number; status?: string };

   597→                min={values.startDate || undefined}
   598→                className="h-11 rounded-xl"
   599→              />
   600→            </div>
   601→          </div>
   602→
   603→          <DialogFooter className="pt-1">
   604→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   605→              Cancel
   606→            </Button>
   607→            <Button type="submit" disabled={submitting}>
   608→              {goal ? "Save changes" : "Create goal"}
   609→            </Button>
   610→          </DialogFooter>
   611→        </form>
   612→      </DialogContent>
   613→    </Dialog>
   614→  );
   615→}
   616→
   617→// ── Main view ────────────────────────────────────────────────
   618→
   619→type PeriodTab = "all" | GoalPeriod;
   620→type StatusFilter = GoalStatus | "all";
   621→
   622→export function GoalsView() {
   623→  const today = todayKey();
   624→  const queryClient = useQueryClient();
   625→
   626→  const goalsQuery = useQuery({
   627→    queryKey: ["goals"],
   628→    queryFn: () => goalsApi.list({ status: "all" }),
   629→  });
   630→  const goals = goalsQuery.data ?? [];
   631→
   632→  const [periodTab, setPeriodTab] = React.useState<PeriodTab>("all");
   633→  const [statusFilter, setStatusFilter] = React.useState<StatusFilter>("active");
   634→
   635→  const [dialogOpen, setDialogOpen] = React.useState(false);
   636→  const [editingGoal, setEditingGoal] = React.useState<Goal | null>(null);
   637→  const [goalToDelete, setGoalToDelete] = React.useState<Goal | null>(null);
   638→  const [resetConfirm, setResetConfirm] = React.useState<GoalPeriod | null>(null);
   639→  /** Goal that just hit 100% — drives the one-shot .celebrate pulse. */
   640→  const [celebrateId, setCelebrateId] = React.useState<string | null>(null);
   641→
   642→  const invalidate = (keys: string[]) => {
   643→    for (const key of keys) void queryClient.invalidateQueries({ queryKey: [key] });
   644→  };
   645→
   646→  /** Fire the celebration pulse when a goal crosses its target. */
   647→  const celebrateCompletion = (goal: Goal, nextProgress: number) => {
   648→    if (nextProgress >= goal.target && goal.progress < goal.target) {
   649→      setCelebrateId(goal.id);
   650→      window.setTimeout(() => setCelebrateId(null), 1000);
   651→    }
   652→  };
   653→
   654→  // ── Mutations ──
   655→
   656→  const incrementGoal = useMutation({
   657→    mutationFn: ({ goal, delta }: { goal: Goal; delta: number }) =>
   658→      goalsApi.increment(goal.id, delta),
   659→    onMutate: async ({ goal, delta }) => {
   660→      await queryClient.cancelQueries({ queryKey: ["goals"] });
   661→      const previous = queryClient.getQueryData<Goal[]>(["goals"]);
   662→      const nextProgress = Math.min(Math.max(goal.progress + delta, 0), goal.target);
   663→      celebrateCompletion(goal, nextProgress);
   664→      queryClient.setQueryData<Goal[]>(["goals"], (old) =>
   665→        (old ?? []).map((g) =>
   666→          g.id === goal.id
   667→            ? applyStatusRules(g, { progress: nextProgress })
   668→            : g
   669→        )
   670→      );
   671→      return { previous };
   672→    },
   673→    onError: (error, _vars, context) => {
   674→      if (context?.previous) queryClient.setQueryData(["goals"], context.previous);
   675→      toast.error(error.message || "Could not update progress");
   676→    },
   677→    onSuccess: (updated, { goal }) => {
   678→      if (updated.progress >= updated.target && goal.progress < goal.target) {
   679→        toast.success("🎉 Goal completed!");
   680→      }
   681→    },
   682→    onSettled: () => invalidate(["goals", "stats"]),
   683→  });
   684→
   685→  const patchGoal = useMutation({
   686→    mutationFn: (vars: { goal: Goal; patch: GoalPatch; successToast?: string }) =>
   687→      goalsApi.update(vars.goal.id, vars.patch),
   688→    onMutate: async ({ goal, patch }) => {
   689→      await queryClient.cancelQueries({ queryKey: ["goals"] });
   690→      const previous = queryClient.getQueryData<Goal[]>(["goals"]);
   691→      if (patch.progress !== undefined) celebrateCompletion(goal, patch.progress);
   692→      queryClient.setQueryData<Goal[]>(["goals"], (old) =>
   693→        (old ?? []).map((g) => (g.id === goal.id ? applyStatusRules(g, patch) : g))
   694→      );
   695→      return { previous };
   696→    },
   697→    onError: (error, _vars, context) => {
   698→      if (context?.previous) queryClient.setQueryData(["goals"], context.previous);
   699→      toast.error(error.message || "Could not update goal");
   700→    },
   701→    onSuccess: (_updated, vars) => {
   702→      if (vars.successToast) toast.success(vars.successToast);
   703→    },
   704→    onSettled: () => invalidate(["goals", "stats"]),
   705→  });
   706→
   707→  const saveGoal = useMutation({
   708→    mutationFn: (vars: { id?: string; input: GoalInput }) =>
   709→      vars.id ? goalsApi.update(vars.id, vars.input) : goalsApi.create(vars.input),
   710→    onSuccess: (_result, vars) => {
   711→      invalidate(["goals", "stats"]);
   712→      toast.success(vars.id ? "Goal updated" : "Goal created");
   713→      setDialogOpen(false);
   714→    },
   715→    onError: (e) => toast.error(e.message || "Could not save goal"),
   716→  });
   717→
   718→  const removeGoal = useMutation({
   719→    mutationFn: (id: string) => goalsApi.remove(id),
   720→    onSuccess: () => {
   721→      invalidate(["goals", "stats"]);
   722→      setGoalToDelete(null);
   723→      toast.success("Goal deleted");
   724→    },
   725→    onError: (e) => toast.error(e.message || "Could not delete goal"),
   726→  });
   727→
   728→  const resetPeriodProgress = useMutation({
   729→    mutationFn: async (period: GoalPeriod) => {
   730→      const targets = goals.filter(
   731→        (g) => g.period === period && g.status !== "archived" && g.progress > 0
   732→      );
   733→      await Promise.all(targets.map((g) => goalsApi.update(g.id, { progress: 0 })));
   734→      return targets.length;
   735→    },
   736→    onMutate: async (period) => {
   737→      await queryClient.cancelQueries({ queryKey: ["goals"] });
   738→      const previous = queryClient.getQueryData<Goal[]>(["goals"]);
   739→      queryClient.setQueryData<Goal[]>(["goals"], (old) =>
   740→        (old ?? []).map((g) =>
   741→          g.period === period && g.status !== "archived"
   742→            ? { ...g, progress: 0, status: "active" }
   743→            : g
   744→        )
   745→      );
   746→      return { previous };
   747→    },
   748→    onError: (error, _period, context) => {
   749→      if (context?.previous) queryClient.setQueryData(["goals"], context.previous);
   750→      toast.error(error.message || "Could not reset progress");
   751→    },
   752→    onSuccess: (count, period) => {
   753→      invalidate(["goals", "stats"]);
   754→      toast.success(
   755→        count === 0
   756→          ? `No ${PERIOD_LABEL[period].toLowerCase()} goals to reset`
   757→          : `${PERIOD_LABEL[period]} progress reset — ${count} goal${count === 1 ? "" : "s"} back to zero`
   758→      );
   759→    },
   760→  });
   761→
   762→  const addSampleGoals = useMutation({
   763→    mutationFn: async () => {
   764→      await goalsApi.create({
   765→        title: "Read 10 pages daily",
   766→        category: "learning",
   767→        period: "daily",
   768→        target: 10,
   769→        unit: "pages",
   770→        startDate: todayKey(),
   771→      });
   772→      await goalsApi.create({
   773→        title: "Complete 1 online course module",
   774→        category: "learning",
   775→        period: "weekly",
   776→        target: 3,
   777→        unit: "modules",
   778→        startDate: todayKey(),
   779→      });
   780→    },
   781→    onSuccess: () => {
   782→      invalidate(["goals", "stats"]);
   783→      toast.success("Sample goals added 🎯");
   784→    },
   785→    onError: (e) => toast.error(e.message || "Could not add sample goals"),
   786→  });
   787→
   788→  // ── Derived data ──
   789→
   790→  const activeCount = goals.filter((g) => g.status === "active").length;
   791→  const completedCount = goals.filter((g) => g.status === "completed").length;
   792→  const weekStart = weekStartKey(today);
   793→  const completedThisWeek = goals.filter(
   794→    (g) => g.status === "completed" && dateToKey(new Date(g.updatedAt)) >= weekStart
   795→  ).length;
   796→
   797→  const filtered = goals.filter(
   798→    (g) =>
   799→      (periodTab === "all" || g.period === periodTab) &&
   800→      (statusFilter === "all" || g.status === statusFilter)
   801→  );
   802→
   803→  const openNewGoal = () => {
   804→    setEditingGoal(null);
   805→    setDialogOpen(true);
   806→  };
   807→  const openEditGoal = (goal: Goal) => {
   808→    setEditingGoal(goal);
   809→    setDialogOpen(true);
   810→  };
   811→
   812→  const submitGoal = (values: GoalFormValues) => {
   813→    saveGoal.mutate({
   814→      id: editingGoal?.id,
   815→      input: {
   816→        title: values.title,
   817→        description: values.description.trim() || null,
   818→        category: values.category,
   819→        period: values.period,
   820→        target: Number.parseInt(values.target, 10),
   821→        unit: values.unit.trim() || null,
   822→        startDate: values.startDate || todayKey(),
   823→        endDate: values.endDate || null,
   824→      },
   825→    });
   826→  };
   827→
   828→  const changingId =
   829→    incrementGoal.isPending ? incrementGoal.variables?.goal.id : undefined;
   830→
   831→  return (
   832→    <div>
   833→      <ViewHeader
   834→        title="Goals"
   835→        subtitle="Daily learning goals, weekly targets, monthly ambitions"
   836→        actions={
   837→          <>
   838→            <DropdownMenu>
   839→              <DropdownMenuTrigger asChild>
   840→                <Button
   841→                  variant="outline"
   842→                  size="sm"
   843→                  className="h-11 rounded-xl"
   844→                  aria-label="Reset progress"
   845→                >
   846→                  <RotateCcw aria-hidden="true" />
   847→                  Reset
   848→                </Button>
   849→              </DropdownMenuTrigger>
   850→              <DropdownMenuContent align="end" className="w-52">
   851→                <DropdownMenuLabel>Set progress back to zero</DropdownMenuLabel>
   852→                {GOAL_PERIODS.map((p) => (
   853→                  <DropdownMenuItem key={p.value} onSelect={() => setResetConfirm(p.value)}>
   854→                    <RotateCcw aria-hidden="true" /> Reset {p.label.toLowerCase()} progress
   855→                  </DropdownMenuItem>
   856→                ))}
   857→              </DropdownMenuContent>
   858→            </DropdownMenu>
   859→            <Button size="sm" className="h-11 rounded-xl" onClick={openNewGoal}>
   860→              <Plus aria-hidden="true" />
   861→              New goal
   862→            </Button>
   863→          </>
   864→        }
   865→      />
   866→
   867→      {goalsQuery.isLoading ? (
   868→        <GoalsSkeleton />
   869→      ) : goalsQuery.isError ? (
   870→        <QueryError onRetry={() => goalsQuery.refetch()} />
   871→      ) : (
   872→        <div className="space-y-4">
   873→          {/* Summary strip */}
   874→          <div className="grid grid-cols-3 gap-2 sm:gap-3">
   875→            <StatChip label="Active" value={activeCount} hint="Goals currently in progress" />
   876→            <StatChip label="Completed" value={completedCount} hint="Completed all time" />
   877→            <StatChip
   878→              label="This week"
   879→              value={completedThisWeek}
   880→              hint="Completed this calendar week"
   881→            />
   882→          </div>
   883→
   884→          {/* Filters */}
   885→          <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   886→            <Tabs
   887→              value={periodTab}
   888→              onValueChange={(v) => setPeriodTab(v as PeriodTab)}
   889→            >
   890→              <TabsList className="h-11 rounded-xl p-1">
   891→                <TabsTrigger value="all" className="px-3 text-sm">
   892→                  All
   893→                </TabsTrigger>
   894→                {GOAL_PERIODS.map((p) => (
   895→                  <TabsTrigger key={p.value} value={p.value} className="px-3 text-sm">
   896→                    {p.label}
   897→                  </TabsTrigger>
   898→                ))}
   899→              </TabsList>
   900→            </Tabs>
   901→            <div
   902→              className="flex items-center gap-1.5"
   903→              role="group"
   904→              aria-label="Filter goals by status"
   905→            >
   906→              {STATUS_FILTERS.map((status) => (
   907→                <button
   908→                  key={status}
   909→                  type="button"
   910→                  onClick={() =>
   911→                    setStatusFilter((current) => (current === status ? "all" : status))
   912→                  }
   913→                  aria-pressed={statusFilter === status}
   914→                  className={cn(
   915→                    "h-11 rounded-full border px-3.5 text-xs font-medium capitalize transition-all active:scale-95",
   916→                    "focus-visible:outline-none focus-visible:ring-[3px] focus-visible:ring-ring/50",
   917→                    statusFilter === status
   918→                      ? "border-primary bg-primary/10 text-primary shadow-sm"
   919→                      : "border-border bg-card text-muted-foreground hover:bg-muted/50"
   920→                  )}
   921→                >
   922→                  {status}
   923→                </button>
   924→              ))}
   925→            </div>
   926→          </div>
   927→
   928→          {/* Goal list */}
   929→          {goals.length === 0 ? (
   930→            <EmptyState
   931→              icon={Target}
   932→              title="No goals yet"
   933→              description="Set a daily, weekly, or monthly target and watch the progress add up."
   934→              actionLabel="Set a goal"
   935→              onAction={openNewGoal}
   936→              secondaryLabel={addSampleGoals.isPending ? "Adding…" : "Add sample learning goals"}
   937→              onSecondary={() => addSampleGoals.mutate()}
   938→            />
   939→          ) : filtered.length === 0 ? (
   940→            <div className="rounded-2xl border border-dashed bg-muted/30 px-6 py-10 text-center">
   941→              <p className="text-sm font-medium">No goals match these filters</p>
   942→              <p className="mt-1 text-xs text-muted-foreground">
   943→                Try a different period or status.
   944→              </p>
   945→              <Button
   946→                variant="outline"
   947→                size="sm"
   948→                className="mt-4"
   949→                onClick={() => {
   950→                  setPeriodTab("all");
   951→                  setStatusFilter("all");
   952→                }}
   953→              >
   954→                Clear filters
   955→              </Button>
   956→            </div>
   957→          ) : (
   958→            <ul className="stagger-list space-y-3">
   959→              {filtered.map((goal) => (
   960→                <GoalCard
   961→                  key={goal.id}
   962→                  goal={goal}
   963→                  today={today}
   964→                  changing={changingId === goal.id}
   965→                  celebrating={celebrateId === goal.id}
   966→                  onIncrement={(g, delta) => incrementGoal.mutate({ goal: g, delta })}
   967→                  onEdit={openEditGoal}
   968→                  onMarkComplete={(g) =>
   969→                    patchGoal.mutate({
   970→                      goal: g,
   971→                      patch: { progress: g.target },
   972→                      successToast: "🎉 Goal completed!",
   973→                    })
   974→                  }
   975→                  onSetStatus={(g, status) =>
   976→                    patchGoal.mutate({
   977→                      goal: g,
   978→                      patch: { status },
   979→                      successToast: status === "archived" ? "Goal archived" : "Goal restored",
   980→                    })
   981→                  }
   982→                  onDelete={setGoalToDelete}
   983→                />
   984→              ))}
   985→            </ul>
   986→          )}
   987→        </div>
   988→      )}
   989→
   990→      {/* ── Dialogs ── */}
   991→      <GoalFormDialog
   992→        open={dialogOpen}
   993→        onOpenChange={setDialogOpen}
   994→        goal={editingGoal}
   995→        submitting={saveGoal.isPending}
   996→        onSubmit={submitGoal}
   997→      />
   998→
   999→      <AlertDialog
  1000→        open={resetConfirm !== null}
  1001→        onOpenChange={(open) => {
  1002→          if (!open) setResetConfirm(null);
  1003→        }}
  1004→      >
  1005→        <AlertDialogContent className="rounded-2xl">
  1006→          <AlertDialogHeader>
  1007→            <AlertDialogTitle>
  1008→              Reset {resetConfirm ? PERIOD_LABEL[resetConfirm].toLowerCase() : ""} progress?
  1009→            </AlertDialogTitle>
  1010→            <AlertDialogDescription>
  1011→              Progress for every {resetConfirm ? PERIOD_LABEL[resetConfirm].toLowerCase() : ""}{" "}
  1012→              goal goes back to zero, including completed ones. Nothing is deleted and the
  1013→              targets stay the same.
  1014→            </AlertDialogDescription>
  1015→          </AlertDialogHeader>
  1016→          <AlertDialogFooter>
  1017→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1018→            <AlertDialogAction
  1019→              onClick={() => resetConfirm && resetPeriodProgress.mutate(resetConfirm)}
  1020→            >
  1021→              Reset progress
  1022→            </AlertDialogAction>
  1023→          </AlertDialogFooter>
  1024→        </AlertDialogContent>
  1025→      </AlertDialog>
  1026→
  1027→      <AlertDialog
  1028→        open={goalToDelete !== null}
  1029→        onOpenChange={(open) => {
  1030→          if (!open) setGoalToDelete(null);
  1031→        }}
  1032→      >
  1033→        <AlertDialogContent className="rounded-2xl">
  1034→          <AlertDialogHeader>
  1035→            <AlertDialogTitle>Delete goal?</AlertDialogTitle>
  1036→            <AlertDialogDescription>
  1037→              &ldquo;{goalToDelete?.title}&rdquo; and its progress will be permanently removed.
  1038→              This cannot be undone.
  1039→            </AlertDialogDescription>
  1040→          </AlertDialogHeader>
  1041→          <AlertDialogFooter>
  1042→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1043→            <AlertDialogAction
  1044→              className="bg-destructive text-white hover:bg-destructive/90"
  1045→              onClick={() => goalToDelete && removeGoal.mutate(goalToDelete.id)}
  1046→            >
  1047→              Delete
  1048→            </AlertDialogAction>
  1049→          </AlertDialogFooter>
  1050→        </AlertDialogContent>
  1051→      </AlertDialog>
  1052→    </div>
  1053→  );
  1054→}
